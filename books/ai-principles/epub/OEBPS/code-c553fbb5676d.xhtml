"""A finite RNN-T-style alignment lattice, without claiming to train RNN-T."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import torch
from torch.nn import functional as F
 
 
ROOT = Path(__file__).resolve().parents[2]
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    out = args.out if args.out.is_absolute() else ROOT / args.out
    if out.exists():
        raise FileExistsError(f"preserve existing transducer probe: {out}")
    out.parent.mkdir(parents=True, exist_ok=True)
    # Two acoustic steps t=0,1; text A,B. At (t,u), probabilities are
    # blank/A/B. A label advances u without advancing t, blank advances t.
    raw = torch.tensor([
        [[1.3, 0.8, -0.2], [0.7, -0.3, 1.1], [1.5, 0.1, -0.4]],
        [[0.9, 1.2, -0.5], [0.6, 0.1, 1.4], [1.0, 0.2, 0.0]],
    ], dtype=torch.float64, requires_grad=True)
    probs = F.softmax(raw, dim=-1)
    target_ids = (1, 2)
    T, U = 2, 2
 
    def enumerate_paths(t: int, u: int, product: torch.Tensor,
                        names: list[str], found: list) -> None:
        if t == T:
            if u == U:
                found.append((product, names))
            return
        enumerate_paths(t + 1, u, product * probs[t, u, 0],
                        names + [f"blank at ({t},{u})"], found)
        if u < U:
            symbol = "A" if u == 0 else "B"
            enumerate_paths(t, u + 1, product * probs[t, u, target_ids[u]],
                            names + [f"emit {symbol} at ({t},{u})"], found)
 
    paths = []
    enumerate_paths(0, 0, probs.new_ones(()), [], paths)
    direct = torch.stack([weight for weight, _ in paths]).sum()
 
    alpha = [[None for _ in range(U + 1)] for _ in range(T + 1)]
    alpha[0][0] = probs.new_ones(())
    for t in range(T + 1):
        for u in range(U + 1):
            here = alpha[t][u]
            if here is None or t == T:
                continue
            to_blank = here * probs[t, u, 0]
            alpha[t + 1][u] = to_blank if alpha[t + 1][u] is None else alpha[t + 1][u] + to_blank
            if u < U:
                to_label = here * probs[t, u, target_ids[u]]
                alpha[t][u + 1] = to_label if alpha[t][u + 1] is None else alpha[t][u + 1] + to_label
    dynamic = alpha[T][U]
    grad_direct = torch.autograd.grad(-direct.log(), raw, retain_graph=True)[0]
    grad_dynamic = torch.autograd.grad(-dynamic.log(), raw)[0]
    if abs(float((direct - dynamic).detach())) > 1e-14:
        raise AssertionError("direct/dynamic transducer path sums differ")
    report = {
        "identity": "finite math probe only; probabilities supplied by a fixed table, not a trained RNN-T",
        "acoustic_steps": T,
        "target_labels": ["A", "B"],
        "node_distribution_order": ["blank", "A", "B"],
        "node_probabilities": probs.detach().tolist(),
        "paths": [{"edges": names, "probability": float(weight.detach())}
                  for weight, names in paths],
        "path_count": len(paths),
        "direct_total_probability": float(direct.detach()),
        "dynamic_total_probability": float(dynamic.detach()),
        "negative_log_probability": float(-dynamic.detach().log()),
        "maximum_logit_gradient_difference": float(
            (grad_direct - grad_dynamic).abs().max()),
    }
    out.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"path_count": len(paths),
                      "probability": report["direct_total_probability"],
                      "gradient_difference": report["maximum_logit_gradient_difference"]},
                     ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
