"""加载三条已选路线，追踪事前指定验证句和固定规则的测试样本。"""
 
from __future__ import annotations
 
import hashlib
import json
from pathlib import Path
 
import torch
from torch.nn import functional as F
 
from prepare_tatoeba_fra import tokens
from translate_tatoeba import ARMS, EOS, Translator, load_data, source_for_arm, trim_eos
 
 
WORK = Path(__file__).resolve().parents[1]
DATA = WORK / "data" / "tatoeba_fra_eng"
RESULT = WORK / "results" / "tatoeba_translation_examples.json"
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def chosen_models(source_size: int, target_size: int, device: torch.device, manifest_hash: str):
    models = {}
    for arm in ARMS:
        path = WORK / "results" / f"tatoeba_{arm}_best.pt"
        saved = torch.load(path, map_location="cpu", weights_only=True)
        if saved["identity"]["manifest_sha256"] != manifest_hash:
            raise ValueError("模型权重和当前句对来源不配套")
        model = Translator(source_size, target_size, arm == "soft_read").to(device)
        model.load_state_dict(saved["state_dict"])
        model.eval()
        models[arm] = (model, sha256(path), saved["selected_epoch"])
    return models
 
 
@torch.no_grad()
def decode_one(model: Translator, source: torch.Tensor, source_len: torch.Tensor, vocab: list[str]) -> list[str]:
    return [vocab[index] for index in trim_eos(model.greedy(source, source_len, 20)[0].cpu().tolist())]
 
 
def main() -> None:
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    manifest, parts, source_vocab, target_vocab = load_data(device)
    manifest_hash = sha256(DATA / "manifest.json")
    models = chosen_models(len(source_vocab), len(target_vocab), device, manifest_hash)
    validation = parts["validation"]
    fixed_row = manifest["alignment_visualization_validation_source_row"]
    matches = [i for i, item in enumerate(validation["metadata"]) if item["source_row_zero_based"] == fixed_row]
    if len(matches) != 1:
        raise ValueError("固定验证句身份不符")
    row = matches[0]
    item = validation["metadata"][row]
    target_len = int(validation["target_len"][row])
    source_len = int(validation["source_len"][row])
    chosen = {
        "source_row_zero_based": fixed_row,
        "english_original": item["english_original"],
        "french_original": item["french_original"],
        "attribution": item["attribution"],
        "source_tokens_mapped": [source_vocab[int(i)] for i in validation["source_ids"][row, :source_len].cpu()],
        "teacher_decoder_input": [target_vocab[int(i)] for i in validation["target_input"][row, :target_len].cpu()],
        "teacher_target": [target_vocab[int(i)] for i in validation["target_output"][row, :target_len].cpu()],
        "predicted_greedy": {},
    }
    for arm, (model, _, _) in models.items():
        src = source_for_arm(validation, arm)[row:row + 1]
        chosen["predicted_greedy"][arm] = decode_one(
            model, src, validation["source_len"][row:row + 1], target_vocab
        )
    model = models["soft_read"][0]
    with torch.no_grad():
        logits, weights = model(
            validation["source_ids"][row:row + 1],
            validation["source_len"][row:row + 1],
            validation["target_input"][row:row + 1, :target_len].long(),
            capture_weights=True,
        )
        probabilities = F.softmax(logits[0], dim=-1)
        targets = validation["target_output"][row, :target_len].long()
        target_probs = probabilities.gather(1, targets[:, None]).squeeze(1)
    chosen["soft_teacher_target_probabilities"] = target_probs.cpu().tolist()
    chosen["soft_teacher_attention_row_sums"] = weights[0, :, :source_len].sum(dim=-1).cpu().tolist()
 
    test = parts["test"]
    groups = {}
    for index, meta in enumerate(test["metadata"]):
        key = meta["source_group_sha256"]
        groups.setdefault(key, {"row": index, "english": meta["english_original"],
                                "refs": [], "length": int(test["source_len"][index]) - 1})
        groups[key]["refs"].append({"french": meta["french_original"], "attribution": meta["attribution"]})
    selected = []
    for name, bounds in (("2-5", (2, 5)), ("6-10", (6, 10)), ("11-18", (11, 18))):
        choices = [
            (key, info) for key, info in groups.items() if bounds[0] <= info["length"] <= bounds[1]
        ]
        for key, info in sorted(choices, key=lambda pair: pair[0])[:3]:
            row = info["row"]
            generated = {
                arm: decode_one(model, source_for_arm(test, arm)[row:row + 1],
                                test["source_len"][row:row + 1], target_vocab)
                for arm, (model, _, _) in models.items()
            }
            selected.append({"length_bin": name, "source_group_sha256": key,
                             "source": info["english"], "first_reference": info["refs"][0],
                             "reference_count": len(info["refs"]), "generated": generated})
    output = {
        "identity": "load-only qualitative audit; selected validation sentence was fixed before training, 3 test source groups per length bin chosen by source hash, not by model outputs",
        "manifest_sha256": manifest_hash,
        "checkpoints": {
            arm: {"sha256": value[1], "selected_epoch": value[2]} for arm, value in models.items()
        },
        "preselected_validation_sentence": chosen,
        "test_hash_selected_examples": selected,
        "limits": "Teacher-forced source weights and probabilities are not manual word-alignment labels; greedy decoding may differ, references are incomplete and often non-unique.",
    }
    RESULT.parent.mkdir(exist_ok=True)
    RESULT.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"validation": chosen, "first_test_examples": selected[:3]}, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
