"""Check the local old/new gradient relation at the fixed C27 math model."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
import torch.nn.functional as F
from tokenizers import Tokenizer
 
import arithmetic_curriculum as math_task
import code_expression_lab as code_task
 
ROOT = math_task.ROOT
PARENT = ROOT / "work/runs/c27_digit_addition_long_trial/best.pt"
EXPECTED = "02be020308fabdc0b7ea05f800ac424d8c062e525130f91cae9aa30825464115"
 
 
def main(out: Path) -> None:
    if out.exists():
        raise FileExistsError(out)
    if math_task.file_sha(PARENT) != EXPECTED:
        raise RuntimeError("Parent math weights changed")
    device = math_task.choose_device()
    model, _, identity = math_task.start_model(device)
    saved = torch.load(PARENT, map_location="cpu", weights_only=True)
    if saved["start_model_identity"] != identity:
        raise RuntimeError("Parent identity changed")
    model.load_state_dict(saved["model"])
    model.eval()
    tokenizer = Tokenizer.from_file(str(math_task.TOKENIZER_FILE))
    rng = np.random.default_rng(4604)
    math_train = math_task.load_split("train")
    code_train = code_task.load_split("train")
    old_batch = [math_train[int(i)] for i in rng.choice(len(math_train), 16, replace=False)]
    new_batch = [code_train[int(i)] for i in rng.choice(len(code_train), 16, replace=False)]
 
    def loss(rows, batch_fn):
        ids, labels, real = batch_fn(rows, tokenizer, device)
        # Use FP32 throughout this Taylor check; BF16 rounding can dominate a tiny step.
        with torch.autocast(device_type=device.type, enabled=False):
            logits = model(ids, real)
        return F.cross_entropy(logits.float().transpose(1, 2), labels,
                               ignore_index=-100, reduction="sum") / (labels != -100).sum()
 
    old_loss = loss(old_batch, math_task.batch)
    model.zero_grad(set_to_none=True)
    old_loss.backward()
    grad_old = [p.grad.detach().float().clone() for p in model.parameters()]
    new_loss = loss(new_batch, code_task.batch)
    model.zero_grad(set_to_none=True)
    new_loss.backward()
    grad_new = [p.grad.detach().float().clone() for p in model.parameters()]
    dot = float(sum((a*b).sum() for a,b in zip(grad_old,grad_new)))
    norm_old = float(torch.sqrt(sum((a*a).sum() for a in grad_old)))
    norm_new = float(torch.sqrt(sum((b*b).sum() for b in grad_new)))
    rate = 1e-6
    predicted_change = -rate * dot
    with torch.no_grad():
        for p,g in zip(model.parameters(), grad_new):
            p.add_(g, alpha=-rate)
    actual_old = float(loss(old_batch, math_task.batch).detach())
    result = {
        "scope": "One fixed old-math/new-code train minibatch at C27 math checkpoint; FP32 eval-mode gradients and one tiny plain SGD step, not the actual BF16/AdamW branch update",
        "parent_sha256": EXPECTED, "old_batch_n": 16, "new_batch_n": 16, "rng_seed": 4604,
        "old_math_loss_before": float(old_loss.detach()), "new_code_loss_before": float(new_loss.detach()),
        "gradient_dot": dot, "old_gradient_norm": norm_old, "new_gradient_norm": norm_new,
        "gradient_cosine": dot / (norm_old * norm_new),
        "tiny_sgd_step": rate, "first_order_predicted_old_loss_change": predicted_change,
        "actual_same_old_batch_loss_after": actual_old,
        "actual_old_loss_change": actual_old - float(old_loss.detach()),
        "code_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    }
    math_task.write_json(out, result)
    print(json.dumps(result, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", type=Path, required=True)
    args = ap.parse_args()
    main(ROOT / args.out)
