"""Verify DDPM Gaussian identities and distinguish conditional from marginal fields."""
 
from __future__ import annotations
 
import argparse
import json
import math
from pathlib import Path
 
import numpy as np
 
 
ROOT = Path(__file__).resolve().parents[2]
TIMESTEPS = 300
BETA = np.linspace(0.0001, 0.05, TIMESTEPS, dtype=np.float64)
ALPHA = 1 - BETA
BAR_ALPHA = np.concatenate(([1.0], np.cumprod(ALPHA)))
 
 
def posterior(t: int, x_t: float, x_0: float):
    beta, alpha = BETA[t - 1], ALPHA[t - 1]
    abar, abar_before = BAR_ALPHA[t], BAR_ALPHA[t - 1]
    variance = beta * (1 - abar_before) / (1 - abar)
    mean_direct = (
        math.sqrt(abar_before) * beta / (1 - abar) * x_0 +
        math.sqrt(alpha) * (1 - abar_before) / (1 - abar) * x_t
    )
    epsilon = (x_t - math.sqrt(abar) * x_0) / math.sqrt(1 - abar)
    mean_noise = (x_t - beta / math.sqrt(1 - abar) * epsilon) / math.sqrt(alpha)
    return mean_direct, mean_noise, variance
 
 
def normal_density(x: float, mean: float, variance: float) -> float:
    return math.exp(-0.5 * (x - mean) ** 2 / variance) / math.sqrt(
        2 * math.pi * variance
    )
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c31_gaussian_path_probe_300.json"))
    args = parser.parse_args()
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError(f"preserve existing C31 probe: {out}")
    # Conditioning a joint pair of Gaussian forward states on x_t and x_0.
    rows = []
    for t in (1, 50, 200, 300):
        x_0, x_t = 0.8, -0.2
        direct, via_noise, variance = posterior(t, x_t, x_0)
        if abs(direct - via_noise) > 1e-11:
            raise RuntimeError("posterior mean parameterizations disagree")
        rows.append({
            "step": t, "bar_alpha": float(BAR_ALPHA[t]),
            "signal_coefficient": float(math.sqrt(BAR_ALPHA[t])),
            "noise_coefficient": float(math.sqrt(1 - BAR_ALPHA[t])),
            "posterior_mean_direct": direct,
            "posterior_mean_from_noise": via_noise,
            "posterior_variance": variance,
        })
    # A two-point image-data analogue: X0 is -1 or +1, each with probability 1/2.
    # The clean point is unknown when only one noisy observation x_t is given.
    t, x_t = 100, 0.3
    a, sigma = math.sqrt(BAR_ALPHA[t]), math.sqrt(1 - BAR_ALPHA[t])
    clean = (-1.0, 1.0)
    densities = [normal_density(x_t, a * x0, sigma ** 2) for x0 in clean]
    weights = [density / sum(densities) for density in densities]
    realized_epsilons = [(x_t - a * x0) / sigma for x0 in clean]
    conditional_scores = [-epsilon / sigma for epsilon in realized_epsilons]
    marginal_score = sum(
        weights[i] * conditional_scores[i] for i in range(2)
    )
    expected_epsilon = sum(
        weights[i] * realized_epsilons[i] for i in range(2)
    )
    if abs(marginal_score + expected_epsilon / sigma) > 1e-12:
        raise RuntimeError("mixture marginal score mismatch")
    # With different possible source/target endpoints, a time point need not
    # determine its particular paired velocity.
    crossing_time = 1 / 3
    pair_a = (-1.0, 2.0)
    pair_b = (1.0, -2.0)
    same_position_a = (1 - crossing_time) * pair_a[0] + crossing_time * pair_a[1]
    same_position_b = (1 - crossing_time) * pair_b[0] + crossing_time * pair_b[1]
    if abs(same_position_a - same_position_b) > 1e-12:
        raise RuntimeError("flow crossing arithmetic wrong")
    result = {
        "scope": "closed-form Gaussian/finite-mixture mathematics, no trained image model",
        "schedule": {
            "steps": TIMESTEPS, "beta_first": float(BETA[0]),
            "beta_last": float(BETA[-1]),
            "bar_alpha_at_terminal": float(BAR_ALPHA[-1]),
            "forward": "x_t=sqrt(bar_alpha_t)*x_0+sqrt(1-bar_alpha_t)*epsilon",
        },
        "posterior_checks": rows,
        "two_clean_point_noisy_observation": {
            "step": t, "observed_x_t": x_t,
            "possible_clean_x_0": list(clean),
            "posterior_weights_for_possible_x_0": weights,
            "realized_epsilon_if_clean_x_0_known": realized_epsilons,
            "conditional_gaussian_scores": conditional_scores,
            "unconditional_mixture_score": marginal_score,
            "posterior_mean_epsilon": expected_epsilon,
            "minus_posterior_mean_epsilon_over_sigma": -expected_epsilon / sigma,
        },
        "flow_pair_crossing": {
            "time": crossing_time, "position_from_pair_a": same_position_a,
            "position_from_pair_b": same_position_b,
            "paired_velocity_a": pair_a[1] - pair_a[0],
            "paired_velocity_b": pair_b[1] - pair_b[0],
            "mean_velocity_if_pair_is_equiprobable_at_crossing": 0.0,
            "scope": "four discrete endpoints illustrate paired target ambiguity, not a smooth ODE theorem",
        },
    }
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n",
                   encoding="utf-8")
    print("bar_alpha_T", BAR_ALPHA[-1],
          "mixture_score", marginal_score,
          "crossing paired velocities", 3, -3)
 
 
if __name__ == "__main__":
    main()
