"""Separate ODE solver step error from any neural vector-field learning error."""
 
from __future__ import annotations
 
import argparse
import json
import math
from pathlib import Path
 
 
ROOT = Path(__file__).resolve().parents[2]
 
 
def velocity(x: float, t: float) -> float:
    return t * x
 
 
def integrate(steps: int, method: str) -> float:
    x = 1.0
    dt = 1 / steps
    for index in range(steps):
        t = index * dt
        first = velocity(x, t)
        if method == "euler":
            x += dt * first
        elif method == "heun":
            proposed = x + dt * first
            x += dt * (first + velocity(proposed, t + dt)) / 2
        else:
            raise ValueError(method)
    return x
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c31_ode_step_probe.json"))
    args = parser.parse_args()
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError("preserve existing ODE check")
    exact = math.exp(0.5)  # d(log x)/dt=t, x(0)=1.
    rows = []
    for steps in (4, 16, 64):
        for method in ("euler", "heun"):
            value = integrate(steps, method)
            rows.append({
                "steps": steps, "method": method,
                "function_evaluations": steps if method == "euler" else 2 * steps,
                "value_at_1": value,
                "absolute_error": abs(value - exact),
            })
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps({
        "scope": "analytic one-dimensional ODE numerical-method check, not a trained image flow",
        "ode": "dx/dt=t*x, x(0)=1",
        "exact_x1": exact,
        "rows": rows,
        "limitation": "Heun better on this smooth known field; no automatic ranking for a learned high-dimensional field or fixed wall-clock budget",
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("exact", exact, "Euler4", rows[0]["value_at_1"],
          "Heun4", rows[1]["value_at_1"])
 
 
if __name__ == "__main__":
    main()
