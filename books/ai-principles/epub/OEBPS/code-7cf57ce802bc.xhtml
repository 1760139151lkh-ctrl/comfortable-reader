"""Small-context DPO on genuine CoVal majority comparisons from one C25 base.
 
Only 58 of 1,078 prompts survive the declared complete-context filter. This
experiment teaches the objective and its data boundary; it cannot represent
general user preference, image aesthetics, or the full CoVal distribution.
"""
 
from __future__ import annotations
 
import argparse
import copy
import hashlib
import json
import math
from pathlib import Path
 
import numpy as np
import torch
import torch.nn.functional as F
from tokenizers import Tokenizer
 
from arithmetic_curriculum import (
    ROOT, TOKENIZER_FILE, BOS, EOS, PAD, autocast_for,
    choose_device, file_sha, start_model, write_json,
)
 
 
DATA = ROOT / "work/data/coval_eec757/short_dpo_v2"
SEED = 20260924
BETA = 0.1
 
 
def manifest() -> tuple[dict, str]:
    path = DATA / "manifest.json"
    m = json.loads(path.read_text(encoding="utf-8"))
    if file_sha(TOKENIZER_FILE) != m["tokenizer_sha256"]:
        raise RuntimeError("tokenizer changed")
    for split, expected in m["files"].items():
        if file_sha(DATA / f"{split}.json") != expected:
            raise RuntimeError(f"{split} changed")
    return m, file_sha(path)
 
 
def split_pairs(name: str) -> list[dict]:
    rows = json.loads((DATA / f"{name}.json").read_text(encoding="utf-8"))
    return [{
        "prompt_id": row["prompt_id"],
        "prompt": row["prompt"],
        **pair,
    } for row in rows for pair in row["pairs"]]
 
 
def encoded_stream(prompt: str, response: str, tokenizer: Tokenizer) -> tuple[list[int], list[int], list[bool]]:
    p = tokenizer.encode(prompt, add_special_tokens=False).ids
    a = tokenizer.encode(response, add_special_tokens=False).ids
    stream = [BOS] + p + a + [EOS]
    if not a or len(stream) - 1 > 128:
        raise RuntimeError("derived pair exceeds complete context")
    inputs = stream[:-1]
    labels = stream[1:]
    labels[:len(p)] = [-100] * len(p)
    return inputs, labels, [True] * len(inputs)
 
 
def batch(sequences: list[tuple[list[int], list[int], list[bool]]],
          device: torch.device):
    width = max(len(seq[0]) for seq in sequences)
    ids = torch.full((len(sequences), width), PAD, dtype=torch.long, device=device)
    labels = torch.full((len(sequences), width), -100, dtype=torch.long, device=device)
    real = torch.zeros((len(sequences), width), dtype=torch.bool, device=device)
    for i, (input_ids, target_ids, visible) in enumerate(sequences):
        size = len(input_ids)
        ids[i, :size] = torch.tensor(input_ids, dtype=torch.long, device=device)
        labels[i, :size] = torch.tensor(target_ids, dtype=torch.long, device=device)
        real[i, :size] = torch.tensor(visible, dtype=torch.bool, device=device)
    return ids, labels, real
 
 
def log_probability(model, tensors: tuple[torch.Tensor, ...],
                    device: torch.device):
    ids, labels, real = tensors
    with autocast_for(device):
        logits = model(ids, real)
    logp = F.log_softmax(logits.float(), dim=-1)
    selected = logp.gather(2, labels.clamp(min=0).unsqueeze(-1)).squeeze(-1)
    return (selected * (labels != -100)).sum(dim=-1)
 
 
def tensors_for(pairs: list[dict], tokenizer: Tokenizer, device: torch.device):
    chosen = batch([encoded_stream(row["prompt"], row["chosen"], tokenizer)
                    for row in pairs], device)
    rejected = batch([encoded_stream(row["prompt"], row["rejected"], tokenizer)
                      for row in pairs], device)
    return chosen, rejected
 
 
@torch.no_grad()
def measure(model, reference, pairs: list[dict], tokenizer: Tokenizer,
            device: torch.device) -> dict:
    model.eval()
    reference.eval()
    losses = []
    majority_correct = 0
    probabilities = []
    votes = 0
    sample = []
    for offset in range(0, len(pairs), 16):
        part = pairs[offset:offset + 16]
        chosen, rejected = tensors_for(part, tokenizer, device)
        delta = (
            log_probability(model, chosen, device)
            - log_probability(model, rejected, device)
            - log_probability(reference, chosen, device)
            + log_probability(reference, rejected, device)
        )
        probs = torch.sigmoid(BETA * delta).tolist()
        for row, p in zip(part, probs):
            n_a, n_b = row["world_votes_for_pair_AB_order"]
            n = n_a + n_b
            majority_fraction = max(n_a, n_b) / n
            clipped = min(max(p, 1e-8), 1 - 1e-8)
            losses.append(
                -majority_fraction * math.log(clipped)
                -(1-majority_fraction) * math.log(1-clipped)
            )
            majority_correct += int(p > 0.5)
            probabilities.append(p)
            votes += n
            if len(sample) < 3:
                sample.append({
                    "prompt_id": row["prompt_id"],
                    "majority_fraction": majority_fraction,
                    "predicted_probability_of_majority_chosen": p,
                    "response_text_omitted": True,
                })
    return {
        "pairs": len(pairs), "source_world_non_tie_votes": votes,
        "mean_vote_fraction_cross_entropy": sum(losses) / len(losses),
        "majority_direction_correct": majority_correct,
        "mean_probability_of_majority_chosen": sum(probabilities) / len(probabilities),
        "sample_without_text": sample,
    }
 
 
def train(out_dir: Path, steps: int, learning_rate: float,
          soft_votes: bool = False, validate_every: int = 20) -> None:
    if out_dir.exists() and any(out_dir.iterdir()):
        raise FileExistsError(f"preserve existing run: {out_dir}")
    out_dir.mkdir(parents=True)
    m, m_sha = manifest()
    train_pairs = split_pairs("train")
    validation = split_pairs("validation")
    tokenizer = Tokenizer.from_file(str(TOKENIZER_FILE))
    device = choose_device()
    model, cfg, identity = start_model(device)
    reference = copy.deepcopy(model).eval()
    for p in reference.parameters():
        p.requires_grad_(False)
    torch.manual_seed(SEED)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED)
        torch.cuda.reset_peak_memory_stats()
    rng = np.random.default_rng(SEED)
    optimizer = torch.optim.AdamW(model.parameters(), lr=learning_rate, weight_decay=0.0)
    baseline = measure(model, reference, validation, tokenizer, device)
    best_value = baseline["mean_vote_fraction_cross_entropy"]
    best_step = 0
    best_state = copy.deepcopy(model.state_dict())
    history = [{"step": 0, "validation": baseline}]
    first_update = None
    for step in range(1, steps + 1):
        indices = rng.integers(0, len(train_pairs), size=8)
        part = [train_pairs[int(i)] for i in indices]
        chosen, rejected = tensors_for(part, tokenizer, device)
        model.eval()  # no dropout or mode change between score and update
        with torch.no_grad():
            reference_chosen = log_probability(reference, chosen, device)
            reference_rejected = log_probability(reference, rejected, device)
        chosen_logp = log_probability(model, chosen, device)
        rejected_logp = log_probability(model, rejected, device)
        logits = BETA * (
            (chosen_logp - rejected_logp)
            - (reference_chosen - reference_rejected)
        )
        if soft_votes:
            majority_share = torch.tensor([
                max(row["world_votes_for_pair_AB_order"])
                / sum(row["world_votes_for_pair_AB_order"])
                for row in part
            ], dtype=logits.dtype, device=device)
            loss = -(
                majority_share * F.logsigmoid(logits)
                + (1 - majority_share) * F.logsigmoid(-logits)
            ).mean()
        else:
            loss = -F.logsigmoid(logits).mean()
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        norm = float(torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0))
        before = model.output.weight.detach().clone() if step == 1 else None
        optimizer.step()
        if step == 1:
            first_update = {
                "pairs": len(part), "reference_relative_margin_at_start": logits.detach().tolist(),
                "loss": float(loss.detach()), "gradient_norm_before_clip": norm,
                "output_weight_change_norm": float(torch.linalg.vector_norm(
                    model.output.weight.detach() - before
                )),
            }
        if step % validate_every == 0:
            current = measure(model, reference, validation, tokenizer, device)
            history.append({"step": step, "validation": current})
            if current["mean_vote_fraction_cross_entropy"] < best_value:
                best_value = current["mean_vote_fraction_cross_entropy"]
                best_step = step
                best_state = copy.deepcopy(model.state_dict())
            print("step", step, "val_vote_loss",
                  round(current["mean_vote_fraction_cross_entropy"], 4),
                  "majority", current["majority_direction_correct"],
                  "/", current["pairs"], flush=True)
    model.load_state_dict(best_state)
    checkpoint = {
        "model": {name: value.detach().cpu().clone()
                  for name, value in model.state_dict().items()},
        "selected_step": best_step,
        "short_manifest_sha256": m_sha,
        "base_identity": identity,
        "config": cfg.__dict__,
    }
    checkpoint_path = out_dir / "best.pt"
    torch.save(checkpoint, checkpoint_path)
    report = {
        "scope": "C25 parent LM DPO on only short CoVal world-majority pairs after generic-developer prompt filter; no general aesthetics claim or free generation success",
        "short_manifest_sha256": m_sha,
        "data_counts": {k: m[k] for k in ("prompt_counts", "pair_counts")},
        "base_identity": identity,
        "training": {
            "steps": steps, "batch_pairs": 8,
            "learning_rate": learning_rate, "beta": BETA,
            "hard_majority_labels": not soft_votes,
            "soft_world_vote_fractions": soft_votes,
            "validate_every": validate_every,
            "validation_selection": "lowest vote-fraction cross-entropy at fixed intervals",
        },
        "first_update": first_update,
        "history": history,
        "selected_step": best_step,
        "selected_validation": measure(model, reference, validation, tokenizer, device),
        "checkpoint": str(checkpoint_path.relative_to(ROOT)).replace("\\", "/"),
        "checkpoint_sha256": file_sha(checkpoint_path),
        "peak_gpu_allocated_bytes": (
            torch.cuda.max_memory_allocated() if device.type == "cuda" else None
        ),
        "test_opened": False,
    }
    write_json(out_dir / "train.json", report)
    print("selected", best_step, best_value)
 
 
def evaluate(run_dir: Path, out: Path) -> None:
    if out.exists():
        raise FileExistsError(f"test already opened: {out}")
    _, m_sha = manifest()
    report = json.loads((run_dir / "train.json").read_text(encoding="utf-8"))
    checkpoint_path = ROOT / report["checkpoint"]
    if (report["short_manifest_sha256"] != m_sha
            or report["test_opened"] is not False
            or file_sha(checkpoint_path) != report["checkpoint_sha256"]):
        raise RuntimeError("training identity changed")
    device = choose_device()
    baseline, _, identity = start_model(device)
    if identity != report["base_identity"]:
        raise RuntimeError("C25 parent changed")
    saved = torch.load(checkpoint_path, map_location="cpu", weights_only=True)
    if (saved["short_manifest_sha256"] != m_sha
            or saved["base_identity"] != identity
            or saved["selected_step"] != report["selected_step"]):
        raise RuntimeError("selected checkpoint metadata changed")
    trained = copy.deepcopy(baseline)
    trained.load_state_dict(saved["model"])
    tokenizer = Tokenizer.from_file(str(TOKENIZER_FILE))
    # The only code path that opens source-derived heldout pair labels.
    test = split_pairs("test")
    output = {
        "scope": "first heldout short-pair test after validation checkpoint choice; real human aggregate majority, highly selected context subset, no new annotator/user evaluation",
        "short_manifest_sha256": m_sha,
        "base_identity": identity,
        "selected_checkpoint_sha256": report["checkpoint_sha256"],
        "selected_step": report["selected_step"],
        "baseline": measure(baseline, baseline, test, tokenizer, device),
        "trained": measure(trained, baseline, test, tokenizer, device),
    }
    write_json(out, output)
    print("test vote loss", output["baseline"]["mean_vote_fraction_cross_entropy"],
          "to", output["trained"]["mean_vote_fraction_cross_entropy"],
          "majority", output["trained"]["majority_direction_correct"],
          "/", output["trained"]["pairs"])
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("mode", choices=("train", "evaluate"))
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c28_coval_short_dpo_trial"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c28_coval_short_dpo_test.json"))
    parser.add_argument("--steps", type=int, default=200)
    parser.add_argument("--learning-rate", type=float, default=0.00005)
    parser.add_argument("--soft-votes", action="store_true")
    parser.add_argument("--validate-every", type=int, default=20)
    args = parser.parse_args()
    if args.mode == "train":
        train(ROOT / args.run_dir, args.steps, args.learning_rate,
              soft_votes=args.soft_votes, validate_every=args.validate_every)
    else:
        evaluate(ROOT / args.run_dir, ROOT / args.out)
 
 
if __name__ == "__main__":
    main()
