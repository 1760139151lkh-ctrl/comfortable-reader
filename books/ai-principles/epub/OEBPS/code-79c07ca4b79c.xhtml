"""One real C38 training trace: input visibility, target origin, and two losses.
 
This is a gradient/role probe. It does not execute an optimizer update or
claim that full-record language modeling is invalid; Toolformer used one.
"""
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import torch
from torch.nn import functional as F
from tokenizers import Tokenizer
 
from c38_tool_data import encode
from c38_tool_data_v3 import segments_v3, tokenized_v3
from c38_tool_train import DATA, ROOT, TOKENIZER, sha, parent_model
 
V3 = ROOT / "data/c38_tool_trajectories_v3"
MODEL = ROOT / "runs/c38_tool_aligned_copy_2400/best.pt"
 
 
def main(out):
    if out.exists():
        raise RuntimeError("output exists")
    manifest = json.loads((V3 / "manifest.json").read_text(encoding="utf-8"))
    if sha(V3 / "train.jsonl") != manifest["files"]["train.jsonl"]:
        raise RuntimeError("source drift")
    row = next(json.loads(text) for text in (V3 / "train.jsonl").read_text(encoding="utf-8").splitlines()
               if '"kind": "title"' in text and '"author_simulated_catalog_snapshot"' not in text)
    tok = Tokenizer.from_file(str(TOKENIZER))
    encoded = tokenized_v3(tok, row)
    chunks = [encode(tok, text) for text in segments_v3(row)]
    seq = [1] + sum(chunks, [])
    full_targets = seq[1:] + [-100] * (128 - len(seq) + 1)
    assistant = torch.tensor(encoded["labels"], dtype=torch.long)[None]
    whole = torch.tensor(full_targets, dtype=torch.long)[None]
    ids = torch.tensor(encoded["input_ids"], dtype=torch.long)[None]
    real = torch.tensor(encoded["attention_mask"], dtype=torch.bool)[None]
    model, cfg, _ = parent_model(torch.device("cpu"))
    state = torch.load(MODEL, map_location="cpu", weights_only=True)
    model.load_state_dict(state["model"])
    model.eval()
 
    def gradient(labels):
        model.zero_grad(set_to_none=True)
        logits = model(ids, real)
        total = F.cross_entropy(logits.transpose(1, 2).float(), labels, ignore_index=-100, reduction="sum")
        count = int((labels != -100).sum())
        loss = total / count
        loss.backward()
        grad = model.output.weight.grad.detach().clone()
        return float(loss.detach()), count, grad
 
    selected_loss, selected_count, selected_grad = gradient(assistant)
    all_loss, all_count, all_grad = gradient(whole)
    obs_start = encoded["copy_observation_value_start"]
    obs_length = encoded["copy_observation_value_length"]
    assert obs_length > 0
    assert all(assistant[0, i] == -100 for i in range(obs_start - 1, obs_start + obs_length - 1))
    result = {"scope": "one actual v3 train title trajectory, frozen chosen action model, two gradient calculations, no optimizer step",
              "model_sha256": sha(MODEL), "manifest_sha256": sha(V3 / "manifest.json"),
              "question": row["question"], "assistant_action": row["action"], "tool_return": row["observation"], "assistant_final": row["answer"],
              "segments_token_counts": {"user_and_role_prefix_plus_BOS": 1 + len(chunks[0]), "assistant_call": len(chunks[1]), "tool_return_plus_role_prefix": len(chunks[2]), "assistant_final": len(chunks[3])},
              "tool_value_token_ids_in_input": ids[0, obs_start:obs_start + obs_length].tolist(),
              "tool_value_target_positions_all_masked_in_assistant_objective": True,
              "assistant_loss_nll": selected_loss, "assistant_targets": selected_count,
              "whole_trace_loss_nll": all_loss, "whole_trace_targets": all_count,
              "output_head_gradient_cosine": float(F.cosine_similarity(selected_grad.flatten(), all_grad.flatten(), dim=0)),
              "output_head_gradient_difference_l2": float((selected_grad - all_grad).norm()),
              "note": "masking a tool-return target does not remove those token IDs from the causal input to the final answer; full-trace LM is another legal objective with different gradients"}
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({k: result[k] for k in ("assistant_loss_nll", "assistant_targets", "whole_trace_loss_nll", "whole_trace_targets", "output_head_gradient_cosine", "output_head_gradient_difference_l2")}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    args = p.parse_args()
    main(args.out)
