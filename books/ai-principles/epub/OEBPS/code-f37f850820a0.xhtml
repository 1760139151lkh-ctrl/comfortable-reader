"""Train one C25-descended model on both narrow math and code examples.
 
The earlier specialist tests have been opened by the author. Reusing them here
is a comparative diagnostic, never a fresh blind test for method selection.
"""
 
from __future__ import annotations
 
import argparse
import copy
import json
from pathlib import Path
 
import numpy as np
import torch
import torch.nn.functional as F
from tokenizers import Tokenizer
 
import arithmetic_curriculum as math_task
import code_expression_lab as code_task
 
 
ROOT = math_task.ROOT
SEED = 20260924
LEARNING_RATE = 0.0002
MATH_WEIGHT = 0.5
CODE_WEIGHT = 0.5
 
 
def validation(model, tokenizer, device, math_rows, code_rows) -> dict:
    math_nll = math_task.teacher_nll(model, math_rows, tokenizer, device)
    code_nll = code_task.teacher_nll(model, code_rows, tokenizer, device)
    math_free = math_task.free_generate(model, math_rows, tokenizer, device)
    code_free = code_task.free_generate(
        model, code_rows, tokenizer, device, code_task.VALIDATION_TEST_INPUTS
    )
    average_fraction = (
        math_free["exact_answer_and_eos"] / len(math_rows)
        + code_free["symbolically_equivalent"] / len(code_rows)
    ) / 2
    return {
        "math_response_nll": math_nll,
        "code_response_nll": code_nll,
        "math_free": math_free,
        "code_free": code_free,
        "average_task_fraction": average_fraction,
    }
 
 
def train(out_dir: Path, steps: int) -> None:
    if out_dir.exists() and any(out_dir.iterdir()):
        raise FileExistsError(f"preserve existing run: {out_dir}")
    out_dir.mkdir(parents=True)
    _, math_sha = math_task.load_manifest()
    _, code_sha = code_task.load_manifest()
    math_train = math_task.load_split("train")
    code_train = code_task.load_split("train")
    math_val = math_task.load_split("validation")
    code_val = code_task.load_split("validation")
    tokenizer = Tokenizer.from_file(str(math_task.TOKENIZER_FILE))
    device = math_task.choose_device()
    model, cfg, start_identity = math_task.start_model(device)
    torch.manual_seed(SEED)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED)
        torch.cuda.reset_peak_memory_stats()
    math_rng = np.random.default_rng(SEED)
    code_rng = np.random.default_rng(SEED + 1)
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=LEARNING_RATE, betas=(0.9, 0.95),
        weight_decay=0.01,
    )
    before = validation(model, tokenizer, device, math_val, code_val)
    best_key = (before["average_task_fraction"],
                -(before["math_response_nll"]["nll"]
                  + before["code_response_nll"]["nll"]))
    best_step = 0
    best_state = copy.deepcopy(model.state_dict())
    history = [{"step": 0, "validation": before}]
    for step in range(1, steps + 1):
        math_ids = math_rng.integers(0, len(math_train), size=16)
        code_ids = code_rng.integers(0, len(code_train), size=16)
        mb = [math_train[int(index)] for index in math_ids]
        cb = [code_train[int(index)] for index in code_ids]
        m_inputs, m_labels, m_real = math_task.batch(mb, tokenizer, device)
        c_inputs, c_labels, c_real = code_task.batch(cb, tokenizer, device)
        model.train()
        with math_task.autocast_for(device):
            m_logits = model(m_inputs, m_real)
            c_logits = model(c_inputs, c_real)
        math_loss = F.cross_entropy(
            m_logits.float().transpose(1, 2), m_labels,
            ignore_index=-100, reduction="sum",
        ) / (m_labels != -100).sum()
        code_loss = F.cross_entropy(
            c_logits.float().transpose(1, 2), c_labels,
            ignore_index=-100, reduction="sum",
        ) / (c_labels != -100).sum()
        loss = MATH_WEIGHT * math_loss + CODE_WEIGHT * code_loss
        if not torch.isfinite(loss):
            raise RuntimeError("nonfinite joint loss")
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
        if step % 200 == 0:
            current = validation(model, tokenizer, device, math_val, code_val)
            history.append({"step": step, "validation": current})
            key = (current["average_task_fraction"],
                   -(current["math_response_nll"]["nll"]
                     + current["code_response_nll"]["nll"]))
            if key > best_key:
                best_key = key
                best_step = step
                best_state = copy.deepcopy(model.state_dict())
            print(
                "step", step,
                "math", current["math_free"]["exact_answer_and_eos"], "/", len(math_val),
                "code", current["code_free"]["symbolically_equivalent"], "/", len(code_val),
                flush=True,
            )
    model.load_state_dict(best_state)
    checkpoint = {
        "model": {name: value.detach().cpu().clone()
                  for name, value in model.state_dict().items()},
        "selected_step": best_step,
        "math_manifest_sha256": math_sha,
        "code_manifest_sha256": code_sha,
        "start_model_identity": start_identity,
        "config": cfg.__dict__,
    }
    best_path = out_dir / "best.pt"
    torch.save(checkpoint, best_path)
    result = {
        "scope": "single model trained jointly on author-generated digit addition and AST-limited Python expression demonstrations; specialist tests already seen, selection uses validation only",
        "math_manifest_sha256": math_sha,
        "code_manifest_sha256": code_sha,
        "start_model_identity": start_identity,
        "training": {
            "steps": steps, "batch_size_math": 16, "batch_size_code": 16,
            "mean_loss_weights": [MATH_WEIGHT, CODE_WEIGHT],
            "learning_rate": LEARNING_RATE,
            "selection": "max average of math free exact fraction and code symbolic fraction on validation every 200 steps, tie by summed NLL",
        },
        "history": history,
        "selected_step": best_step,
        "selected_validation": validation(model, tokenizer, device, math_val, code_val),
        "checkpoint": str(best_path.relative_to(ROOT)).replace("\\", "/"),
        "checkpoint_sha256": math_task.file_sha(best_path),
        "device": str(device),
        "peak_gpu_allocated_bytes": (
            torch.cuda.max_memory_allocated() if device.type == "cuda" else None
        ),
        "independent_test_status": "specialist test rows known to author before joint design; no new blind test claim",
    }
    math_task.write_json(out_dir / "train.json", result)
    print("selected", best_step, best_key)
 
 
def compare(run_dir: Path, out: Path) -> None:
    if out.exists():
        raise FileExistsError(f"preserve existing diagnostic: {out}")
    _, math_sha = math_task.load_manifest()
    _, code_sha = code_task.load_manifest()
    report = json.loads((run_dir / "train.json").read_text(encoding="utf-8"))
    checkpoint_path = ROOT / report["checkpoint"]
    if (report["math_manifest_sha256"] != math_sha
            or report["code_manifest_sha256"] != code_sha
            or math_task.file_sha(checkpoint_path) != report["checkpoint_sha256"]):
        raise RuntimeError("joint checkpoint identity changed")
    device = math_task.choose_device()
    base, _, identity = math_task.start_model(device)
    if identity != report["start_model_identity"]:
        raise RuntimeError("C25 start identity changed")
    saved = torch.load(checkpoint_path, map_location="cpu", weights_only=True)
    if saved["selected_step"] != report["selected_step"]:
        raise RuntimeError("joint selected step changed")
    model = copy.deepcopy(base)
    model.load_state_dict(saved["model"])
    tokenizer = Tokenizer.from_file(str(math_task.TOKENIZER_FILE))
    # These two test splits were already opened for their respective specialist arms.
    math_test = math_task.load_split("test")
    code_test = code_task.load_split("test")
    math_paraphrase = [{
        **row, "prompt": math_task.prompt(
            row["tens"], row["units"], row["addend"], paraphrase=True
        )
    } for row in math_test]
    code_paraphrase = [{
        **row, "prompt": code_task.prompt(
            row["family"], row["variable"], row["constant"], paraphrase=True
        )
    } for row in code_test]
    result = {
        "scope": "posthoc comparison on already-opened specialist test rows; joint model chosen on validation only, not a new blind acceptance set",
        "selected_checkpoint_sha256": report["checkpoint_sha256"],
        "selected_step": report["selected_step"],
        "math_same_template": math_task.free_generate(
            model, math_test, tokenizer, device
        ),
        "code_same_template": code_task.free_generate(
            model, code_test, tokenizer, device, code_task.HELDOUT_TEST_INPUTS
        ),
        "math_paraphrase": math_task.free_generate(
            model, math_paraphrase, tokenizer, device
        ),
        "code_paraphrase": code_task.free_generate(
            model, code_paraphrase, tokenizer, device,
            code_task.HELDOUT_TEST_INPUTS
        ),
    }
    math_task.write_json(out, result)
    print("math", result["math_same_template"]["exact_answer_and_eos"],
          "/", len(math_test), "code",
          result["code_same_template"]["symbolically_equivalent"],
          "/", len(code_test))
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("mode", choices=("train", "compare"))
    parser.add_argument("--steps", type=int, default=2400)
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c27_joint_math_code_trial"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c27_joint_math_code_comparison.json"))
    args = parser.parse_args()
    if args.mode == "train":
        train(ROOT / args.run_dir, args.steps)
    else:
        compare(ROOT / args.run_dir, ROOT / args.out)
 
 
if __name__ == "__main__":
    main()
