"""核对 UCI 原包，只读取 day.csv 和 Readme.txt 到本地工作区。"""
 
import hashlib
import urllib.request
from pathlib import Path
from zipfile import ZipFile
 
 
WORK = Path(__file__).resolve().parents[1]
ARCHIVE = WORK / "data/uci_bike_sharing_original.zip"
FOLDER = WORK / "data/uci_bike_sharing"
URL = "https://archive.ics.uci.edu/static/public/275/bike%2Bsharing%2Bdataset.zip"
EXPECTED = {
    "archive": "b70182d0d0508e9abbb79306ce5c0cec34869000f8220175ac83d11dbe845401",
    "day.csv": "a6bcf826782d3c0fbfdcbeead17cd0884185a0dafe8ff10cd48a874ee7ba18be",
    "Readme.txt": "b92c8628622948bf0828c43a1b315d1517c3d7fd63654730d0dea379b8e88175",
}
 
 
def digest(data):
    return hashlib.sha256(data).hexdigest()
 
 
def main():
    FOLDER.mkdir(parents=True, exist_ok=True)
    if not ARCHIVE.exists():
        ARCHIVE.write_bytes(urllib.request.urlopen(URL, timeout=30).read())
    archive_bytes = ARCHIVE.read_bytes()
    if digest(archive_bytes) != EXPECTED["archive"]:
        raise ValueError("UCI ZIP 哈希不符，停止读取")
    with ZipFile(ARCHIVE) as source:
        for name in ("day.csv", "Readme.txt"):
            data = source.read(name)  # 固定文件名，不按包内路径展开
            if digest(data) != EXPECTED[name]:
                raise ValueError(f"{name} 原始字节哈希不符")
            target = FOLDER / name
            if target.exists() and target.read_bytes() != data:
                raise ValueError(f"{target} 已有不同内容，不覆盖")
            if not target.exists():
                target.write_bytes(data)
    print("原始 ZIP、日记录与说明文字的字节身份均已核对")
 
 
if __name__ == "__main__":
    main()
