"""Measure old-math squared gradients before sequential code-task training."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
import torch.nn.functional as F
from tokenizers import Tokenizer
 
import arithmetic_curriculum as math_task
import code_expression_lab as code_task
 
ROOT = math_task.ROOT
MATH_CHECKPOINT = ROOT / "work/runs/c27_digit_addition_long_trial/best.pt"
EXPECTED_MATH_SHA = "02be020308fabdc0b7ea05f800ac424d8c062e525130f91cae9aa30825464115"
SAMPLE_SEED = 4601
 
 
def measure(out_dir: Path, examples: int = 96) -> None:
    if out_dir.exists() and any(out_dir.iterdir()):
        raise FileExistsError(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    _, math_manifest_sha = math_task.load_manifest()
    _, code_manifest_sha = code_task.load_manifest()
    if math_task.file_sha(MATH_CHECKPOINT) != EXPECTED_MATH_SHA:
        raise RuntimeError("C27 mathematics checkpoint changed")
    device = math_task.choose_device()
    model, config, identity = math_task.start_model(device)
    saved = torch.load(MATH_CHECKPOINT, map_location="cpu", weights_only=True)
    if saved["task_manifest_sha256"] != math_manifest_sha or saved["start_model_identity"] != identity:
        raise RuntimeError("C27 math source identity changed")
    model.load_state_dict(saved["model"])
    tokenizer = Tokenizer.from_file(str(math_task.TOKENIZER_FILE))
    math_rows = math_task.load_split("train")
    code_rows = code_task.load_split("train")
    rng = np.random.default_rng(SAMPLE_SEED)
    selected = rng.choice(len(math_rows), size=examples, replace=False)
    squared = {name: torch.zeros_like(parameter, dtype=torch.float32)
               for name, parameter in model.named_parameters()}
    model.train()
    for index in selected:
        inputs, labels, real = math_task.batch([math_rows[int(index)]], tokenizer, device)
        model.zero_grad(set_to_none=True)
        with math_task.autocast_for(device):
            logits = model(inputs, real)
        loss = F.cross_entropy(logits.float().transpose(1, 2), labels,
                               ignore_index=-100, reduction="sum") / (labels != -100).sum()
        loss.backward()
        for name, parameter in model.named_parameters():
            if parameter.grad is not None:
                squared[name] += parameter.grad.detach().float().square()
    for value in squared.values():
        value.div_(examples)
    all_values = torch.cat([value.flatten() for value in squared.values()])
    raw_mean = float(all_values.mean())
    if raw_mean <= 0:
        raise RuntimeError("Empirical importance is zero")
    normalized = {name: (value / raw_mean).detach().cpu() for name, value in squared.items()}
 
    code_batch = [code_rows[int(i)] for i in rng.choice(len(code_rows), size=16, replace=False)]
    inputs, labels, real = code_task.batch(code_batch, tokenizer, device)
    model.zero_grad(set_to_none=True)
    with math_task.autocast_for(device):
        logits = model(inputs, real)
    code_loss = F.cross_entropy(logits.float().transpose(1, 2), labels,
                                ignore_index=-100, reduction="sum") / (labels != -100).sum()
    code_loss.backward()
    code_grad = torch.cat([p.grad.detach().float().flatten() for p in model.parameters() if p.grad is not None])
    checkpoint = {"importance": normalized, "normalization_raw_mean": raw_mean,
                  "math_checkpoint_sha256": EXPECTED_MATH_SHA,
                  "math_manifest_sha256": math_manifest_sha,
                  "code_manifest_sha256": code_manifest_sha,
                  "math_training_indices": [int(i) for i in selected]}
    importance_path = out_dir / "importance.pt"
    torch.save(checkpoint, importance_path)
    normalized_values = all_values / raw_mean
    report = {
        "scope": "Empirical diagonal mean of per-example old-math response-loss squared gradients; not exact Hessian or posterior",
        "math_checkpoint_sha256": EXPECTED_MATH_SHA,
        "math_manifest_sha256": math_manifest_sha,
        "code_manifest_sha256": code_manifest_sha,
        "sample_seed": SAMPLE_SEED, "old_math_train_examples": examples,
        "old_math_response_targets_per_example": 4,
        "parameter_count": int(all_values.numel()),
        "importance_raw_mean": raw_mean,
        "importance_normalized_quantiles": np.quantile(normalized_values.cpu().numpy(), [0, .5, .9, .99, .999, 1]).tolist(),
        "first_code_train_batch_nll": float(code_loss.detach()),
        "first_code_gradient_rms": float(torch.sqrt(torch.mean(code_grad.square()))),
        "first_code_gradient_norm": float(torch.linalg.vector_norm(code_grad)),
        "importance_path": str(importance_path.relative_to(ROOT)).replace("\\", "/"),
        "importance_sha256": math_task.file_sha(importance_path),
        "code_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "device": str(device),
    }
    math_task.write_json(out_dir / "report.json", report)
    print(json.dumps(report, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--out-dir", type=Path, required=True)
    ap.add_argument("--examples", type=int, default=96)
    args = ap.parse_args()
    measure(ROOT / args.out_dir, args.examples)
