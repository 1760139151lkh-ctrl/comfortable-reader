"""Visualize the measured Q1/Q2 light curves after the locked first Q2 test."""
from __future__ import annotations
 
import argparse
import hashlib
import importlib.util
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
source_path = Path(__file__).with_name("c45_kepler_transit.py")
spec = importlib.util.spec_from_file_location("c45_kepler_transit", source_path)
assert spec is not None and spec.loader is not None
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
 
 
def main(run_dir: Path, out_dir: Path) -> None:
    if out_dir.exists() and any(out_dir.iterdir()):
        raise FileExistsError(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    q1_path = run_dir / "q1_train_validation.json"
    q2_path = run_dir / "q2_first_test.json"
    q1 = json.loads(q1_path.read_text(encoding="utf-8"))
    q2 = json.loads(q2_path.read_text(encoding="utf-8"))
    if module.sha(source_path) != q1["code_sha256"] or module.sha(q1_path) != q2["q1_report_sha256"]:
        raise ValueError("Figure inputs changed from frozen model/test")
    t1, f1, _, _ = module.load("q1")
    t2, f2, _, _ = module.load("q2")
    y1 = f1 / q1["normalization_train_median_e_per_s"]
    y2 = f2 / q2["calibration_median_e_per_s"]
    q2_scored = t2 >= t2[0] + q2["calibration_days"]
    model = q1["models"]["box"]
    period = model["period_days"]
    epoch = model["epoch_bkjd"]
    duration = model["duration_days"]
    phase1 = (t1 - epoch + period/2) % period - period/2
    phase2 = (t2[q2_scored] - epoch + period/2) % period - period/2
    fig, axes = plt.subplots(3, 1, figsize=(11, 9), constrained_layout=True)
    for ax, t, y, name, c in (
        (axes[0], t1, y1, "Q1: fitted first 70% / later same-quarter check", "#27669d"),
        (axes[1], t2[q2_scored], y2[q2_scored], "Q2: later quarter, after four-day baseline calibration", "#21827b"),
    ):
        ax.scatter(t, y, s=6, alpha=0.65, color=c, rasterized=True)
        order = np.argsort(t)
        ax.plot(t[order], module.box(t[order], model), color="#d14e43", linewidth=1.2)
        ax.set(ylabel="relative flux", title=name)
        ax.grid(alpha=0.2)
    axes[0].axvline(q1["split_bkjd"], color="black", linewidth=1, linestyle="--", label="fit / validation")
    axes[0].legend(loc="lower left", fontsize=8)
    axes[1].set(xlabel="BJD - 2454833 (days)")
    ax = axes[2]
    ax.scatter(phase1, y1, s=5, alpha=0.35, color="#27669d", label="Q1")
    ax.scatter(phase2, y2[q2_scored], s=5, alpha=0.25, color="#21827b", label="Q2 scored")
    bins = np.linspace(-period/2, period/2, 61)
    for phase, y, color, label in ((phase1, y1, "#153d62", "Q1 bin median"),
                                   (phase2, y2[q2_scored], "#0d564f", "Q2 bin median")):
        inds = np.digitize(phase, bins) - 1
        center = (bins[:-1] + bins[1:]) / 2
        median = np.array([np.median(y[inds == i]) if np.any(inds == i) else np.nan for i in range(len(center))])
        ax.plot(center, median, color=color, linewidth=1.5, label=label)
    ax.axvspan(-duration/2, duration/2, color="#d14e43", alpha=0.12, label="Q1 box interval")
    ax.plot(np.linspace(-period/2, period/2, 1000), module.box(epoch + np.linspace(-period/2, period/2, 1000), model),
            color="#d14e43", linewidth=1.3)
    ax.set(xlabel="days from predicted transit center (phase folded)", ylabel="relative flux",
           title="Same Q1 period and epoch applied to both quarters")
    ax.set_xlim(-0.35, 0.35)
    ax.grid(alpha=0.2)
    ax.legend(fontsize=7, ncol=3, loc="lower right")
    png = out_dir / "hat_p_7b_q1_q2_measured_and_predicted.png"
    fig.savefig(png, dpi=175)
    plt.close(fig)
    receipt = {"q1_report_sha256": module.sha(q1_path), "q2_report_sha256": module.sha(q2_path),
               "q1_fits_sha256": module.sha(module.FILES["q1"]), "q2_fits_sha256": module.sha(module.FILES["q2"]),
               "figure_sha256": hashlib.sha256(png.read_bytes()).hexdigest(),
               "figure": str(png), "period_days": period, "duration_days": duration,
               "q2_calibration_excluded": int((~q2_scored).sum()), "q2_scored": int(q2_scored.sum())}
    (out_dir / "receipt.json").write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(receipt, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--run-dir", type=Path, required=True)
    ap.add_argument("--out-dir", type=Path, required=True)
    args = ap.parse_args()
    main(args.run_dir, args.out_dir)
