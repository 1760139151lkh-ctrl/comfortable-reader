"""Fetch two fixed public MAST HAT-P-7 light-curve files; never replace changed bytes."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import requests
 
BASE = "https://archive.stsci.edu/pub/kepler/lightcurves/0106/010666592/"
FILES = {
    "q1": "kplr010666592-2009166043257_llc.fits",
    "q2": "kplr010666592-2009259160929_llc.fits",
}
EXPECTED_SHA256 = {
    "q1": "37faa7dc82fa5d9ec6d794369d93def0d78a49c91e61cf19bb4cafe571bb8f6c",
    "q2": "2a10ce0fa87e8a757e6d66a740669bcf94a704eecf0646bbf898ad91b402fc08",
}
DEST = Path(__file__).resolve().parents[1] / "data" / "c45_kepler_hatp7"
 
 
def get_one(which: str) -> dict:
    DEST.mkdir(parents=True, exist_ok=True)
    name = FILES[which]
    target = DEST / name
    url = BASE + name
    if not target.exists():
        response = requests.get(url, timeout=90)
        response.raise_for_status()
        payload = response.content
        if not payload.startswith(b"SIMPLE  ="):
            raise ValueError("Download does not look like FITS")
        target.write_bytes(payload)
    payload = target.read_bytes()
    if not payload.startswith(b"SIMPLE  ="):
        raise ValueError("Existing file is not FITS")
    digest = hashlib.sha256(payload).hexdigest()
    if digest != EXPECTED_SHA256[which]:
        raise ValueError(f"Official file bytes changed for {which}: {digest}")
    return {
        "url": url, "path": str(target.relative_to(DEST.parents[1])),
        "bytes": len(payload), "sha256": digest,
    }
 
 
if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("which", choices=["q1", "q2", "both"])
    args = parser.parse_args()
    rows = [get_one(key) for key in FILES if args.which in ("both", key)]
    print(json.dumps(rows, ensure_ascii=False, indent=2))
