"""Rigid-motion invariance/equivariance on C34's real laser-derived 3D points."""
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import numpy as np
from scipy.spatial.distance import cdist
from scipy.spatial.transform import Rotation
 
ROOT = Path(__file__).resolve().parents[1]
PLY = ROOT / "runs" / "c34_eth3d_delivery_geometry_current" / "reference_partial_points.ply"
 
 
def load_points(path):
    with path.open("r", encoding="utf-8") as f:
        header = []
        for line in f:
            header.append(line.strip())
            if line.strip() == "end_header":
                break
        if "format ascii 1.0" not in header or "element vertex 22485" not in header:
            raise RuntimeError("unexpected C34 PLY header")
        rows = []
        for _ in range(22485):
            fields = f.readline().split()
            if len(fields) != 6:
                raise RuntimeError("bad PLY vertex row")
            rows.append([float(v) for v in fields[:3]] + [int(v) / 255 for v in fields[3:]])
    data = np.array(rows, dtype=np.float64)
    if not np.isfinite(data).all():
        raise RuntimeError("nonfinite point")
    return data[:, :3], data[:, 3:]
 
 
def messages(points, color, neighbors, sigma):
    differences = points[neighbors] - points[:, None, :]
    squared = (differences**2).sum(axis=-1)
    weights = np.exp(-squared / sigma**2)
    scalar = (weights * color[neighbors, 0]).sum(axis=1)
    vector = (weights[:, :, None] * differences).sum(axis=1)
    return scalar, vector
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    p.add_argument("--rotation-deg-xyz", nargs=3, type=float, default=[23.0, -41.0, 17.0])
    p.add_argument("--translation-m", nargs=3, type=float, default=[0.7, -0.3, 1.2])
    p.add_argument("--sigma-scale", type=float, default=1.0)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("result exists")
    if a.sigma_scale <= 0:
        raise ValueError("--sigma-scale must be positive")
    points, color = load_points(PLY)
    rng = np.random.default_rng(20260925)
    chosen = rng.choice(len(points), size=256, replace=False)
    positions, rgb = points[chosen], color[chosen]
    squared_distances = cdist(positions, positions, metric="sqeuclidean")
    np.fill_diagonal(squared_distances, np.inf)
    neighbors = np.argsort(squared_distances, axis=1)[:, :8]
    sigma = float(np.median(np.sqrt(np.take_along_axis(squared_distances, neighbors, axis=1)))) * a.sigma_scale
    scalar, vector = messages(positions, rgb, neighbors, sigma)
    rotation = Rotation.from_euler("xyz", a.rotation_deg_xyz, degrees=True).as_matrix()
    shift = np.array(a.translation_m)
    moved = positions @ rotation.T + shift
    moved_distances = cdist(moved, moved, metric="sqeuclidean")
    np.fill_diagonal(moved_distances, np.inf)
    moved_neighbors = np.argsort(moved_distances, axis=1)[:, :8]
    if not np.array_equal(neighbors, moved_neighbors):
        raise RuntimeError("kNN changed under rigid motion")
    scalar_moved, vector_moved = messages(moved, rgb, moved_neighbors, sigma)
    scalar_error = float(np.max(np.abs(scalar_moved - scalar)))
    vector_error = float(np.max(np.abs(vector_moved - vector @ rotation.T)))
    scale_changed, _ = messages(positions * 2.0, rgb, neighbors, sigma)
    scale_effect = float(np.mean(np.abs(scale_changed - scalar)))
    if scalar_error > 1e-10 or vector_error > 1e-10:
        raise RuntimeError("rigid-motion property failed")
    result = {"source": "C34 ETH3D delivery_area_1s laser reference back-projected partial PLY, not trained GNN", "points_in_source": len(points), "points_sampled": len(chosen), "graph_choice": "8 nearest neighbours by Euclidean distance among sampled positions; PLY itself did not define these edges", "median_neighbour_distance_m": sigma / a.sigma_scale, "sigma_scale": a.sigma_scale, "sigma_m": sigma, "rigid_rotation_degrees_xyz": a.rotation_deg_xyz, "rigid_translation_m": shift.tolist(), "same_knn_indices_after_rigid_transform": True, "scalar_message_rotation_translation_invariance_max_error": scalar_error, "vector_message_equivariance_max_error": vector_error, "doubling_space_coordinates_while_fixing_sigma_scalar_message_mean_change": scale_effect, "scope": "exact numerical property of this hand-designed distance/difference message; no training, no semantic 3D component recognition"}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(result, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
