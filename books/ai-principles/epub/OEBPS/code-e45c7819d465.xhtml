"""C19: inspect the trained model's causal dependency and saved resume state.
 
Uses only C18 train blocks; it does not open the test arrays.
"""
 
from __future__ import annotations
 
import argparse
import json
 
import numpy as np
import torch
 
import train_causal_wikitext as t
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--output-dir", default="work/results")
    args = parser.parse_args()
    t.select_output_dir(args.output_dir)
    result_path = t.RESULTS / "causal_lm_verification.json"
    cfg = t.TrainConfig()
    manifest, manifest_sha = t.load_manifest()
    part = t.load_split(manifest, "train")
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model, _ = t.create(cfg, device)
    best = t.load_checkpoint(t.BEST, device, cfg, manifest_sha)
    latest = t.load_checkpoint(t.LATEST, device, cfg, manifest_sha)
    model.load_state_dict(best["model"])
    model.eval()
    with torch.no_grad():
        ids, _, real = t.batch(part, np.array([0]), device)
        with t.autocast_for(device):
            original = model(ids, real)
            future_edit = ids.clone()
            future_edit[0, 10] = 303 if int(ids[0, 10]) != 303 else 304
            after_future = model(future_edit, real)
            prefix_edit = ids.clone()
            prefix_edit[0, 1] = 303 if int(ids[0, 1]) != 303 else 304
            after_prefix = model(prefix_edit, real)
        future_max = float((original[:, :10] - after_future[:, :10]).abs().max())
        later_change = float((original[:, 10:20] - after_prefix[:, 10:20]).abs().max())
        first_article = np.flatnonzero(part["article_index"] == 0)
        tail_ids, _, tail_real = t.batch(part, np.array([first_article[-1]]), device)
        with t.autocast_for(device):
            tail_original = model(tail_ids, tail_real)
            pad_edit = tail_ids.clone()
            pad_edit[0, -1] = 303
            tail_after = model(pad_edit, tail_real)
        real_length = int(tail_real.sum())
        pad_change = float((tail_original[:, :real_length] - tail_after[:, :real_length]).abs().max())
    if future_max != 0 or pad_change != 0 or later_change <= 0:
        raise AssertionError("causal, PAD or context-dependence check failed")
    if latest["epoch"] != cfg.epochs or latest["next_batch"] != 0 or latest["global_step"] != 1336:
        raise AssertionError("unexpected latest resume cursor")
    if len(latest["optimizer"]["state"]) == 0:
        raise AssertionError("optimizer moments were not saved")
    report = {
        "scope": "one full train block future/prefix edits and first article padded tail; no test arrays",
        "c18_manifest_sha256": manifest_sha,
        "best_checkpoint_sha256": t.sha256(t.BEST),
        "first_full_block_index": 0,
        "future_edited_position": 10,
        "earlier_logits_max_change_after_future_edit": future_max,
        "later_logits_max_change_after_prefix_edit": later_change,
        "tail_block_index": int(first_article[-1]),
        "tail_real_positions": real_length,
        "real_logits_max_change_after_masked_pad_edit": pad_change,
        "latest_checkpoint_epoch": latest["epoch"],
        "latest_checkpoint_global_step": latest["global_step"],
        "optimizer_state_parameter_entries": len(latest["optimizer"]["state"]),
        "saved_cpu_rng": latest["torch_cpu_rng"].numel() > 0,
        "saved_cuda_rng_count": len(latest["torch_cuda_rng_all"]),
    }
    t.write_json(result_path, report)
    print(json.dumps(report, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
