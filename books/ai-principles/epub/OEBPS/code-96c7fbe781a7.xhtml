"""Show the same base noise after several numerical solvers/step counts."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
from c31_straight_flow import ROOT, digest
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c31_fashion_flow"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/figures/c31_fashion_flow_solvers.png"))
    parser.add_argument("--receipt", type=Path,
                        default=Path("work/results/c31_fashion_flow_figure.json"))
    args = parser.parse_args()
    run, out, receipt = ROOT / args.run_dir, ROOT / args.out, ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing flow figure")
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    if digest(ROOT / report["checkpoint"]) != report["checkpoint_sha256"]:
        raise RuntimeError("flow checkpoint changed")
    real_path = ROOT / report["real_validation_16_path"]
    if digest(real_path) != report["real_validation_16_sha256"]:
        raise RuntimeError("flow real validation sample changed")
    rows = [("真实验证图", np.load(real_path, allow_pickle=False)[:8])]
    for item in report["samplers"]:
        path = ROOT / item["path"]
        if digest(path) != item["sha256"]:
            raise RuntimeError("flow generated array changed")
        rows.append((
            f"{item['method'].title()} {item['steps']}步",
            np.load(path, allow_pickle=False)[:8],
        ))
    fig, axes = plt.subplots(len(rows), 8, figsize=(13, 1.8 * len(rows)))
    for row, (title, images) in enumerate(rows):
        for col in range(8):
            if report["channels"] == 1:
                axes[row, col].imshow(
                    images[col], cmap="gray", vmin=0, vmax=255,
                    interpolation="nearest",
                )
            else:
                axes[row, col].imshow(images[col], interpolation="nearest")
            axes[row, col].axis("off")
        fig.text(0.01, 1 - (row + 0.5) / len(rows), title,
                 va="center", fontsize=10, fontproperties="Microsoft YaHei")
    fig.tight_layout(rect=(0.13, 0, 1, 1))
    out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=180)
    plt.close(fig)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps({
        "scope": "same first8 latent noise for five fixed numerical integrators; real validation images only as reference, not paired targets; visual comparison",
        "source_identity": report["source_identity"],
        "checkpoint_sha256": report["checkpoint_sha256"],
        "generated_array_sha256": {
            item["name"]: item["sha256"] for item in report["samplers"]
        },
        "row_titles": [name for name, _ in rows],
        "figure": str(out.relative_to(ROOT)).replace("\\", "/"),
        "official_test_opened": False,
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
 
 
if __name__ == "__main__":
    main()
