"""Check old C25/C19 validation targets after the selected C26 restricted RL run."""
 
from __future__ import annotations
 
import argparse
import copy
import json
from pathlib import Path
 
import torch
 
from binary_feedback_lab import (
    ROOT, choose_device, file_sha, load_manifest, start_model, write_json,
)
from train_dolly_sft import (
    evaluate_sft, evaluate_wiki, load_base, load_sft_manifest, load_sft_split,
)
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path, default=Path("work/runs/c26_binary_v2_balanced_trial"))
    parser.add_argument("--data-dir", type=Path, default=Path("work/data/c26_binary_digits_v2"))
    parser.add_argument("--out", type=Path, default=Path("work/results/c26_binary_v2_retention.json"))
    args = parser.parse_args()
    run_dir = ROOT / args.run_dir
    task_manifest = ROOT / args.data_dir / "manifest.json"
    report = json.loads((run_dir / "train.json").read_text(encoding="utf-8"))
    if report["task_manifest_sha256"] != file_sha(task_manifest):
        raise RuntimeError("task version changed")
    checkpoint_path = ROOT / report["checkpoint"]
    if file_sha(checkpoint_path) != report["checkpoint_sha256"]:
        raise RuntimeError("selected C26 checkpoint changed")
    device = choose_device()
    base, cfg, identity = start_model(device)
    if identity != report["start_model_identity"]:
        raise RuntimeError("starting C25 checkpoint changed")
    checkpoint = torch.load(checkpoint_path, map_location="cpu", weights_only=True)
    if checkpoint["selected_step"] != report["selected_step"]:
        raise RuntimeError("selected step changed")
    trained = copy.deepcopy(base)
    trained.load_state_dict(checkpoint["model"])
    trained.eval()
    sft_manifest, sft_sha = load_sft_manifest()
    if sft_sha != identity["c25_sft_manifest_sha256"]:
        raise RuntimeError("C25 response validation split changed")
    dolly_validation = load_sft_split(sft_manifest, "validation")
    _, _, c19 = load_base()
    results = {}
    for name, model in (("start_c25_full_mix", base), ("selected_c26_policy", trained)):
        results[name] = {
            "dolly_validation_response": evaluate_sft(
                model, dolly_validation, device, with_categories=False
            ),
            "old_wikitext_validation": evaluate_wiki(
                model, c19["wiki_manifest"], cfg, device
            ),
        }
    output = {
        "scope": "already-used validation data only; retention diagnostic after C26 checkpoint fixed, no C26 test-based tuning",
        "selected_step": report["selected_step"],
        "task_manifest_sha256": report["task_manifest_sha256"],
        "checkpoint_sha256": report["checkpoint_sha256"],
        "results": results,
    }
    write_json(ROOT / args.out, output)
    print("Dolly validation NLL", results["start_c25_full_mix"]["dolly_validation_response"]["response_nll"],
          "to", results["selected_c26_policy"]["dolly_validation_response"]["response_nll"])
    print("Wiki validation NLL", results["start_c25_full_mix"]["old_wikitext_validation"]["nll"],
          "to", results["selected_c26_policy"]["old_wikitext_validation"]["nll"])
 
 
if __name__ == "__main__":
    main()
