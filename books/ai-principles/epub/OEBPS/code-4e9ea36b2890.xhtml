"""Repeated fixed normalized neighbor averaging on real Cora word features."""
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import networkx as nx
import numpy as np
import torch
 
from c36_cora_node_models import DATA, normalized_adjacency
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    p.add_argument("--max-rounds", type=int, default=32)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("result exists")
    if a.max_rounds < 0:
        raise ValueError("--max-rounds must be nonnegative")
    torch.set_num_threads(8)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    x = torch.from_numpy(np.load(DATA / "features_uint8.npy").astype(np.float32)).to(device)
    x = x / x.sum(dim=1, keepdim=True).clamp_min(1)
    edges = np.load(DATA / "undirected_unique_edges_int64.npy")
    s = normalized_adjacency(edges, len(x), device)
    graph = nx.Graph()
    graph.add_nodes_from(range(len(x)))
    graph.add_edges_from(map(tuple, edges.tolist()))
    components = sorted((len(part) for part in nx.connected_components(graph)), reverse=True)
    rng = np.random.default_rng(20260925)
    pairs = torch.from_numpy(rng.integers(0, len(x), (4096, 2))).long().to(device)
    records = []
    h = x
    with torch.inference_mode():
        for k in range(a.max_rounds + 1):
            if k in (0, 1, 2, 4, 8, 16, 32, 64, 128) or k == a.max_rounds:
                sh = torch.sparse.mm(s, h)
                energy = float((h * (h - sh)).sum())
                pair_rms = float((h[pairs[:, 0]] - h[pairs[:, 1]]).square().sum(dim=1).mean().sqrt())
                records.append({"averaging_rounds": k, "normalized_dirichlet_energy": energy, "sampled_pair_feature_rms_distance": pair_rms, "feature_l2_norm": float(h.square().sum().sqrt())})
            h = torch.sparse.mm(s, h)
    if any(records[i + 1]["normalized_dirichlet_energy"] > records[i]["normalized_dirichlet_energy"] + 1e-5 for i in range(len(records) - 1)):
        raise RuntimeError("the expected fixed-S energy decrease did not occur")
    output = {"input": "all2708 real Cora anonymous word features, row-normalized, no labels, no trained network weights", "operator": "S = D^-1/2 (A+I) D^-1/2 for undirected/symmetrized citation graph", "max_rounds": a.max_rounds, "connected_components": len(components), "largest_component": components[0], "meaning": "This fixed linear diffusion can erase distinction; it is not a theorem that every learned deep GNN must over-smooth", "records": records}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(output, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
