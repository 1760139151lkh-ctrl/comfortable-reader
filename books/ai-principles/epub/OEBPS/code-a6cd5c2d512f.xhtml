"""One-model C27 math/code training with C25 response and C19 text replay."""
 
from __future__ import annotations
 
import argparse
import copy
import json
from pathlib import Path
 
import numpy as np
import torch
import torch.nn.functional as F
from tokenizers import Tokenizer
 
import arithmetic_curriculum as math_task
import code_expression_lab as code_task
from joint_math_code_sft import validation
from train_dolly_sft import (
    load_base, load_sft_manifest, load_sft_split, load_wiki_split,
    torch_batch as dolly_batch, wiki_batch,
)
 
 
ROOT = math_task.ROOT
SEED = 20260924
LOSS_WEIGHTS = {"math": 0.35, "code": 0.35, "dolly": 0.15, "wiki": 0.15}
 
 
def mean_nll(model, inputs, labels, real, device):
    with math_task.autocast_for(device):
        logits = model(inputs, real)
    return F.cross_entropy(
        logits.float().transpose(1, 2), labels,
        ignore_index=-100, reduction="sum",
    ) / (labels != -100).sum()
 
 
def train(out_dir: Path, steps: int) -> None:
    if out_dir.exists() and any(out_dir.iterdir()):
        raise FileExistsError(f"preserve existing run: {out_dir}")
    out_dir.mkdir(parents=True)
    _, math_sha = math_task.load_manifest()
    _, code_sha = code_task.load_manifest()
    sft_manifest, sft_sha = load_sft_manifest()
    _, _, c19 = load_base()
    wiki_sha = c19["c18_manifest_sha256"]
    math_train = math_task.load_split("train")
    code_train = code_task.load_split("train")
    math_val = math_task.load_split("validation")
    code_val = code_task.load_split("validation")
    dolly_train = load_sft_split(sft_manifest, "train")
    wiki_train = load_wiki_split(c19["wiki_manifest"], "train")
    tokenizer = Tokenizer.from_file(str(math_task.TOKENIZER_FILE))
    device = math_task.choose_device()
    model, cfg, start_identity = math_task.start_model(device)
    if (start_identity["c25_sft_manifest_sha256"] != sft_sha
            or start_identity["c19_checkpoint_sha256"] != c19["c19_checkpoint_sha256"]):
        raise RuntimeError("parent source mismatch")
    torch.manual_seed(SEED)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED)
        torch.cuda.reset_peak_memory_stats()
    math_rng = np.random.default_rng(SEED)
    code_rng = np.random.default_rng(SEED + 1)
    dolly_rng = np.random.default_rng(SEED + 2)
    wiki_rng = np.random.default_rng(SEED + 3)
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=0.0002, betas=(0.9, 0.95),
        weight_decay=0.01,
    )
    before = validation(model, tokenizer, device, math_val, code_val)
    best_key = (before["average_task_fraction"],
                -(before["math_response_nll"]["nll"]
                  + before["code_response_nll"]["nll"]))
    best_step = 0
    best_state = copy.deepcopy(model.state_dict())
    history = [{"step": 0, "validation": before}]
    seen = {"math": 0, "code": 0, "dolly": 0, "wiki": 0}
    for step in range(1, steps + 1):
        mi = math_rng.integers(0, len(math_train), size=16)
        ci = code_rng.integers(0, len(code_train), size=16)
        di = dolly_rng.integers(0, len(dolly_train["input_ids"]), size=16)
        wi = wiki_rng.integers(0, len(wiki_train["input_ids"]), size=16)
        mb = [math_train[int(index)] for index in mi]
        cb = [code_train[int(index)] for index in ci]
        m = math_task.batch(mb, tokenizer, device)
        c = code_task.batch(cb, tokenizer, device)
        d = dolly_batch(dolly_train, di, device)
        w = wiki_batch(wiki_train, wi, device)
        seen["math"] += int((m[1] != -100).sum())
        seen["code"] += int((c[1] != -100).sum())
        seen["dolly"] += int((d[1] != -100).sum())
        seen["wiki"] += int((w[1] != -100).sum())
        model.train()
        objective = (
            LOSS_WEIGHTS["math"] * mean_nll(model, *m, device)
            + LOSS_WEIGHTS["code"] * mean_nll(model, *c, device)
            + LOSS_WEIGHTS["dolly"] * mean_nll(model, *d, device)
            + LOSS_WEIGHTS["wiki"] * mean_nll(model, *w, device)
        )
        if not torch.isfinite(objective):
            raise RuntimeError("nonfinite four-way loss")
        optimizer.zero_grad(set_to_none=True)
        objective.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
        if step % 200 == 0:
            current = validation(model, tokenizer, device, math_val, code_val)
            history.append({"step": step, "validation": current})
            key = (current["average_task_fraction"],
                   -(current["math_response_nll"]["nll"]
                     + current["code_response_nll"]["nll"]))
            if key > best_key:
                best_key = key
                best_step = step
                best_state = copy.deepcopy(model.state_dict())
            print("step", step,
                  "math", current["math_free"]["exact_answer_and_eos"],
                  "code", current["code_free"]["symbolically_equivalent"],
                  flush=True)
    model.load_state_dict(best_state)
    checkpoint = {
        "model": {name: value.detach().cpu().clone()
                  for name, value in model.state_dict().items()},
        "selected_step": best_step,
        "math_manifest_sha256": math_sha,
        "code_manifest_sha256": code_sha,
        "dolly_manifest_sha256": sft_sha,
        "wiki_manifest_sha256": wiki_sha,
        "start_model_identity": start_identity,
        "config": cfg.__dict__,
    }
    best_path = out_dir / "best.pt"
    torch.save(checkpoint, best_path)
    report = {
        "scope": "one C25-descended model with four train-split mean losses; math/code specialist tests already seen before this design, selection uses math/code validation only",
        "math_manifest_sha256": math_sha,
        "code_manifest_sha256": code_sha,
        "dolly_manifest_sha256": sft_sha,
        "wiki_manifest_sha256": wiki_sha,
        "start_model_identity": start_identity,
        "training": {
            "steps": steps, "batch_size_each": 16,
            "loss_weights_on_separate_target_means": LOSS_WEIGHTS,
            "learning_rate": 0.0002,
            "selection": "max average math/code validation free-correct fractions every 200 steps; old tasks measured only after checkpoint selected",
        },
        "targets_seen_all_steps": seen,
        "history": history,
        "selected_step": best_step,
        "selected_validation": validation(model, tokenizer, device, math_val, code_val),
        "checkpoint": str(best_path.relative_to(ROOT)).replace("\\", "/"),
        "checkpoint_sha256": math_task.file_sha(best_path),
        "device": str(device),
        "peak_gpu_allocated_bytes": (
            torch.cuda.max_memory_allocated() if device.type == "cuda" else None
        ),
        "independent_test_status": "specialist tests known to author; any same-row comparison is posthoc",
    }
    math_task.write_json(out_dir / "train.json", report)
    print("selected", best_step, best_key)
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--steps", type=int, default=2400)
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c27_joint_replay_trial"))
    args = parser.parse_args()
    train(ROOT / args.run_dir, args.steps)
 
 
if __name__ == "__main__":
    main()
