"""Analyze aggregate CoVal rankings without copying annotator identities or demographics."""
 
from __future__ import annotations
 
import argparse
import collections
import hashlib
import itertools
import json
import re
from pathlib import Path
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/coval_eec757"
ALTERNATIVES = "ABCD"
 
 
def ranking(text: str) -> dict[str, int]:
    groups = [re.findall(r"[ABCD]", part) for part in text.replace(" ", "").split(">")]
    flattened = [letter for group in groups for letter in group]
    if sorted(flattened) != list(ALTERNATIVES):
        raise ValueError(f"unknown rank format: {text!r}")
    return {letter: position for position, group in enumerate(groups) for letter in group}
 
 
def strict_top(rank: dict[str, int]) -> str | None:
    first = [letter for letter, place in rank.items() if place == 0]
    return first[0] if len(first) == 1 else None
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c28_coval_disagreement.json"))
    args = parser.parse_args()
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError(f"preserve existing result: {out}")
    manifest = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    source = DATA / "comparisons.jsonl"
    if hashlib.sha256(source.read_bytes()).hexdigest() != manifest["files"]["comparisons.jsonl"]["sha256"]:
        raise RuntimeError("source changed")
    prompt_count = 0
    assessment_count = 0
    assessments_missing_personal = 0
    prompts_with_two_personal_ranks = 0
    personal_world_strict_both = 0
    personal_world_changed = 0
    prompt_multiple_personal_tops = 0
    prompt_multiple_world_tops = 0
    prompt_majority_changed = 0
    pair_decisions = 0
    pair_with_both_sides = 0
    close_pairs = 0
    pair_majority_strength = []
    world_pair_decisions = 0
    world_pairs_with_both_sides = 0
    world_close_pairs = 0
    subjectivity = collections.Counter()
    subjectivity_disagreement = collections.Counter()
    fixed_example = None
    for line in source.open(encoding="utf-8"):
        item = json.loads(line)
        prompt_count += 1
        personal = []
        world = []
        votes: dict[tuple[str, str], collections.Counter] = {
            pair: collections.Counter() for pair in itertools.combinations(ALTERNATIVES, 2)
        }
        world_votes: dict[tuple[str, str], collections.Counter] = {
            pair: collections.Counter() for pair in itertools.combinations(ALTERNATIVES, 2)
        }
        for assessment in item["metadata"]["assessments"]:
            assessment_count += 1
            blocks = assessment["ranking_blocks"]
            if not blocks["world"]:
                raise RuntimeError("a world ranking is missing from current source")
            w = ranking(blocks["world"][0]["ranking"])
            p = ranking(blocks["personal"][0]["ranking"]) if blocks["personal"] else None
            assessments_missing_personal += int(p is None)
            p_top, w_top = (strict_top(p) if p is not None else None), strict_top(w)
            if p_top is not None:
                personal.append(p_top)
            if w_top is not None:
                world.append(w_top)
            if p_top is not None and w_top is not None:
                personal_world_strict_both += 1
                personal_world_changed += int(p_top != w_top)
            subjectivity[assessment["subjectivity"]] += 1
            for a, b in votes:
                if p is not None:
                    if p[a] < p[b]:
                        votes[(a, b)][a] += 1
                    elif p[b] < p[a]:
                        votes[(a, b)][b] += 1
                if w[a] < w[b]:
                    world_votes[(a, b)][a] += 1
                elif w[b] < w[a]:
                    world_votes[(a, b)][b] += 1
        distinct_tops = len(set(personal))
        prompts_with_two_personal_ranks += int(len(personal) >= 2)
        prompt_multiple_personal_tops += int(len(personal) >= 2 and distinct_tops >= 2)
        prompt_multiple_world_tops += int(len(set(world)) >= 2)
        majority_subjectivity = collections.Counter(
            assessment["subjectivity"]
            for assessment in item["metadata"]["assessments"]
        ).most_common(1)[0][0]
        subjectivity_disagreement[majority_subjectivity] += int(distinct_tops >= 2)
        pc = collections.Counter(personal)
        wc = collections.Counter(world)
        if pc and wc:
            p_winners = [k for k, v in pc.items() if v == max(pc.values())]
            w_winners = [k for k, v in wc.items() if v == max(wc.values())]
            if len(p_winners) == len(w_winners) == 1:
                prompt_majority_changed += int(p_winners[0] != w_winners[0])
        for choices in votes.values():
            if sum(choices.values()) < 2:
                continue
            pair_decisions += 1
            pair_with_both_sides += int(len(choices) == 2)
            strength = max(choices.values()) / sum(choices.values())
            pair_majority_strength.append(strength)
            close_pairs += int(strength <= 0.6)
        for choices in world_votes.values():
            if sum(choices.values()) < 2:
                continue
            world_pair_decisions += 1
            world_pairs_with_both_sides += int(len(choices) == 2)
            world_close_pairs += int(max(choices.values()) / sum(choices.values()) <= 0.6)
        if fixed_example is None:
            fixed_example = {
                "prompt_id": item["prompt_id"],
                "assessments": len(item["metadata"]["assessments"]),
                "personal_top_counts": dict(pc),
                "world_top_counts": dict(wc),
                "pair_AB_personal_counts": dict(votes[("A", "B")]),
                "subjectivity_counts": dict(collections.Counter(
                    assessment["subjectivity"]
                    for assessment in item["metadata"]["assessments"]
                )),
                "topic": "first record; prompt asks about eating beef; no response/rationale text copied",
            }
    result = {
        "scope": "descriptive human-rank analysis of CoVal; synthetic prompts and model candidates, no learner/user labels, no demographics or free-text rationales exported",
        "source_sha256": manifest["files"]["comparisons.jsonl"]["sha256"],
        "prompts": prompt_count,
        "assessments": assessment_count,
        "assessments_missing_personal_rank_in_file": assessments_missing_personal,
        "prompts_with_at_least_two_personal_strict_tops": prompts_with_two_personal_ranks,
        "personal_and_world_strict_tops_both": personal_world_strict_both,
        "personal_to_world_top_changed_same_annotator": personal_world_changed,
        "prompts_with_multiple_personal_top_choices": prompt_multiple_personal_tops,
        "prompts_with_multiple_world_top_choices": prompt_multiple_world_tops,
        "prompts_personal_vs_world_unique_majority_changed": prompt_majority_changed,
        "personal_pair_comparisons_with_two_or_more_non_tie_votes": pair_decisions,
        "pairs_with_both_preferences_observed": pair_with_both_sides,
        "pair_majority_at_most_60_percent": close_pairs,
        "pair_majority_strength_mean": sum(pair_majority_strength) / len(pair_majority_strength),
        "world_pair_comparisons_with_two_or_more_non_tie_votes": world_pair_decisions,
        "world_pairs_with_both_preferences_observed": world_pairs_with_both_sides,
        "world_pair_majority_at_most_60_percent": world_close_pairs,
        "assessment_subjectivity_counts": dict(subjectivity),
        "prompts_with_multiple_personal_tops_by_majority_subjectivity": dict(subjectivity_disagreement),
        "example_without_text": fixed_example,
    }
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("prompts", prompt_count, "assessments", assessment_count,
          "personal_top_conflict", prompt_multiple_personal_tops,
          "pair_both_sides", pair_with_both_sides, "/", pair_decisions)
 
 
if __name__ == "__main__":
    main()
