"""用本机训练器的同一函数比较四轮与假设八轮计划的学习率。"""
 
from __future__ import annotations
 
import hashlib
import json
import math
from pathlib import Path
 
import matplotlib
 
matplotlib.use("Agg")
import matplotlib.pyplot as plt
 
from train_causal_wikitext import TrainConfig, learning_rate
 
 
WORK = Path(__file__).resolve().parents[1]
MANIFEST = WORK / "data" / "wikitext2_causal" / "manifest.json"
RESULT = WORK / "results" / "learning_rate_plan_probe.json"
FIGURE = WORK / "figures" / "causal_lm_learning_rate_plan.png"
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def main() -> None:
    manifest = json.loads(MANIFEST.read_text(encoding="utf-8"))
    cfg = TrainConfig()
    blocks = manifest["splits"]["train"]["blocks"]
    steps_per_epoch = math.ceil(blocks / cfg.batch_size)
    plans = {"four_epochs_executed": steps_per_epoch * 4,
             "eight_epochs_counterfactual_only": steps_per_epoch * 8}
    inspect_steps = [1, 50, 100, 334, 335, 668, 1336, 2672]
    values = {name: {str(step): learning_rate(step, total, cfg) for step in inspect_steps if step <= total}
              for name, total in plans.items()}
    if steps_per_epoch != 334 or values["four_epochs_executed"]["1336"] != cfg.final_lr:
        raise AssertionError("当前训练块数或学习率端点与本章计划不符")
 
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    fig, ax = plt.subplots(figsize=(8.3, 4.2), layout="constrained")
    for name, total in plans.items():
        steps = range(1, total + 1)
        rates = [learning_rate(step, total, cfg) for step in steps]
        label = "实际四轮计划" if name == "four_epochs_executed" else "假设八轮计划（未训练）"
        ax.plot(steps, rates, label=label, linewidth=2)
    ax.axvline(334, color="#888888", linestyle="--", linewidth=1, label="第一轮结束·334步")
    ax.set(xlabel="全局更新步数", ylabel="学习率", title="恢复时总计划不能无声改变")
    ax.legend()
    FIGURE.parent.mkdir(exist_ok=True)
    fig.savefig(FIGURE, dpi=180)
    plt.close(fig)
    report = {
        "c18_manifest_sha256": sha256(MANIFEST),
        "train_blocks": blocks,
        "batch_size_blocks": cfg.batch_size,
        "steps_per_epoch": steps_per_epoch,
        "warmup_steps": cfg.warmup_steps,
        "peak_lr": cfg.peak_lr,
        "final_lr": cfg.final_lr,
        "plans_total_steps": plans,
        "selected_step_learning_rates": values,
        "scope": "直接调用真实C19训练器的learning_rate函数；八轮曲线只是假设计划，不是已运行八轮模型的成绩。",
    }
    RESULT.parent.mkdir(exist_ok=True)
    RESULT.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"steps_per_epoch": steps_per_epoch, "step_335_four": values["four_epochs_executed"]["335"],
                      "step_335_eight": values["eight_epochs_counterfactual_only"]["335"]}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
