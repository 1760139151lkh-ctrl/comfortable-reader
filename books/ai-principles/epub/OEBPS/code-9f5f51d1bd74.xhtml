"""Fetch the pinned UCI OULAD archive and extract only required CSV members."""
from __future__ import annotations
 
import hashlib
import json
import shutil
import zipfile
from pathlib import Path
 
import requests
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/c47_oulad"
URL = "https://archive.ics.uci.edu/static/public/349/open+university+learning+analytics+dataset.zip"
ARCHIVE = DATA / "oulad_uci_349.zip"
ALLOWED = {"studentInfo.csv", "studentVle.csv", "courses.csv", "assessments.csv",
           "studentAssessment.csv", "studentRegistration.csv", "vle.csv", "OULAD.names"}
NEEDED = {"studentInfo.csv", "studentVle.csv", "studentRegistration.csv", "courses.csv", "OULAD.names"}
EXPECTED_ARCHIVE_SHA256 = "f2ed1902616c1fe8d2824d872c0b7d2d72be435bf0124d077044fe4be2c6d3e4"
MAX_BYTES = 100_000_000
 
 
def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def main() -> None:
    DATA.mkdir(parents=True, exist_ok=True)
    if not ARCHIVE.exists():
        response = requests.get(URL, stream=True, timeout=90)
        response.raise_for_status()
        temp = DATA / "oulad_download.part"
        if temp.exists():
            raise FileExistsError(temp)
        total = 0
        with temp.open("wb") as out:
            for block in response.iter_content(chunk_size=1 << 20):
                if not block:
                    continue
                total += len(block)
                if total > MAX_BYTES:
                    raise ValueError("Unexpected OULAD archive size")
                out.write(block)
        temp.replace(ARCHIVE)
    digest = sha(ARCHIVE)
    if digest != EXPECTED_ARCHIVE_SHA256:
        raise ValueError(f"UCI archive changed: {digest}")
    output = {"archive": str(ARCHIVE.relative_to(ROOT)).replace("\\", "/"),
              "bytes": ARCHIVE.stat().st_size, "sha256": digest, "members": {}}
    with zipfile.ZipFile(ARCHIVE) as archive:
        infos = archive.infolist()
        if {i.filename for i in infos} != ALLOWED:
            raise ValueError("Unexpected ZIP member set")
        if archive.testzip() is not None:
            raise ValueError("ZIP member CRC failed")
        for info in infos:
            if info.filename not in NEEDED:
                continue
            if info.file_size > 500_000_000 or "/" in info.filename or "\\" in info.filename:
                raise ValueError("Unsafe member")
            target = DATA / info.filename
            if not target.exists():
                with archive.open(info) as source, target.open("wb") as sink:
                    shutil.copyfileobj(source, sink)
            if target.stat().st_size != info.file_size:
                raise ValueError(f"Member size changed: {info.filename}")
            output["members"][info.filename] = {"bytes": target.stat().st_size, "sha256": sha(target)}
    manifest = DATA / "manifest.json"
    if manifest.exists():
        old = json.loads(manifest.read_text(encoding="utf-8"))
        if (old["archive"] != output["archive"] or old["bytes"] != output["bytes"]
                or old["sha256"] != output["sha256"]
                or any(output["members"].get(name) != row for name, row in old["members"].items())):
            raise ValueError("Existing manifest differs from current source bytes")
        if old != output:
            manifest.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    else:
        manifest.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(output, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
