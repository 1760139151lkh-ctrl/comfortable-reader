"""Q2 evaluation after a documented time/quality-only calibration amendment."""
from __future__ import annotations
 
import argparse
import importlib.util
import json
from pathlib import Path
 
import numpy as np
 
source_path = Path(__file__).with_name("c45_kepler_transit.py")
spec = importlib.util.spec_from_file_location("c45_kepler_transit", source_path)
assert spec is not None and spec.loader is not None
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
FILES, load, metrics, sha = module.FILES, module.load, module.metrics, module.sha
 
 
def main(run_dir: Path) -> None:
    q1_path = run_dir / "q1_train_validation.json"
    q2_path = run_dir / "q2_first_test.json"
    if q2_path.exists():
        raise FileExistsError("Q2 first result already exists")
    q1 = json.loads(q1_path.read_text(encoding="utf-8"))
    original_code = Path(__file__).with_name("c45_kepler_transit.py")
    if sha(original_code) != q1["code_sha256"] or sha(FILES["q1"]) != q1["q1"]["file_sha256"]:
        raise RuntimeError("Frozen Q1 training code or data changed")
    t, f, e, meta = load("q2")
    calibration = t < t[0] + 4.0
    if int(calibration.sum()) < 50:
        raise ValueError("Amended Q2 calibration still lacks 50 good points")
    scale = float(np.median(f[calibration]))
    output = {
        "scope": "First Q2 evaluation with documented four-day quality-coverage amendment; Q1 models unchanged",
        "pretest_decision": "work/verification/C45_kepler_pretest_decision.md",
        "q1_report_sha256": sha(q1_path), "q1_code_sha256": q1["code_sha256"],
        "q2_evaluator_sha256": sha(Path(__file__)), "q2": meta,
        "calibration_days": 4.0, "calibration_n": int(calibration.sum()),
        "calibration_median_e_per_s": scale,
        "scored_time_range_bkjd": [float(t[~calibration][0]), float(t[-1])],
        "metrics": metrics(t[~calibration], f[~calibration] / scale, q1["models"]),
    }
    q2_path.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(output, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--run-dir", type=Path, required=True)
    args = ap.parse_args()
    main(args.run_dir)
