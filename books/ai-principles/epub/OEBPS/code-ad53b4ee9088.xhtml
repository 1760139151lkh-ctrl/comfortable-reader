"""第三章：把一条隐藏路径的解析导数与有限差分对上。"""
from __future__ import annotations
import math
 
x1 = 1.0
target = 1.0
v = -2.0
b = 1.2
a = math.atanh(0.5)  # 让当前隐藏活动 tanh(a*x1) 恰好为 0.5
step = 1e-6
 
 
def loss(at_weight):
    hidden = math.tanh(at_weight * x1)
    score = b + v * hidden
    return 0.5 * (score - target) ** 2
 
 
hidden = math.tanh(a * x1)
score = b + v * hidden
residual = score - target
analytic = residual * v * (1.0 - hidden ** 2) * x1
finite_difference = (loss(a + step) - loss(a - step)) / (2 * step)
 
print("隐藏活动、末端分数和残差:", hidden, score, residual)
print("链式法则给出的导数:", analytic)
print("有限差分给出的近似:", finite_difference)
print("步长 0.1 的更新方向:", a - 0.1 * analytic)
assert abs(analytic - finite_difference) < 1e-8
