"""Compare real speech and non-speech waveforms at the same sample rate."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
from scipy.signal import resample_poly
import soundfile as sf
 
from c32_audio_features import SAMPLE_RATE, complex_stft
 
 
ROOT = Path(__file__).resolve().parents[2]
SPEECH = ROOT / "work/data/librispeech_dev_clean/chapter_split.json"
EVENTS = ROOT / "work/data/esc10_33c8ce9/chapter_features/manifest.json"
 
 
def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path, required=True)
    parser.add_argument("--receipt", type=Path, required=True)
    args = parser.parse_args()
    out = args.out if args.out.is_absolute() else ROOT / args.out
    receipt = args.receipt if args.receipt.is_absolute() else ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing sound comparison artifacts")
    speech = json.loads(SPEECH.read_text(encoding="utf-8"))
    events = json.loads(EVENTS.read_text(encoding="utf-8"))
    speech_row = next(row for row in speech["rows"] if row["id"] == "1272-128104-0000")
    dog_row = next(row for row in events["rows"] if row["split"] == "train" and row["category"] == "dog")
    rain_row = next(row for row in events["rows"] if row["split"] == "train" and row["category"] == "rain")
    wave_speech, sr = sf.read(ROOT / speech_row["flac"], dtype="float32")
    if sr != SAMPLE_RATE:
        raise ValueError("speech sample rate changed")
    chosen = [("朗读语音：目标是一串字", wave_speech, speech_row["id"])]
    for title, row in (("狗叫：目标是十选一的事件类别", dog_row),
                       ("下雨：目标是十选一的事件类别", rain_row)):
        wave, original_rate = sf.read(ROOT / row["source_wav"], dtype="float32")
        if original_rate != 44100:
            raise ValueError("ESC original sample rate changed")
        chosen.append((title, resample_poly(wave, 160, 441), row["id"]))
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    fig, axes = plt.subplots(3, 1, figsize=(11, 9), layout="constrained")
    for axis, (title, wave, _) in zip(axes, chosen):
        freqs, times, spectrum = complex_stft(wave.astype(np.float32))
        power = 20 * np.log10(np.maximum(np.abs(spectrum), 1e-7))
        relative = power - power.max()
        axis.pcolormesh(times, freqs, relative, shading="auto", cmap="magma",
                        vmin=-65, vmax=0)
        axis.set(ylim=(0, 4000), xlabel="时间（秒）", ylabel="频率（Hz）",
                 title=title + "；每段单独把最强点定为 0 dB")
    out.parent.mkdir(parents=True, exist_ok=True)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=150)
    plt.close(fig)
    report = {
        "speech_split_sha256": sha256(SPEECH),
        "event_feature_manifest_sha256": sha256(EVENTS),
        "all_examples_from_training_splits": True,
        "speech_id": speech_row["id"],
        "speech_transcript": speech_row["transcript"],
        "event_examples": [{"id": row["id"], "category": row["category"],
                            "source_recording": row["source_recording"]}
                           for row in (dog_row, rain_row)],
        "same_display_sample_rate": SAMPLE_RATE,
        "event_resampling": "scipy.signal.resample_poly 160/441 with antialias filter",
        "plot_magnitude": "per-clip relative dB, cannot compare absolute loudness between panels",
        "figure_sha256": sha256(out),
    }
    receipt.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n",
                       encoding="utf-8")
    print(json.dumps({"speech_id": speech_row["id"],
                      "dog_id": dog_row["id"], "rain_id": rain_row["id"]}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
