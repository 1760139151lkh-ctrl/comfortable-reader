"""Evaluate validation-selected Cora arms once on held-out paper labels."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from collections import Counter
from pathlib import Path
 
import numpy as np
import torch
from torch.nn import functional as F
 
from c36_cora_node_models import DATA, NodeNet, load_data, normalized_adjacency
 
 
@torch.inference_mode()
def evaluate(model, x, y, ids, adjacency, categories):
    model.eval()
    logits = model(x, adjacency)[ids]
    correct = logits.argmax(-1) == y[ids]
    per_class = {categories[k]: {"nodes": int((y[ids] == k).sum()), "accuracy": float(correct[y[ids] == k].float().mean())} for k in range(len(categories))}
    return {"nodes": len(ids), "nll": float(F.cross_entropy(logits, y[ids])), "accuracy": float(correct.float().mean()), "per_class": per_class}
 
 
def homophily(edges, labels):
    return {"same_label_pairs": int((labels[edges[:, 0]] == labels[edges[:, 1]]).sum()), "pairs": len(edges), "fraction": float((labels[edges[:, 0]] == labels[edges[:, 1]]).mean())}
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--mlp-ckpt", type=Path, default=Path("work/runs/c36_cora_mlp250/best.pt"))
    p.add_argument("--gcn-ckpt", type=Path, default=Path("work/runs/c36_cora_gcn250/best.pt"))
    p.add_argument("--rewired-ckpt", type=Path, default=Path("work/runs/c36_cora_rewired250/best.pt"))
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("test report exists")
    torch.set_num_threads(8)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    split = json.loads((DATA / "split.json").read_text(encoding="utf-8"))
    test = torch.tensor(split["test"], device=device)
    if len(test) != 1000:
        raise RuntimeError("test changed")
    labels_np = np.load(DATA / "labels_int64.npy")
    original = np.load(DATA / "undirected_unique_edges_int64.npy")
    rewired = np.load(DATA / "degree_preserving_rewired_edges_int64.npy")
    report = {"identity": "LINQS Cora original source with author-internal 140/500/1000 node-label split; all node words and citation edges visible transductively", "test_labels": len(test), "training_labels": 140, "validation_labels": 500, "remaining_labels_unused": 1068, "graph_input": {"real_undirected_model_edges": len(original), "source_directed_citation_rows": 5429, "real_edge_same_label_after_test": homophily(original, labels_np), "rewired_edge_same_label_after_test": homophily(rewired, labels_np)}, "models": {}}
    for name, path in (("mlp", a.mlp_ckpt), ("gcn", a.gcn_ckpt), ("rewired", a.rewired_ckpt)):
        ckpt = torch.load(path, map_location="cpu", weights_only=True)
        x, y, _, _, adjacency = load_data(name, device, include_test_labels=True)
        model = NodeNet(name).to(device)
        model.load_state_dict(ckpt["model"])
        item = {"validation_selected_epoch": ckpt["epoch"], "checkpoint_sha256": hashlib.sha256(path.read_bytes()).hexdigest(), "test": evaluate(model, x, y, test, adjacency, split["categories_in_label_index_order"])}
        if name == "gcn":
            wrong_adj = normalized_adjacency(rewired, len(x), device)
            item["same_trained_weights_wrong_edges_diagnostic"] = evaluate(model, x, y, test, wrong_adj, split["categories_in_label_index_order"])
        report["models"][name] = item
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"homophily": report["graph_input"], "models": {k:{"epoch":v["validation_selected_epoch"],"test": {key:value for key,value in v["test"].items() if key!='per_class'}} for k,v in report["models"].items()}, "gcn_wrong_edges": {key:value for key,value in report["models"]["gcn"]["same_trained_weights_wrong_edges_diagnostic"].items() if key!='per_class'}}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
