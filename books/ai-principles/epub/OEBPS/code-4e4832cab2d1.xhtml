"""Evaluate frozen single/mean/ordered models on Penn's official action split."""
 
from __future__ import annotations
 
import argparse
import collections
import json
from pathlib import Path
 
import numpy as np
import torch
 
from c33_penn_action_models import ActionModel, MODES, batch_tensor, sha256
from prepare_c33_action_frames import decode_frame
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/penn_action_original"
ACTION = DATA / "action_8frames_v1"
 
 
def test_frames(rows: list[dict]) -> np.ndarray:
    clips = []
    for row in rows:
        indexes = np.rint(np.linspace(0, row["nframes"] - 1, 8)).astype(int)
        clips.append(np.stack([decode_frame(row, int(index)) for index in indexes]))
    return np.stack(clips)
 
 
@torch.no_grad()
def evaluate_one(model: ActionModel, frames: np.ndarray, rows: list[dict],
                 names: list[str], device: torch.device) -> dict:
    model.eval()
    lookup = {name: i for i, name in enumerate(names)}
    matrix = np.zeros((len(names), len(names)), dtype=np.int64)
    loss_sum = 0.0
    details = []
    for start in range(0, len(rows), 16):
        indexes = np.arange(start, min(start + 16, len(rows)))
        x = batch_tensor(frames, indexes, device)
        logits = model(x)
        probs = torch.softmax(logits, dim=1).cpu().numpy()
        for local, index in enumerate(indexes):
            truth = lookup[rows[index]["action"]]
            predicted = int(probs[local].argmax())
            matrix[truth, predicted] += 1
            loss_sum -= float(np.log(max(probs[local, truth], 1e-12)))
            details.append({"video_id": rows[index]["id"],
                            "true_action": rows[index]["action"],
                            "predicted_action": names[predicted]})
    correct = int(np.trace(matrix))
    class_accuracy = {name: {"correct": int(matrix[i, i]),
                             "count": int(matrix[i].sum()),
                             "accuracy": float(matrix[i, i] / matrix[i].sum())}
                      for i, name in enumerate(names)}
    return {"correct": correct, "count": len(rows),
            "accuracy": correct / len(rows),
            "mean_negative_log_probability": loss_sum / len(rows),
            "macro_class_accuracy": float(np.mean([v["accuracy"] for v in class_accuracy.values()])),
            "per_class": class_accuracy,
            "confusion_true_rows_predicted_columns": matrix.tolist(),
            "predictions": details}
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path, required=True)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    run = args.run_dir if args.run_dir.is_absolute() else ROOT / args.run_dir
    out = args.out if args.out.is_absolute() else ROOT / args.out
    if out.exists():
        raise FileExistsError(f"preserve existing action evaluation: {out}")
    action_manifest = ACTION / "manifest.json"
    prepared = json.loads(action_manifest.read_text(encoding="utf-8"))
    training = json.loads((run / "train.json").read_text(encoding="utf-8"))
    if (training["data_manifest_sha256"] != sha256(action_manifest) or
            training["official_test_accessed"]):
        raise RuntimeError("action training/data mismatch")
    inventory_file = DATA / "inventory.json"
    inventory = json.loads(inventory_file.read_text(encoding="utf-8"))
    if prepared["inventory_sha256"] != sha256(inventory_file):
        raise RuntimeError("Penn official inventory changed")
    rows = [row for row in inventory["rows"] if row["official_split"] == "official_test"]
    if len(rows) != 1068:
        raise RuntimeError("Penn test count changed")
    frames = test_frames(rows)
    torch.set_num_threads(4)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    results = {}
    for mode, record in zip(MODES, training["results"]):
        checkpoint_path = run / (mode + "_best.pt")
        if record["best_checkpoint_sha256"] != sha256(checkpoint_path):
            raise RuntimeError(f"selected {mode} checkpoint changed")
        checkpoint = torch.load(checkpoint_path, map_location="cpu", weights_only=False)
        if checkpoint["data_manifest_sha256"] != sha256(action_manifest) or checkpoint["mode"] != mode:
            raise RuntimeError(f"{mode} checkpoint source mismatch")
        model = ActionModel(mode, len(prepared["class_names"])).to(device)
        model.load_state_dict(checkpoint["model"])
        results[mode] = evaluate_one(model, frames, rows, prepared["class_names"], device)
        print("action test", mode, results[mode]["correct"], "/", len(rows), flush=True)
    report = {
        "identity": "first action-class evaluation after all three models fixed on internal validation; official Penn test frames previously accessed by a separate author-derived wrist task, so not globally unseen source",
        "source_archive_sha256": inventory["source_archive_sha256"],
        "inventory_sha256": sha256(inventory_file),
        "action_manifest_sha256": sha256(action_manifest),
        "train_report_sha256": sha256(run / "train.json"),
        "official_test_video_count": len(rows),
        "class_names": prepared["class_names"],
        "modes": results,
        "limits": "15 manually trimmed action types; different model parameters/compute; an action label does not provide per-frame wrist track or future-frame video generation",
    }
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n",
                   encoding="utf-8")
    print(json.dumps({mode: results[mode]["accuracy"] for mode in MODES},
                     ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
