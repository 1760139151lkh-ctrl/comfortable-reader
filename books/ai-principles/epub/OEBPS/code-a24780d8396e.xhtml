"""Check C19 full-prefix logits against this exact model's layerwise KV cache."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
 
from c37_cached_decode import DATA, SERVING, cache_bytes, decode_one, full_logits, generate, load_serving, prefill
from train_causal_wikitext import BOS
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("receipt exists")
    torch.set_num_threads(8)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model, tokenizer, state = load_serving(device)
    train = np.load(DATA / "train_blocks.npz")
    sample = torch.as_tensor(train["input_ids"][0], dtype=torch.long, device=device)
    data = {"device": str(device), "serving_sha256": hashlib.sha256(SERVING.read_bytes()).hexdigest(), "checkpoint_source_sha256": state["source_checkpoint_sha256"], "context": model.cfg.context, "checks": []}
    with torch.inference_mode():
        for prefix_length, increments in ((16, 16), (64, 16), (120, 8), (127, 1), (128, 0)):
            ids = sample[:prefix_length][None]
            cached_logits, cache = prefill(model, ids)
            prefix_error = float((cached_logits - full_logits(model, ids)).abs().max())
            max_step_error = 0.0
            for j in range(increments):
                next_id = sample[prefix_length + j][None]
                cached_logits, cache = decode_one(model, next_id, cache)
                expected = full_logits(model, sample[:prefix_length + j + 1][None])
                max_step_error = max(max_step_error, float((cached_logits - expected).abs().max()))
            data["checks"].append({"prefix": prefix_length, "incremental_steps_without_eviction": increments, "prefill_max_logit_difference": prefix_error, "decode_max_logit_difference": max_step_error, "cache_bytes_last_step": cache_bytes(cache), "analytical_float32_cache_bytes_last_step": 2 * model.cfg.layers * model.cfg.width * (prefix_length + increments) * 4})
        # A tempting but incorrect rolling cache discards only old K/V. C19
        # rebases absolute position encodings and earlier hidden states when
        # its sliding 128-token window advances.
        _, full_cache = prefill(model, sample[:128][None])
        truncated = [(k[:, :, 1:].clone(), v[:, :, 1:].clone()) for k, v in full_cache]
        wrong_logits, _ = decode_one(model, torch.as_tensor(train["input_ids"][1, 0:1], dtype=torch.long, device=device), truncated)
        extended = torch.cat((sample[:128], torch.as_tensor(train["input_ids"][1, 0:1], dtype=torch.long, device=device)))
        correct_logits = full_logits(model, extended[None])
        data["wrong_evict_only_max_logit_difference"] = float((wrong_logits - correct_logits).abs().max())
        text_prefix = [BOS] + tokenizer.encode("The city of", add_special_tokens=False).ids
        short_full = generate(model, text_prefix, 16, "full")
        short_cached = generate(model, text_prefix, 16, "cached")
        long_tokens = sample[:120].cpu().tolist()
        long_full = generate(model, long_tokens, 16, "full")
        long_cached = generate(model, long_tokens, 16, "cached")
        data["greedy_short_full_equals_cache"] = short_full["new_token_ids"] == short_cached["new_token_ids"]
        data["greedy_long_window_full_equals_rebuilding_cache"] = long_full["new_token_ids"] == long_cached["new_token_ids"]
        data["short_generation"] = {"prompt": "The city of", "full": short_full, "cached": short_cached, "decoded": tokenizer.decode([n for n in short_cached["new_token_ids"] if n != 2], skip_special_tokens=False)}
        data["long_generation_rebuilds"] = long_cached["context_rebuilds_after_window_overflow"]
    worst = max(max(x["prefill_max_logit_difference"], x["decode_max_logit_difference"]) for x in data["checks"])
    if worst > 1e-4 or not data["greedy_short_full_equals_cache"] or not data["greedy_long_window_full_equals_rebuilding_cache"]:
        raise RuntimeError("cached implementation does not agree with C19 full-window model")
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"max_logit_difference": worst, "wrong_evict_only_max_logit_difference": data["wrong_evict_only_max_logit_difference"], "short_same": data["greedy_short_full_equals_cache"], "long_same": data["greedy_long_window_full_equals_rebuilding_cache"], "long_rebuilds": data["long_generation_rebuilds"]}))
 
 
if __name__ == "__main__":
    main()
