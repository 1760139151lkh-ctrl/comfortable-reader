"""用马尔可夫1913年报告的元音/辅音计数核一项长序列比例平衡。"""
 
from __future__ import annotations
 
import argparse
import json
from fractions import Fraction
from pathlib import Path
 
 
WORK = Path(__file__).resolve().parents[1]
DEFAULT = WORK / "results" / "c21_markov_vowel_balance.json"
 
 
def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--out", type=Path, default=DEFAULT)
    args = parser.parse_args()
    output = args.out if args.out.is_absolute() else WORK.parent / args.out
    if output.exists():
        raise FileExistsError(f"结果已存在，请另选输出路径：{output}")
    vowel_count, total_letters = 8638, 20000
    after_vowel_vowel = Fraction(1104, 8638)
    after_consonant_vowel = Fraction(7534, 11361)
    balanced_vowel_fraction = after_consonant_vowel / (1 - after_vowel_vowel + after_consonant_vowel)
    observed_vowel_fraction = Fraction(vowel_count, total_letters)
    if abs(float(balanced_vowel_fraction - observed_vowel_fraction)) > 0.0001:
        raise AssertionError("历史计数与两类相邻概率的比例平衡不相称")
    report = {
        "source": "Markov 1913 Eugene Onegin study, 2006 English translation pp.591-597",
        "reported_vowel_count": vowel_count,
        "reported_total_letters": total_letters,
        "reported_vowel_after_vowel_count_ratio": "1104/8638",
        "reported_vowel_after_consonant_count_ratio": "7534/11361",
        "observed_overall_vowel_fraction": float(observed_vowel_fraction),
        "balance_equation": "p = P(V|V)*p + P(V|C)*(1-p)",
        "balance_solution_fraction": str(balanced_vowel_fraction),
        "balance_solution_decimal": float(balanced_vowel_fraction),
        "balance_minus_observed": float(balanced_vowel_fraction - observed_vowel_fraction),
        "limit": "由长序列相邻计数的频率平衡得到，不独立证明只看前一字母就是充分的一步马尔可夫模型；边界字母和原文四舍五入均会影响末位。",
    }
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(report, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
