"""同一原始图／固定乱序图上，从随机初值训练全连接和卷积分类器。"""
 
from __future__ import annotations
 
import csv
import hashlib
import json
import platform
import time
from pathlib import Path
 
import matplotlib
 
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import torch
from torch import nn
from torch.nn import functional as F
 
from prepare_fashion_mnist import DATA, WORK, load_images, load_labels
 
 
SEED = 20260924
EPOCHS = 5
BATCH_SIZE = 256
LEARNING_RATE = 0.001
ARMS = ("mlp_original", "mlp_permuted", "cnn_original", "cnn_permuted")
RESULT = WORK / "results" / "fashion_spatial_training.json"
PREDICTIONS = WORK / "results" / "fashion_test_predictions.csv"
FIGURE = WORK / "figures" / "fashion_spatial_training.png"
 
 
class MLP(nn.Module):
    def __init__(self) -> None:
        super().__init__()
        self.net = nn.Sequential(nn.Flatten(), nn.Linear(784, 128), nn.ReLU(), nn.Linear(128, 10))
 
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.net(x)
 
 
class CNN(nn.Module):
    def __init__(self) -> None:
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(1, 16, 3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(16, 32, 3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
        )
        self.classifier = nn.Sequential(nn.Flatten(), nn.Linear(32 * 7 * 7, 64), nn.ReLU(), nn.Linear(64, 10))
 
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.classifier(self.features(x))
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def load_verified_data(device: torch.device) -> dict:
    manifest = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    for name, meta in manifest["files"].items():
        if sha256(DATA / name) != meta["sha256"]:
            raise ValueError(f"数据原包字节不符：{name}")
    split_path = DATA / "chapter_split.npz"
    if sha256(split_path) != manifest["split_file_sha256"]:
        raise ValueError("固定划分或置换文件字节身份改变")
    with np.load(split_path) as prepared:
        train_indices = prepared["train_indices"].copy()
        validation_indices = prepared["validation_indices"].copy()
        permutation = prepared["permutation"].copy()
    if not np.array_equal(np.sort(permutation), np.arange(784)):
        raise ValueError("像素置换不完整")
    if len(train_indices) != 50_000 or len(validation_indices) != 10_000:
        raise ValueError("训练／验证行数不符")
 
    original_training_images = load_images(DATA / "train-images-idx3-ubyte.gz", 60_000)
    official_test_images = load_images(DATA / "t10k-images-idx3-ubyte.gz", 10_000)
    original_training_labels = load_labels(DATA / "train-labels-idx1-ubyte.gz", 60_000)
    official_test_labels = load_labels(DATA / "t10k-labels-idx1-ubyte.gz", 10_000)
 
    x_all = torch.from_numpy(original_training_images).to(device=device, dtype=torch.float32).unsqueeze(1) / 255.0
    y_all = torch.from_numpy(original_training_labels.astype(np.int64)).to(device)
    x_test = torch.from_numpy(official_test_images).to(device=device, dtype=torch.float32).unsqueeze(1) / 255.0
    y_test = torch.from_numpy(official_test_labels.astype(np.int64)).to(device)
    perm_index = torch.from_numpy(permutation.astype(np.int64)).to(device)
    x_all_permuted = x_all.flatten(1).index_select(1, perm_index).view(-1, 1, 28, 28)
    x_test_permuted = x_test.flatten(1).index_select(1, perm_index).view(-1, 1, 28, 28)
    assert torch.equal(torch.sort(x_all[0].flatten()).values, torch.sort(x_all_permuted[0].flatten()).values)
    return {
        "manifest": manifest,
        "train_indices": train_indices,
        "validation_indices": validation_indices,
        "permutation": permutation,
        "x_all": x_all,
        "x_all_permuted": x_all_permuted,
        "y_all": y_all,
        "x_test": x_test,
        "x_test_permuted": x_test_permuted,
        "y_test": y_test,
        "example_original": original_training_images[int(train_indices[0])],
        "example_permuted": original_training_images[int(train_indices[0])].flatten()[permutation].reshape(28, 28),
        "example_label": int(original_training_labels[int(train_indices[0])]),
    }
 
 
@torch.no_grad()
def evaluate(model: nn.Module, images: torch.Tensor, labels: torch.Tensor, indices: np.ndarray | None = None) -> dict:
    model.eval()
    if indices is None:
        indices = np.arange(len(labels))
    total_loss, correct = 0.0, 0
    predictions = []
    for start in range(0, len(indices), BATCH_SIZE):
        batch_ids = torch.as_tensor(indices[start:start + BATCH_SIZE], device=images.device, dtype=torch.long)
        scores = model(images.index_select(0, batch_ids))
        targets = labels.index_select(0, batch_ids)
        total_loss += float(F.cross_entropy(scores, targets, reduction="sum").item())
        guessed = scores.argmax(dim=1)
        correct += int((guessed == targets).sum().item())
        predictions.extend(guessed.cpu().tolist())
    return {
        "loss": total_loss / len(indices),
        "accuracy": correct / len(indices),
        "correct": correct,
        "total": len(indices),
        "predictions": predictions,
    }
 
 
def image_shift(images: torch.Tensor) -> torch.Tensor:
    shifted = torch.zeros_like(images)
    shifted[:, :, 2:, 2:] = images[:, :, :-2, :-2]
    return shifted
 
 
def make_model(arm: str, device: torch.device) -> nn.Module:
    torch.manual_seed(SEED)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED)
    return (MLP() if arm.startswith("mlp") else CNN()).to(device)
 
 
def train_one_arm(arm: str, data: dict, orders: list[np.ndarray], device: torch.device) -> dict:
    model = make_model(arm, device)
    images = data["x_all_permuted"] if arm.endswith("permuted") else data["x_all"]
    labels = data["y_all"]
    optimizer = torch.optim.Adam(model.parameters(), lr=LEARNING_RATE)
    train_indices = data["train_indices"]
    validation_indices = data["validation_indices"]
    best_loss, best_epoch = float("inf"), None
    best_state = None
    epochs = []
    if device.type == "cuda":
        torch.cuda.reset_peak_memory_stats(device)
        torch.cuda.synchronize(device)
    start_time = time.perf_counter()
    for epoch, order in enumerate(orders, start=1):
        model.train()
        loss_sum, correct = 0.0, 0
        epoch_indices = train_indices[order]
        for begin in range(0, len(epoch_indices), BATCH_SIZE):
            ids = torch.as_tensor(epoch_indices[begin:begin + BATCH_SIZE], device=device, dtype=torch.long)
            inputs = images.index_select(0, ids)
            targets = labels.index_select(0, ids)
            optimizer.zero_grad(set_to_none=True)
            logits = model(inputs)
            loss = F.cross_entropy(logits, targets)
            loss.backward()
            optimizer.step()
            batch_n = len(ids)
            loss_sum += float(loss.detach().item()) * batch_n
            correct += int((logits.detach().argmax(dim=1) == targets).sum().item())
        validation = evaluate(model, images, labels, validation_indices)
        epoch_result = {
            "epoch": epoch,
            "train_loss": loss_sum / len(train_indices),
            "train_accuracy": correct / len(train_indices),
            "validation_loss": validation["loss"],
            "validation_accuracy": validation["accuracy"],
            "validation_correct": validation["correct"],
        }
        epochs.append(epoch_result)
        print(f"{arm} epoch {epoch}: train loss {epoch_result['train_loss']:.4f}, "
              f"val loss {validation['loss']:.4f}, val acc {validation['accuracy']:.4f}", flush=True)
        if validation["loss"] < best_loss:
            best_loss, best_epoch = validation["loss"], epoch
            best_state = {name: value.detach().cpu().clone() for name, value in model.state_dict().items()}
    if device.type == "cuda":
        torch.cuda.synchronize(device)
    seconds = time.perf_counter() - start_time
    if best_state is None or best_epoch is None:
        raise AssertionError("没有可选的验证模型")
    checkpoint = WORK / "results" / f"fashion_{arm}_best.pt"
    torch.save(
        {
            "state_dict": best_state,
            "arm": arm,
            "chosen_epoch": best_epoch,
            "data_split_sha256": data["manifest"]["split_file_sha256"],
            "seed": SEED,
        },
        checkpoint,
    )
    # 重新载入的核对仅使用验证行；官方测试到四臂训练全部结束后才触碰。
    saved = torch.load(checkpoint, map_location="cpu", weights_only=True)
    reloaded = (MLP() if arm.startswith("mlp") else CNN())
    reloaded.load_state_dict(saved["state_dict"])
    reloaded = reloaded.to(device)
    model.load_state_dict(best_state)
    model.eval()
    reloaded.eval()
    one_validation_id = int(validation_indices[0])
    with torch.no_grad():
        expected = model(images[one_validation_id:one_validation_id + 1])
        actual = reloaded(images[one_validation_id:one_validation_id + 1])
    if not torch.equal(expected, actual):
        raise AssertionError("保存后重新载入的验证预测不一致")
    return {
        "arm": arm,
        "parameters": sum(parameter.numel() for parameter in model.parameters() if parameter.requires_grad),
        "epochs": epochs,
        "chosen_epoch": best_epoch,
        "chosen_validation_loss": best_loss,
        "checkpoint": str(checkpoint.relative_to(WORK)).replace("\\", "/"),
        "checkpoint_sha256": sha256(checkpoint),
        "train_seconds": seconds,
        "peak_cuda_allocated_mib_including_resident_data": (
            torch.cuda.max_memory_allocated(device) / 1024**2 if device.type == "cuda" else None
        ),
        "_selected_model": reloaded,
    }
 
 
def draw(data: dict, outcomes: dict) -> None:
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    figure, axes = plt.subplots(1, 3, figsize=(11.0, 3.7), layout="constrained")
    axes[0].imshow(data["example_original"], cmap="gray", vmin=0, vmax=255)
    axes[0].set_title(f"原图：类别 {data['example_label']}（踝靴）")
    axes[1].imshow(data["example_permuted"], cmap="gray", vmin=0, vmax=255)
    axes[1].set_title("像素值不变，位置固定乱序")
    for ax in axes[:2]:
        ax.set_xticks([])
        ax.set_yticks([])
    colors = {
        "mlp_original": "#4e79a7",
        "mlp_permuted": "#76b7b2",
        "cnn_original": "#e15759",
        "cnn_permuted": "#f28e2b",
    }
    display_names = {
        "mlp_original": "MLP 原图",
        "mlp_permuted": "MLP 乱序",
        "cnn_original": "CNN 原图",
        "cnn_permuted": "CNN 乱序",
    }
    for arm in ARMS:
        values = [entry["validation_accuracy"] for entry in outcomes[arm]["epochs"]]
        axes[2].plot(range(1, EPOCHS + 1), values, marker="o", ms=3, label=display_names[arm], color=colors[arm])
    axes[2].set(xlabel="训练轮次", ylabel="验证集准确率", xticks=range(1, EPOCHS + 1), ylim=(0.4, 1.0))
    axes[2].legend(fontsize=7)
    figure.savefig(FIGURE, dpi=180)
    plt.close(figure)
 
 
def main() -> None:
    if torch.cuda.is_available():
        device = torch.device("cuda")
        torch.backends.cudnn.benchmark = False
        torch.backends.cudnn.deterministic = True
    else:
        device = torch.device("cpu")
    data = load_verified_data(device)
    ordering_rng = np.random.default_rng(SEED + 77)
    orders = [ordering_rng.permutation(len(data["train_indices"])) for _ in range(EPOCHS)]
    outcomes = {arm: train_one_arm(arm, data, orders, device) for arm in ARMS}
 
    # 四臂训练和验证选择均已完成。此处才打开官方测试。
    test_labels = data["y_test"].cpu().numpy()
    test_predictions = {}
    for arm in ARMS:
        selected = outcomes[arm].pop("_selected_model")
        images = data["x_test_permuted"] if arm.endswith("permuted") else data["x_test"]
        test = evaluate(selected, images, data["y_test"])
        prediction = np.asarray(test.pop("predictions"), dtype=np.int64)
        confusion = np.zeros((10, 10), dtype=np.int64)
        np.add.at(confusion, (test_labels, prediction), 1)
        test["confusion_true_rows_predicted_columns"] = confusion.tolist()
        test["per_class_accuracy"] = (np.diag(confusion) / confusion.sum(axis=1)).tolist()
        if arm.endswith("original"):
            shifted = evaluate(selected, image_shift(data["x_test"]), data["y_test"])
            test["two_pixel_right_down_shift"] = {key: shifted[key] for key in ("loss", "accuracy", "correct", "total")}
        outcomes[arm]["official_test"] = test
        test_predictions[arm] = prediction
        print(f"{arm} selected epoch {outcomes[arm]['chosen_epoch']}: "
              f"official test {test['accuracy']:.4f}", flush=True)
 
    RESULT.parent.mkdir(exist_ok=True)
    FIGURE.parent.mkdir(exist_ok=True)
    with PREDICTIONS.open("w", newline="", encoding="utf-8") as stream:
        writer = csv.writer(stream)
        writer.writerow(["official_test_row_zero_based", "true_label", *ARMS])
        for index, label in enumerate(test_labels):
            writer.writerow([index, int(label), *[int(test_predictions[arm][index]) for arm in ARMS]])
    record = {
        "purpose": "four fixed architectures/input-order arms on 2017 Fashion-MNIST; no pretrained weights",
        "data_manifest": str((DATA / "manifest.json").relative_to(WORK)).replace("\\", "/"),
        "data_manifest_sha256": sha256(DATA / "manifest.json"),
        "official_source": data["manifest"]["official_readme"],
        "split_seed": data["manifest"]["split_seed"],
        "pixel_permutation_seed": data["manifest"]["permutation_seed"],
        "training_seed": SEED,
        "epochs": EPOCHS,
        "batch_size": BATCH_SIZE,
        "optimizer": "Adam",
        "learning_rate": LEARNING_RATE,
        "selection": "lowest validation cross-entropy; tie keeps earlier epoch",
        "normalization": "uint8 / 255, no fitted mean or std",
        "device": str(device),
        "gpu_name": torch.cuda.get_device_name(device) if device.type == "cuda" else None,
        "torch_version": torch.__version__,
        "python_version": platform.python_version(),
        "arms": outcomes,
        "test_predictions_csv": str(PREDICTIONS.relative_to(WORK)).replace("\\", "/"),
        "test_predictions_csv_sha256": sha256(PREDICTIONS),
        "limits": "one seed and five epochs, fixed architectures and optimizer; no natural-scene, detection or image-language evaluation",
    }
    RESULT.write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    draw(data, outcomes)
    print(f"saved {RESULT}, {PREDICTIONS}, {FIGURE}", flush=True)
 
 
if __name__ == "__main__":
    main()
