"""沿一条训练原文逐词核对上下文、目标、词表映射和两种概率。"""
 
from __future__ import annotations
 
import json
import math
from pathlib import Path
 
import numpy as np
import torch
from torch.nn import functional as F
 
from word_neural_lm import TwoWordLM
from word_ngram import InterpolatedKN, documents
 
 
WORK = Path(__file__).resolve().parents[1]
DATA = WORK / "data" / "wikitext2_raw"
RESULT = WORK / "results" / "word_source_trace.json"
 
 
def main() -> None:
    manifest = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    vocab = json.loads((DATA / "word_vocab.json").read_text(encoding="utf-8"))
    stoi = {word: index for index, word in enumerate(vocab)}
    source = manifest["first_training_body_line"]
    fields = source["split_tokens"]
    with np.load(DATA / "word_documents.npz") as prepared:
        train_docs = list(documents(prepared["train_tokens"], prepared["train_offsets"]))
    count_model = InterpolatedKN(train_docs, len(vocab))
    ngram_record = json.loads((WORK / "results" / "word_ngram.json").read_text(encoding="utf-8"))
    discount = float(ngram_record["selected_discount"])
    saved = torch.load(WORK / "results" / "word_neural_lm_best.pt", map_location="cpu", weights_only=True)
    if saved["vocab_sha256"] != manifest["vocab_sha256"]:
        raise ValueError("保存的神经模型词表身份不一致")
    neural = TwoWordLM(len(vocab))
    neural.load_state_dict(saved["state_dict"])
    neural.eval()
    positions = (2, 4)
    triples = []
    for index in positions:
        raw_context = fields[index - 2:index]
        raw_target = fields[index]
        context = [stoi.get(word, 0) for word in raw_context]
        target = stoi.get(raw_target, 0)
        with torch.no_grad():
            probabilities = F.softmax(neural(torch.tensor([context], dtype=torch.long)), dim=-1)[0]
            neural_probability = float(probabilities[target])
        count_probability = count_model.probability(*context, target, discount)
        triples.append({
            "raw_context": raw_context,
            "raw_target": raw_target,
            "mapped_context_ids": context,
            "mapped_context_tokens": [vocab[value] for value in context],
            "mapped_target_id": target,
            "mapped_target_token": vocab[target],
            "training_exact_history_count": count_model.trigram_total[tuple(context)],
            "training_exact_trigram_count": count_model.trigram[context[0], context[1], target],
            "kn_probability": count_probability,
            "kn_negative_log_probability": -math.log(count_probability),
            "neural_probability": neural_probability,
            "neural_negative_log_probability": -math.log(neural_probability),
        })
    output = {
        "source_split": "train",
        "source_article": manifest["first_training_article_title"],
        "source_body_row_zero_based": source["source_row"],
        "source_excerpt_first_18_fields": fields,
        "source_excerpt_mapped_ids": [stoi.get(word, 0) for word in fields],
        "two_selected_prediction_events": triples,
        "count_discount_chosen_on_validation": discount,
        "neural_chosen_epoch_on_validation": saved["chosen_epoch"],
        "note": "This is a training article trace, not an independent quality check. Source fields use supplied WikiText spacing; mapped UNK loses original word identity.",
    }
    RESULT.parent.mkdir(exist_ok=True)
    RESULT.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(triples, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
