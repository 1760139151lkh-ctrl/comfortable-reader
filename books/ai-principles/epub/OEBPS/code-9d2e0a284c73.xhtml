"""第四章：把第三章的一条链式导数原样交给 PyTorch。"""
from __future__ import annotations
import math
import torch
 
x1 = torch.tensor(1.0, dtype=torch.float64)
target = torch.tensor(1.0, dtype=torch.float64)
v = torch.tensor(-2.0, dtype=torch.float64)
b = torch.tensor(1.2, dtype=torch.float64)
a = torch.tensor(math.atanh(0.5), dtype=torch.float64, requires_grad=True)
 
hidden = torch.tanh(a * x1)
score = b + v * hidden
loss = 0.5 * (score - target) ** 2
before = a.item()
loss.backward()  # 只把导数写进 a.grad；尚未更新 a
print("当前活动、分数、损失:", hidden.item(), score.item(), loss.item())
print("backward 后的 a 与导数:", a.item(), a.grad.item())
assert a.item() == before
assert abs(a.grad.item() - 1.2) < 1e-12
 
with torch.no_grad():
    a -= 0.1 * a.grad  # 到这里才改变会在下一次前向使用的参数
print("手动更新后的 a:", a.item())
