"""从本章已选中的两词模型按概率逐步采样，不重新训练。"""
 
from __future__ import annotations
 
import hashlib
import json
from pathlib import Path
 
import torch
from torch.nn import functional as F
 
from word_neural_lm import TwoWordLM
 
 
WORK = Path(__file__).resolve().parents[1]
DATA = WORK / "data" / "wikitext2_raw"
RESULT = WORK / "results" / "word_generated_sample.json"
CHECKPOINT = WORK / "results" / "word_neural_lm_best.pt"
SEED = 20260924
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def main() -> None:
    vocab = json.loads((DATA / "word_vocab.json").read_text(encoding="utf-8"))
    manifest = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    if sha256(DATA / "word_vocab.json") != manifest["vocab_sha256"]:
        raise ValueError("词表字节身份不符")
    saved = torch.load(CHECKPOINT, map_location="cpu", weights_only=True)
    if saved["vocab_sha256"] != manifest["vocab_sha256"]:
        raise ValueError("权重与词表不配套")
    model = TwoWordLM(len(vocab))
    model.load_state_dict(saved["state_dict"])
    model.eval()
    generator = torch.Generator(device="cpu").manual_seed(SEED)
    context = [vocab.index("the"), vocab.index("game")]
    sampled = []
    with torch.no_grad():
        for step in range(40):
            logits = model(torch.tensor([context], dtype=torch.long))
            distribution = F.softmax(logits, dim=-1)[0]
            next_id = int(torch.multinomial(distribution, 1, generator=generator).item())
            sampled.append({
                "step": step + 1,
                "context_tokens": [vocab[index] for index in context],
                "sampled_id": next_id,
                "sampled_token": vocab[next_id],
                "probability_assigned_at_sampling": float(distribution[next_id]),
            })
            if vocab[next_id] == "<eos>":
                break
            context = [context[-1], next_id]
    record = {
        "prompt_tokens": ["the", "game"],
        "sampling_seed": SEED,
        "rule": "multinomial from full softmax at temperature 1; stop at EOS or 40 newly sampled units",
        "checkpoint_sha256": sha256(CHECKPOINT),
        "vocab_sha256": manifest["vocab_sha256"],
        "sampled_steps": sampled,
        "joined_normalized_fields": " ".join(item["sampled_token"] for item in sampled),
        "limits": "word-level mapped vocabulary; UNK cannot recover original rare word; two-word context only; generated text is not independent factual or reasoning evidence",
    }
    RESULT.parent.mkdir(exist_ok=True)
    RESULT.write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(record["joined_normalized_fields"])
 
 
if __name__ == "__main__":
    main()
