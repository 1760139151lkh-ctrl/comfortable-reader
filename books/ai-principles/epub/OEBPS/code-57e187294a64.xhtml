"""Call the localhost-only service using actual HTTP requests and audit boundaries."""
from __future__ import annotations
 
import argparse
import json
import time
from pathlib import Path
from urllib.error import HTTPError
from urllib.request import Request, urlopen
 
 
def request(base, path, payload=None):
    body = None if payload is None else json.dumps(payload, ensure_ascii=False).encode("utf-8")
    req = Request(base + path, data=body, headers={"Content-Type": "application/json"} if body is not None else {}, method="POST" if body is not None else "GET")
    start = time.perf_counter()
    try:
        with urlopen(req, timeout=20) as response:
            return {"http_status": response.status, "json": json.loads(response.read()), "client_elapsed_ms": (time.perf_counter() - start) * 1000}
    except HTTPError as error:
        return {"http_status": error.code, "json": json.loads(error.read()), "client_elapsed_ms": (time.perf_counter() - start) * 1000}
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--port", type=int, required=True)
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("probe report exists")
    base = f"http://127.0.0.1:{a.port}"
    health = request(base, "/health")
    warmup = request(base, "/generate", {"prompt": "The city of", "max_new_tokens": 2, "selection": "greedy"})
    single = request(base, "/generate", {"prompt": "The city of", "max_new_tokens": 12, "selection": "greedy"})
    sampled = request(base, "/generate", {"prompt": "The city of", "max_new_tokens": 12, "selection": "sample_top40", "seed": 7})
    sampled_warmer = request(base, "/generate", {"prompt": "The city of", "max_new_tokens": 12, "selection": "sample_top40", "seed": 7, "temperature": 1.0})
    prompts = ["The city of", "The number of", "The history of", "The first of", "In the early", "In the city", "The game of", "The city of"]
    batched = request(base, "/generate_batch", {"prompts": prompts, "max_new_tokens": 12})
    bad_limit = request(base, "/generate", {"prompt": "The city of", "max_new_tokens": 1000})
    bad_lengths = request(base, "/generate_batch", {"prompts": ["A number of", "The city of"], "max_new_tokens": 4})
    bad_temperature = request(base, "/generate", {"prompt": "The city of", "max_new_tokens": 4, "selection": "sample_top40", "temperature": 0})
    long_prompt = request(base, "/generate", {"prompt": "The city of " * 80, "max_new_tokens": 2})
    if [item["http_status"] for item in (health, warmup, single, sampled, sampled_warmer, batched, bad_limit, bad_lengths, bad_temperature, long_prompt)] != [200, 200, 200, 200, 200, 200, 400, 400, 400, 200]:
        raise RuntimeError("HTTP service did not enforce expected routes/limits")
    same = single["json"]["new_token_ids"] == batched["json"]["outputs"][0]["new_token_ids"]
    if not same or not long_prompt["json"]["input_tokens_outside_128_window"] > 0:
        raise RuntimeError("single vs batch or long prompt boundary mismatch")
    report = {"health": health, "one_warmup_request": warmup, "single_greedy": single, "single_sample": sampled, "single_sample_temperature_1": sampled_warmer, "explicit_equal_length_eight_batch": batched, "invalid_max_new_rejected": bad_limit, "different_tokenized_lengths_rejected": bad_lengths, "invalid_temperature_rejected": bad_temperature, "long_prompt_window_boundary": {"status": long_prompt["http_status"], "input_tokens_outside_128_window": long_prompt["json"]["input_tokens_outside_128_window"], "context_rebuilds": long_prompt["json"]["context_rebuilds"]}, "single_first_equals_batch_first_greedy_token_ids": same, "note": "localhost HTTP service, one weight load, one explicit model warmup before quoted single/batch times, lock serializes unrelated handlers; explicit batch endpoint requires equal tokenized lengths and does not implement automatic continuous batching"}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"model_source_sha256": health["json"]["model_source_sha256"], "single_ms": single["json"]["elapsed_model_ms"], "batch8_ms": batched["json"]["elapsed_model_ms"], "same_first": same, "bad_codes": [bad_limit["http_status"], bad_lengths["http_status"], bad_temperature["http_status"]], "long_prompt_trimmed_tokens": report["long_prompt_window_boundary"]["input_tokens_outside_128_window"]}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
