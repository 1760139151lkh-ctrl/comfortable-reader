"""真实 WikiText-2 词序列：一元频率与单折扣插值 Kneser–Ney 三元模型。"""
 
from __future__ import annotations
 
import collections
import hashlib
import json
import math
import time
from pathlib import Path
 
import numpy as np
 
 
WORK = Path(__file__).resolve().parents[1]
DATA = WORK / "data" / "wikitext2_raw"
RESULT = WORK / "results" / "word_ngram.json"
COUNTS = WORK / "results" / "word_ngram_counts.npz"
DISCOUNTS = (0.5, 0.75, 1.0)
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def documents(tokens: np.ndarray, offsets: np.ndarray):
    for start, end in zip(offsets[:-1], offsets[1:]):
        yield tokens[int(start):int(end)].tolist()
 
 
class InterpolatedKN:
    def __init__(self, docs, vocab_size: int) -> None:
        self.vocab_size = vocab_size
        self.trigram = collections.Counter()
        self.bigram = collections.Counter()
        self.unigram = collections.Counter()
        self.trigram_total = collections.Counter()
        self.bigram_total = collections.Counter()
        for doc in docs:
            for index in range(2, len(doc)):
                u, v, word = doc[index - 2], doc[index - 1], doc[index]
                self.trigram[u, v, word] += 1
                self.bigram[v, word] += 1
                self.unigram[word] += 1
                self.trigram_total[u, v] += 1
                self.bigram_total[v] += 1
        self.trigram_types = collections.Counter((u, v) for u, v, _ in self.trigram)
        # 三元模型降到二元时，KN 不再使用二元组的原始出现次数。
        # 每个 (v,w) 只数它曾接在多少种不同的 u 后面。
        self.lower_bigram = collections.Counter((v, w) for u, v, w in self.trigram)
        self.lower_bigram_total = collections.Counter()
        for (v, _), count in self.lower_bigram.items():
            self.lower_bigram_total[v] += count
        self.lower_bigram_types = collections.Counter(v for v, _ in self.lower_bigram)
        self.predecessor_types = collections.Counter(word for _, word in self.bigram)
        self.total_bigram_types = len(self.bigram)
        self.total_targets = sum(self.unigram.values())
        if self.total_targets == 0 or self.total_bigram_types == 0:
            raise ValueError("训练词序列为空")
 
    def continuation(self, word: int) -> float:
        return self.predecessor_types[word] / self.total_bigram_types
 
    def bigram_probability(self, v: int, word: int, discount: float) -> float:
        total = self.lower_bigram_total[v]
        if total == 0:
            return self.continuation(word)
        observed = max(self.lower_bigram[v, word] - discount, 0.0) / total
        backoff_weight = discount * self.lower_bigram_types[v] / total
        return observed + backoff_weight * self.continuation(word)
 
    def probability(self, u: int, v: int, word: int, discount: float) -> float:
        total = self.trigram_total[u, v]
        lower = self.bigram_probability(v, word, discount)
        if total == 0:
            return lower
        observed = max(self.trigram[u, v, word] - discount, 0.0) / total
        backoff_weight = discount * self.trigram_types[u, v] / total
        return observed + backoff_weight * lower
 
    def unigram_probability(self, word: int) -> float:
        return self.unigram[word] / self.total_targets
 
 
def evaluate(model: InterpolatedKN, docs, discount: float | None) -> dict:
    negative_log_sum, count, unseen_trigrams, unseen_histories = 0.0, 0, 0, 0
    unknown_loss, known_loss, unknown_count = 0.0, 0.0, 0
    examples = []
    for doc in docs:
        for index in range(2, len(doc)):
            u, v, word = doc[index - 2], doc[index - 1], doc[index]
            if discount is None:
                probability = model.unigram_probability(word)
            else:
                probability = model.probability(u, v, word, discount)
            if probability <= 0 or probability > 1 + 1e-9:
                raise ValueError(f"无效预测概率：{(u, v, word, probability)}")
            item_loss = -math.log(probability)
            negative_log_sum += item_loss
            count += 1
            if word == 0:
                unknown_loss += item_loss
                unknown_count += 1
            else:
                known_loss += item_loss
            unseen_trigrams += (u, v, word) not in model.trigram
            unseen_histories += (u, v) not in model.trigram_total
            if len(examples) < 6:
                examples.append({"context_ids": [u, v], "target_id": word, "probability": probability})
    return {
        "targets": count,
        "mean_nll_nats": negative_log_sum / count,
        "perplexity_on_mapped_vocabulary": math.exp(negative_log_sum / count),
        "unseen_trigram_target_fraction": unseen_trigrams / count,
        "unseen_context_fraction": unseen_histories / count,
        "unknown_target_fraction": unknown_count / count,
        "unknown_target_mean_nll": unknown_loss / unknown_count if unknown_count else None,
        "known_target_mean_nll": known_loss / (count - unknown_count),
        "first_six_scored_examples": examples,
    }
 
 
def save_counts(model: InterpolatedKN) -> None:
    trigram_keys = np.asarray(list(model.trigram), dtype=np.int32)
    trigram_values = np.asarray(list(model.trigram.values()), dtype=np.int32)
    bigram_keys = np.asarray(list(model.bigram), dtype=np.int32)
    bigram_values = np.asarray(list(model.bigram.values()), dtype=np.int32)
    lower_bigram_keys = np.asarray(list(model.lower_bigram), dtype=np.int32)
    lower_bigram_values = np.asarray(list(model.lower_bigram.values()), dtype=np.int32)
    unigram_keys = np.asarray(list(model.unigram), dtype=np.int32)
    unigram_values = np.asarray(list(model.unigram.values()), dtype=np.int32)
    np.savez_compressed(
        COUNTS,
        trigram_keys=trigram_keys,
        trigram_values=trigram_values,
        bigram_keys=bigram_keys,
        bigram_values=bigram_values,
        lower_bigram_keys=lower_bigram_keys,
        lower_bigram_values=lower_bigram_values,
        unigram_keys=unigram_keys,
        unigram_values=unigram_values,
    )
 
 
def main() -> None:
    started = time.perf_counter()
    manifest = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    if sha256(DATA / "word_documents.npz") != manifest["arrays_sha256"]:
        raise ValueError("文章划分／词序列文件身份不符")
    if sha256(DATA / "word_vocab.json") != manifest["vocab_sha256"]:
        raise ValueError("训练词表身份不符")
    vocab = json.loads((DATA / "word_vocab.json").read_text(encoding="utf-8"))
    with np.load(DATA / "word_documents.npz") as ready:
        arrays = {key: ready[key].copy() for key in ready.files}
    train_docs = list(documents(arrays["train_tokens"], arrays["train_offsets"]))
    validation_docs = list(documents(arrays["validation_tokens"], arrays["validation_offsets"]))
    test_docs = list(documents(arrays["test_tokens"], arrays["test_offsets"]))
    model = InterpolatedKN(train_docs, len(vocab))
    if model.total_targets != manifest["split_stats"]["train"]["next_token_examples_including_eos"]:
        raise ValueError("训练计数未覆盖所有文章目标")
 
    # 对见过和未见过的历史，各核归一化；模型结构决定概率和应为 1。
    checks = []
    for context in ((1, 1), (vocab.index("the"), vocab.index("game")), (99999, 99999)):
        for discount in DISCOUNTS:
            total = sum(model.probability(*context, word, discount) for word in range(len(vocab)))
            if not math.isclose(total, 1.0, rel_tol=1e-8, abs_tol=1e-8):
                raise AssertionError(f"条件分布不归一：{context}, D={discount}, sum={total}")
            checks.append({"context_ids": context, "discount": discount, "probability_sum": total})
 
    validation_by_discount = {
        str(discount): evaluate(model, validation_docs, discount) for discount in DISCOUNTS
    }
    selected = min(DISCOUNTS, key=lambda d: validation_by_discount[str(d)]["mean_nll_nats"])
    # 折扣选择完成后，官方测试只评价一次。
    test = evaluate(model, test_docs, selected)
    unigram_validation = evaluate(model, validation_docs, None)
    unigram_test = evaluate(model, test_docs, None)
    RESULT.parent.mkdir(exist_ok=True)
    save_counts(model)
    context = (vocab.index("the"), vocab.index("game"))
    top = sorted(
        ((model.probability(*context, word, selected), word) for word in range(len(vocab))),
        reverse=True,
    )[:12]
    record = {
        "data_manifest": str((DATA / "manifest.json").relative_to(WORK)).replace("\\", "/"),
        "data_manifest_sha256": sha256(DATA / "manifest.json"),
        "vocab_sha256": manifest["vocab_sha256"],
        "same_mapped_target_definition": "two preceding mapped words, BOS/BOS at each article start, EOS at article end; PARA inside",
        "vocab_size": len(vocab),
        "training_targets": model.total_targets,
        "distinct_trigrams": len(model.trigram),
        "distinct_bigrams": len(model.bigram),
        "lower_order_count_definition": "N1+(.,v,w): number of distinct left words u with c(u,v,w)>0; raw c(v,w) is retained only for comparison",
        "lower_order_example_the_game": {
            "raw_bigram_count": model.bigram[vocab.index("the"), vocab.index("game")],
            "distinct_left_contexts": model.lower_bigram[vocab.index("the"), vocab.index("game")],
        },
        "distinct_histories": len(model.trigram_total),
        "discount_candidates": DISCOUNTS,
        "validation_by_discount": validation_by_discount,
        "selected_discount": selected,
        "test_selected_discount": test,
        "unigram_validation": unigram_validation,
        "unigram_test": unigram_test,
        "normalization_checks": checks,
        "source_context_the_game": {
            "context": ["the", "game"],
            "history_count": model.trigram_total[context],
            "top_next_words": [{"token": vocab[word], "probability": probability} for probability, word in top],
        },
        "saved_counts": str(COUNTS.relative_to(WORK)).replace("\\", "/"),
        "saved_counts_sha256": sha256(COUNTS),
        "seconds_including_count_build_validation_test": time.perf_counter() - started,
        "note": "single-discount interpolated Kneser-Ney with continuation counts at both lower orders; not the original 1995 backoff implementation or modified multi-discount variants; word vocabulary maps unseen words to UNK",
    }
    RESULT.write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({key: record[key] for key in ("distinct_trigrams", "distinct_bigrams", "selected_discount", "unigram_validation", "test_selected_discount", "seconds_including_count_build_validation_test")}, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
