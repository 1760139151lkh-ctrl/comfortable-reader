"""Read one real calibrated ETH3D pair and its laser-derived reference disparity."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
import re
from pathlib import Path
 
import cv2
import matplotlib.pyplot as plt
import numpy as np
 
 
ROOT = Path(__file__).resolve().parents[2]
BASE = ROOT / "work/data/eth3d_two_view"
plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "SimHei", "DejaVu Sans"]
plt.rcParams["axes.unicode_minus"] = False
 
 
def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 20), b""):
            h.update(block)
    return h.hexdigest()
 
 
def read_calib(path: Path) -> dict:
    content = path.read_text(encoding="utf-8")
    def matrix(name: str) -> np.ndarray:
        match = re.search(rf"^{name}=\[([^\]]+)\]", content, re.M)
        if match is None:
            raise ValueError(f"missing {name}")
        return np.array([[float(x) for x in row.split()]
                         for row in match.group(1).split(";")])
    def scalar(name: str) -> float:
        match = re.search(rf"^{name}=([^\n]+)", content, re.M)
        if match is None:
            raise ValueError(f"missing {name}")
        return float(match.group(1))
    return {"K_left": matrix("cam0"), "K_right": matrix("cam1"),
            "doffs": scalar("doffs"), "baseline": scalar("baseline"),
            "width": int(scalar("width")), "height": int(scalar("height"))}
 
 
def read_pfm(path: Path) -> np.ndarray:
    with path.open("rb") as stream:
        marker = stream.readline().strip()
        if marker != b"Pf":
            raise ValueError(f"expected grayscale PFM: {path}")
        width, height = [int(x) for x in stream.readline().split()]
        scale = float(stream.readline())
        if scale >= 0:
            raise ValueError("ETH3D documentation specifies little-endian PFM")
        data = np.fromfile(stream, dtype="<f4", count=width * height)
        if len(data) != width * height or stream.read(1):
            raise ValueError("PFM pixel payload length mismatch")
    return np.flipud(data.reshape(height, width)).copy()
 
 
def read_gray(path: Path) -> np.ndarray:
    data = cv2.imdecode(np.fromfile(path, dtype=np.uint8), cv2.IMREAD_GRAYSCALE)
    if data is None:
        raise RuntimeError(f"cannot decode {path}")
    return data
 
 
def summarize(values: np.ndarray) -> dict:
    return {"min": float(np.min(values)), "p10": float(np.percentile(values, 10)),
            "median": float(np.median(values)), "p90": float(np.percentile(values, 90)),
            "max": float(np.max(values))}
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--scene", default="delivery_area_1s")
    parser.add_argument("--out-dir", type=Path, required=True)
    args = parser.parse_args()
    if not re.fullmatch(r"[a-z0-9_]+", args.scene):
        raise ValueError("invalid scene id")
    out = args.out_dir if args.out_dir.is_absolute() else ROOT / args.out_dir
    if out.exists():
        raise FileExistsError(out)
    scene = BASE / "images" / args.scene
    truth = BASE / "ground_truth" / args.scene
    if not scene.is_dir() or not truth.is_dir():
        raise FileNotFoundError(args.scene)
    calib = read_calib(scene / "calib.txt")
    left, right = read_gray(scene / "im0.png"), read_gray(scene / "im1.png")
    disparity = read_pfm(truth / "disp0GT.pfm")
    mask = read_gray(truth / "mask0nocc.png")
    if (left.shape != right.shape or left.shape != disparity.shape or
            left.shape != mask.shape or left.shape != (calib["height"], calib["width"])):
        raise RuntimeError("camera/image/GT dimensions disagree")
    valid = np.isfinite(disparity) & (disparity + calib["doffs"] > 0)
    visible_both = valid & (mask == 255)
    if not visible_both.any():
        raise RuntimeError("no valid double-visible ground truth")
    f = calib["K_left"][0, 0]
    depth = (f * calib["baseline"]) / (disparity[visible_both] + calib["doffs"])
    out.mkdir(parents=True)
    figure, axes = plt.subplots(2, 2, figsize=(12, 8), layout="constrained")
    axes[0, 0].imshow(left, cmap="gray", vmin=0, vmax=255)
    axes[0, 0].set_title("真实左相机：输入")
    axes[0, 1].imshow(right, cmap="gray", vmin=0, vmax=255)
    axes[0, 1].set_title("真实右相机：输入")
    disp_view = np.where(visible_both, disparity, np.nan)
    color = axes[1, 0].imshow(disp_view, cmap="viridis",
                              vmin=0, vmax=np.percentile(disparity[visible_both], 98))
    axes[1, 0].set_title("激光配准的参考视差；黑色/空白处不评价")
    figure.colorbar(color, ax=axes[1, 0], label="左横坐标－右横坐标（像素）")
    depth_view = np.full(left.shape, np.nan, dtype=np.float32)
    depth_view[visible_both] = f * calib["baseline"] / (disparity[visible_both] + calib["doffs"])
    color = axes[1, 1].imshow(depth_view, cmap="magma_r",
                              vmin=np.percentile(depth, 2), vmax=np.percentile(depth, 98))
    axes[1, 1].set_title("按校准基线换算的相对深度")
    figure.colorbar(color, ax=axes[1, 1], label="与 calib 基线相同的长度单位")
    for ax in axes.flat:
        ax.set_xticks([])
        ax.set_yticks([])
    figure.suptitle(f"ETH3D {args.scene}：同场景两视角不只是两张相似图", fontsize=15)
    figure.savefig(out / "two_views_and_reference.png", dpi=140)
    plt.close(figure)
    receipt = {
        "scene": args.scene,
        "source_manifest_sha256": sha256(BASE / "extraction_manifest.json"),
        "input_sha256": {name: sha256(path) for name, path in {
            "left": scene / "im0.png", "right": scene / "im1.png",
            "calib": scene / "calib.txt", "disparity_gt": truth / "disp0GT.pfm",
            "mask": truth / "mask0nocc.png"}.items()},
        "shape_hw": left.shape,
        "camera": {"fx_pixels": f, "fy_pixels": float(calib["K_left"][1, 1]),
                   "cx_pixels": float(calib["K_left"][0, 2]),
                   "cy_pixels": float(calib["K_left"][1, 2]),
                   "baseline_as_calib": calib["baseline"], "doffs_pixels": calib["doffs"]},
        "ground_truth": {"double_visible_valid_count": int(visible_both.sum()),
                         "valid_any_count": int(valid.sum()),
                         "mask_counts": {str(k): int((mask == k).sum()) for k in np.unique(mask)},
                         "disparity_pixels": summarize(disparity[visible_both]),
                         "depth_in_baseline_units": summarize(depth)},
        "figure_sha256": sha256(out / "two_views_and_reference.png"),
        "limits": "Reference disparity is from registered laser scans and masked; no stereo algorithm trained by this inspection. Calibration baseline unit is left as documented in calib; camera pose/absolute scale need separate verification.",
    }
    (out / "receipt.json").write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"scene": args.scene, "valid_both": receipt["ground_truth"]["double_visible_valid_count"],
                      "median_disparity": receipt["ground_truth"]["disparity_pixels"]["median"],
                      "median_depth_in_baseline_units": receipt["ground_truth"]["depth_in_baseline_units"]["median"]}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
