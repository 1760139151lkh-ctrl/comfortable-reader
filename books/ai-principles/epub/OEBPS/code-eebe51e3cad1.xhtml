"""Visualize one fixed latent grid through training and real CIFAR validation images."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
from cifar_small_gan import ROOT, digest
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c30_cifar_small_gan"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/figures/c30_cifar_gan_progress.png"))
    parser.add_argument("--receipt", type=Path,
                        default=Path("work/results/c30_cifar_gan_visual_check.json"))
    args = parser.parse_args()
    run = ROOT / args.run_dir
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    real_path = ROOT / report["real_validation_sample"]
    if digest(real_path) != report["real_validation_sample_sha256"]:
        raise RuntimeError("CIFAR real validation sample changed")
    rows = [("真实验证图", np.load(real_path, allow_pickle=False)[:8])]
    for preview in report["fixed_noise_previews"]:
        path = ROOT / preview["path"]
        if digest(path) != preview["sha256"]:
            raise RuntimeError("GAN preview changed")
        rows.append((f"第{preview['epoch']}轮固定噪声", np.load(
            path, allow_pickle=False
        )[:8]))
    out, receipt = ROOT / args.out, ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing GAN figure")
    fig, axes = plt.subplots(len(rows), 8, figsize=(13, 1.8 * len(rows)))
    for row, (title, pictures) in enumerate(rows):
        for col in range(8):
            axes[row, col].imshow(pictures[col], interpolation="nearest")
            axes[row, col].axis("off")
        fig.text(0.01, 1 - (row + 0.5) / len(rows), title,
                 va="center", fontsize=10, fontproperties="Microsoft YaHei")
    fig.tight_layout(rect=(0.15, 0, 1, 1))
    out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=180)
    plt.close(fig)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps({
        "scope": "first8 real CIFAR validation images and same first8 latent vectors at each saved epoch; qualitative visual record, no user rating",
        "source_identity": report["source_identity"],
        "checkpoint_sha256": report["final_checkpoint_sha256"],
        "row_titles": [item[0] for item in rows],
        "figure": str(out.relative_to(ROOT)).replace("\\", "/"),
        "official_test_opened_for_this_experiment": False,
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
 
 
if __name__ == "__main__":
    main()
