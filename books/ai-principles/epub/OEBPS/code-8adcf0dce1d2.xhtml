"""检查译码时未来法语输入和英语 PAD 都不能影响当前预测。"""
 
from __future__ import annotations
 
import json
from pathlib import Path
 
import torch
 
from translate_tatoeba import BOS, EOS, PAD, Translator
 
 
OUT = Path(__file__).resolve().parents[1] / "results" / "translation_visibility_probe.json"
 
 
def main() -> None:
    torch.manual_seed(1601)
    source = torch.tensor([[4, 5, EOS, PAD, PAD]])
    source_with_different_padding = torch.tensor([[4, 5, EOS, 10, 11]])
    length = torch.tensor([3])
    reference_prefix = torch.tensor([[BOS, 6, 7, 8]])
    changed_later_prefix = torch.tensor([[BOS, 6, 9, 10]])
    results = {}
    for name, attention in (("fixed_summary", False), ("soft_read", True)):
        model = Translator(source_vocab=12, target_vocab=12, attention=attention).eval()
        with torch.no_grad():
            original, _ = model(source, length, reference_prefix)
            later_changed, _ = model(source, length, changed_later_prefix)
            padding_changed, _ = model(source_with_different_padding, length, reference_prefix)
        early_gap = float((original[:, :2] - later_changed[:, :2]).abs().max())
        changed_step_gap = float((original[:, 2:] - later_changed[:, 2:]).abs().max())
        padding_gap = float((original - padding_changed).abs().max())
        if early_gap != 0.0 or changed_step_gap <= 0.0 or padding_gap != 0.0:
            raise AssertionError(f"{name}: 未来法语或英语填充位的可见性异常")
        results[name] = {
            "unchanged_earlier_target_steps_max_logit_gap": early_gap,
            "changed_later_target_steps_max_logit_gap": changed_step_gap,
            "changed_only_source_padding_max_logit_gap": padding_gap,
        }
    report = {
        "scope": "随机权重、同一合法源句、两种真实译码结构；只核输入可见性，不评价翻译质量。",
        "source_valid_ids": source[0, :3].tolist(),
        "target_input_original": reference_prefix[0].tolist(),
        "target_input_changed_from_third_step": changed_later_prefix[0].tolist(),
        "results": results,
    }
    OUT.parent.mkdir(exist_ok=True)
    OUT.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(report, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
