def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def load_data(device: torch.device) -> tuple[dict, dict, list[str], list[str]]:
    manifest_file = DATA / "manifest.json"
    manifest = json.loads(manifest_file.read_text(encoding="utf-8"))
    if sha256(DATA / "fra-eng.zip") != manifest["source_identity"]["zip_sha256"]:
        raise ValueError("ManyThings 原包身份不符")
    source_vocab = json.loads((DATA / "english_vocab.json").read_text(encoding="utf-8"))
    target_vocab = json.loads((DATA / "french_vocab.json").read_text(encoding="utf-8"))
    if sha256(DATA / "english_vocab.json") != manifest["source_vocab_sha256"]:
        raise ValueError("英语训练词表改变")
    if sha256(DATA / "french_vocab.json") != manifest["target_vocab_sha256"]:
        raise ValueError("法语训练词表改变")
    splits = {}
    for split in ("train", "validation", "test"):
        spec = manifest["files"][split]
        array_file = WORK / spec["arrays"]
        attribution_file = WORK / spec["attribution"]
        if sha256(array_file) != spec["arrays_sha256"] or sha256(attribution_file) != spec["attribution_sha256"]:
            raise ValueError(f"{split} 数组或署名身份改变")
        with np.load(array_file) as prepared:
            arrays = {key: prepared[key].copy() for key in prepared.files}
        metadata = [json.loads(line) for line in attribution_file.read_text(encoding="utf-8").splitlines()]
        if len(metadata) != len(arrays["source_ids"]):
            raise ValueError(f"{split} 原句署名与数组行数不同")
        reversed_source = arrays["source_ids"].copy()
        for row, length in enumerate(arrays["source_len"]):
            # 只反转真实英语字段，仍将 EOS 留在末尾；PAD 不进入。
            reversed_source[row, :int(length) - 1] = reversed_source[row, :int(length) - 1][::-1]
        arrays["source_reversed"] = reversed_source
        splits[split] = {
            "metadata": metadata,
            **{key: torch.from_numpy(value).to(device) for key, value in arrays.items()},
        }
    return manifest, splits, source_vocab, target_vocab
 
 
class Translator(nn.Module):
    def __init__(self, source_vocab: int, target_vocab: int, attention: bool) -> None:
        super().__init__()
        self.attention = attention
        self.source_embedding = nn.Embedding(source_vocab, EMBEDDING, padding_idx=PAD)
        self.encoder = nn.GRU(EMBEDDING, HIDDEN, batch_first=True)
        self.target_embedding = nn.Embedding(target_vocab, EMBEDDING, padding_idx=PAD)
        self.decoder_cell = nn.GRUCell(EMBEDDING + HIDDEN, HIDDEN)
        self.output = nn.Linear(HIDDEN + HIDDEN, target_vocab)
        self.dropout = nn.Dropout(0.1)
        if attention:
            self.source_project = nn.Linear(HIDDEN, HIDDEN, bias=False)
            self.query_project = nn.Linear(HIDDEN, HIDDEN, bias=False)
            self.score_project = nn.Linear(HIDDEN, 1, bias=False)
 
    def encode(self, source: torch.Tensor, length: torch.Tensor):
        embedded = self.dropout(self.source_embedding(source))
        annotations, _ = self.encoder(embedded)
        row = torch.arange(len(source), device=source.device)
        summary = annotations[row, length.long() - 1]
        mask = torch.arange(source.shape[1], device=source.device)[None, :] < length[:, None]
        projected = self.source_project(annotations) if self.attention else None
        return annotations, summary, mask, projected
 
    def read(
        self,
        previous_state: torch.Tensor,
        annotations: torch.Tensor,
        summary: torch.Tensor,
        mask: torch.Tensor,
        projected: torch.Tensor | None,
    ) -> tuple[torch.Tensor, torch.Tensor | None]:
        if not self.attention:
            return summary, None
        energies = self.score_project(
            torch.tanh(projected + self.query_project(previous_state)[:, None, :])
        ).squeeze(-1)
        energies = energies.masked_fill(~mask, float("-inf"))
        weights = torch.softmax(energies, dim=1)
        context = torch.bmm(weights[:, None, :], annotations).squeeze(1)
        return context, weights
 
    def forward(
        self, source: torch.Tensor, source_len: torch.Tensor, target_input: torch.Tensor,
        capture_weights: bool = False,
    ) -> tuple[torch.Tensor, torch.Tensor | None]:
        annotations, summary, mask, projected = self.encode(source, source_len)
        previous_state = summary
        joint_states, attention_rows = [], []
        for step in range(target_input.shape[1]):
            context, weights = self.read(previous_state, annotations, summary, mask, projected)
            target_vector = self.dropout(self.target_embedding(target_input[:, step]))
            state = self.decoder_cell(torch.cat([target_vector, context], dim=1), previous_state)
            joint_states.append(torch.cat([state, context], dim=1))
            if capture_weights and weights is not None:
                attention_rows.append(weights)
            previous_state = state
        weight_tensor = torch.stack(attention_rows, dim=1) if attention_rows else None
        logits = self.output(self.dropout(torch.stack(joint_states, dim=1)))
        return logits, weight_tensor
 
    @torch.no_grad()
    def greedy(self, source: torch.Tensor, source_len: torch.Tensor, max_steps: int) -> torch.Tensor:
        self.eval()
        annotations, summary, mask, projected = self.encode(source, source_len)
        previous_state = summary
        previous_word = torch.full((len(source),), BOS, dtype=torch.long, device=source.device)
        output_ids = []
        finished = torch.zeros(len(source), dtype=torch.bool, device=source.device)
        for _ in range(max_steps):
            context, _ = self.read(previous_state, annotations, summary, mask, projected)
            target_vector = self.target_embedding(previous_word)
            state = self.decoder_cell(torch.cat([target_vector, context], dim=1), previous_state)
            logits = self.output(torch.cat([state, context], dim=1))
            next_word = logits.argmax(dim=1)
            next_word = torch.where(finished, torch.full_like(next_word, EOS), next_word)
            output_ids.append(next_word)
            finished |= next_word == EOS
            previous_word = next_word
            previous_state = state
            if bool(finished.all()):
                break
        return torch.stack(output_ids, dim=1)
 
 
def source_for_arm(part: dict, arm: str) -> torch.Tensor:
    return part["source_reversed"] if arm == "fixed_reversed" else part["source_ids"]
 
 
def batches_for_epoch(count: int, target_lengths: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    permutation = rng.permutation(count)
    result = []
    for start in range(0, count, 4096):
        chunk = permutation[start:start + 4096]
        chunk = chunk[np.argsort(target_lengths[chunk], kind="stable")]
        result.extend(chunk[offset:offset + BATCH] for offset in range(0, len(chunk), BATCH))
    rng.shuffle(result)
    return result
 
 
@torch.no_grad()
def teacher_evaluate(model: Translator, part: dict, arm: str, with_bins: bool = False) -> dict:
    model.eval()
    source = source_for_arm(part, arm)
    loss_sum, correct, target_count = 0.0, 0, 0
    bins = {"2-5": [0.0, 0], "6-10": [0.0, 0], "11-18": [0.0, 0]}
    for start in range(0, len(source), EVAL_BATCH):
        ids = slice(start, start + EVAL_BATCH)
        target_len = part["target_len"][ids]
        longest = int(target_len.max().item())
        logits, _ = model(source[ids], part["source_len"][ids], part["target_input"][ids, :longest].long())
        targets = part["target_output"][ids, :longest].long()
        losses = F.cross_entropy(logits.transpose(1, 2), targets, ignore_index=PAD, reduction="none")
        valid = targets != PAD
        loss_sum += float(losses.sum().item())
        correct += int(((logits.argmax(dim=-1) == targets) & valid).sum().item())
        target_count += int(valid.sum().item())
        if with_bins:
            source_len_no_eos = part["source_len"][ids] - 1
            for label, condition in (
                ("2-5", (source_len_no_eos >= 2) & (source_len_no_eos <= 5)),
                ("6-10", (source_len_no_eos >= 6) & (source_len_no_eos <= 10)),
                ("11-18", source_len_no_eos >= 11),
            ):
                if bool(condition.any()):
                    bins[label][0] += float(losses[condition].sum().item())
                    bins[label][1] += int(valid[condition].sum().item())
    output = {
        "target_tokens_including_eos": target_count,
        "mean_teacher_forced_nll_nats": loss_sum / target_count,
        "teacher_forced_perplexity_on_mapped_french": math.exp(loss_sum / target_count),
        "teacher_forced_top1_accuracy": correct / target_count,
    }
    if with_bins:
        output["source_length_bins"] = {
            key: {"targets": value[1], "mean_nll": value[0] / value[1] if value[1] else None}
            for key, value in bins.items()
        }
    return output
 
 
def train_arm(
    arm: str, data: dict, source_vocab_size: int, target_vocab_size: int,
    orders: list[list[np.ndarray]], device: torch.device, identity: dict,
) -> dict:
    torch.manual_seed(SEED)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED)
        torch.cuda.reset_peak_memory_stats(device)
    model = Translator(source_vocab_size, target_vocab_size, arm == "soft_read").to(device)
    optimizer = torch.optim.Adam(model.parameters(), lr=LEARNING_RATE)
    train = data["train"]
    source = source_for_arm(train, arm)
    history = []
    best_loss, best_epoch, best_model, best_optimizer, no_improve = float("inf"), None, None, None, 0
    first_batch_trace = None
    if device.type == "cuda":
        torch.cuda.synchronize(device)
    started = time.perf_counter()
    for epoch, epoch_batches in enumerate(orders, start=1):
        model.train()
        total_loss, total_targets = 0.0, 0
        for batch_ids in epoch_batches:
            ids = torch.as_tensor(batch_ids, dtype=torch.long, device=device)
            source_len = train["source_len"].index_select(0, ids)
            target_len = train["target_len"].index_select(0, ids)
            longest = int(target_len.max().item())
            src = source.index_select(0, ids)
            target_in = train["target_input"].index_select(0, ids)[:, :longest].long()
            target_out = train["target_output"].index_select(0, ids)[:, :longest].long()
            optimizer.zero_grad(set_to_none=True)
            logits, _ = model(src, source_len, target_in)
            loss_sum = F.cross_entropy(
                logits.transpose(1, 2), target_out, ignore_index=PAD, reduction="sum"
            )
            active = int((target_out != PAD).sum().item())
            loss = loss_sum / active
            loss.backward()
            pre_clip = float(nn.utils.clip_grad_norm_(model.parameters(), GRAD_CLIP))
            optimizer.step()
            total_loss += float(loss_sum.detach().item())
            total_targets += active
            if first_batch_trace is None:
                original_row = int(batch_ids[0])
                first_batch_trace = {
                    "prepared_training_row": original_row,
                    "source_ids": src[0, :int(source_len[0])].detach().cpu().tolist(),
                    "decoder_input_ids": target_in[0, :int(target_len[0])].detach().cpu().tolist(),
                    "decoder_target_ids": target_out[0, :int(target_len[0])].detach().cpu().tolist(),
                    "active_target_steps": int(target_len[0]),
                    "first_batch_mean_loss": float(loss.detach().item()),
                    "pre_clip_gradient_norm": pre_clip,
                }
        validation = teacher_evaluate(model, data["validation"], arm)
        entry = {
            "epoch": epoch,
            "online_train_nll": total_loss / total_targets,
            "validation": validation,
        }
        history.append(entry)
        print(f"{arm} epoch {epoch}: online train {entry['online_train_nll']:.4f}, "
              f"val NLL {validation['mean_teacher_forced_nll_nats']:.4f}, "
              f"val PPL {validation['teacher_forced_perplexity_on_mapped_french']:.2f}", flush=True)
        if validation["mean_teacher_forced_nll_nats"] < best_loss:
            best_loss, best_epoch = validation["mean_teacher_forced_nll_nats"], epoch
            best_model = {name: value.detach().cpu().clone() for name, value in model.state_dict().items()}
            best_optimizer = copy.deepcopy(optimizer.state_dict())
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= PATIENCE:
                break
    if device.type == "cuda":
        torch.cuda.synchronize(device)
    seconds = time.perf_counter() - started
    if best_model is None or best_optimizer is None:
        raise AssertionError("没有验证选中的模型")
    path = WORK / "results" / f"tatoeba_{arm}_best.pt"
    torch.save({
        "arm": arm, "state_dict": best_model, "optimizer_state_dict": best_optimizer,
        "selected_epoch": best_epoch, "seed": SEED, "identity": identity,
        "embedding": EMBEDDING, "hidden": HIDDEN,
    }, path)
    restored = Translator(source_vocab_size, target_vocab_size, arm == "soft_read").to(device)
    restored.load_state_dict(torch.load(path, map_location="cpu", weights_only=True)["state_dict"])
    model.load_state_dict(best_model)
    model.eval()
    restored.eval()
    part = data["validation"]
    example_source = source_for_arm(part, arm)[:1]
    with torch.no_grad():
        expected, _ = model(example_source, part["source_len"][:1], part["target_input"][:1].long())
        actual, _ = restored(example_source, part["source_len"][:1], part["target_input"][:1].long())
        if not torch.equal(expected, actual):
            raise AssertionError("保存恢复后同一验证句 logits 不一致")
    return {
        "parameters": sum(parameter.numel() for parameter in model.parameters()),
        "epochs_run": len(history),
        "history": history,
        "selected_epoch": best_epoch,
        "selected_validation_nll": best_loss,
        "first_batch_trace": first_batch_trace,
        "train_seconds_excluding_data_loading": seconds,
        "peak_cuda_allocated_mib": torch.cuda.max_memory_allocated(device) / 1024**2 if device.type == "cuda" else None,
        "checkpoint": str(path.relative_to(WORK)).replace("\\", "/"),
        "checkpoint_sha256": sha256(path),
        "_model": restored,
    }
 
 
def trim_eos(sequence: list[int]) -> list[int]:
    return sequence[:sequence.index(EOS)] if EOS in sequence else sequence
 
 
def edit_distance(a: list[str], b: list[str]) -> int:
    previous = list(range(len(b) + 1))
    for index, word in enumerate(a, start=1):
        current = [index]
        for other, reference in enumerate(b, start=1):
            current.append(min(previous[other] + 1, current[-1] + 1, previous[other - 1] + (word != reference)))
        previous = current
    return previous[-1]
 
 
@torch.no_grad()
def greedy_evaluate(model: Translator, part: dict, arm: str, target_vocab: list[str], prepare_tokens) -> dict:
    model.eval()
    metadata = part["metadata"]
    groups = {}
    for row, item in enumerate(metadata):
        key = item["source_group_sha256"]
        if key not in groups:
            groups[key] = {"representative": row, "references": [], "english": item["english_original"],
                           "source_length": int(part["source_len"][row]) - 1}
        groups[key]["references"].append(prepare_tokens(item["french_original"]))
    group_list = list(groups.values())
    matches, similarities, bin_stats = 0, [], {
        "2-5": [0, 0, 0.0], "6-10": [0, 0, 0.0], "11-18": [0, 0, 0.0],
    }
    examples = []
    source = source_for_arm(part, arm)
    for start in range(0, len(group_list), EVAL_BATCH):
        chunk = group_list[start:start + EVAL_BATCH]
        indices = torch.as_tensor([item["representative"] for item in chunk], dtype=torch.long, device=source.device)
        src = source.index_select(0, indices)
        length = part["source_len"].index_select(0, indices)
        generated = model.greedy(src, length, max_steps=20).cpu().tolist()
        for item, sequence in zip(chunk, generated):
            predicted = [target_vocab[token] for token in trim_eos(sequence)]
            reference_scores = [
                (predicted == ref, 1 - edit_distance(predicted, ref) / max(len(predicted), len(ref), 1))
                for ref in item["references"]
            ]
            exact = any(pair[0] for pair in reference_scores)
            similarity = max(pair[1] for pair in reference_scores)
            matches += exact
            similarities.append(similarity)
            length = item["source_length"]
            label = "2-5" if length <= 5 else "6-10" if length <= 10 else "11-18"
            bin_stats[label][0] += 1
            bin_stats[label][1] += exact
            bin_stats[label][2] += similarity
            if len(examples) < 12:
                examples.append({
                    "source": item["english"], "reference_first": item["references"][0],
                    "reference_count_for_same_source": len(item["references"]),
                    "predicted_tokens": predicted, "exact_any": exact, "best_edit_similarity": similarity,
                })
    return {
        "unique_source_sentences": len(group_list),
        "exact_match_to_any_reference": matches / len(group_list),
        "mean_best_normalized_token_edit_similarity": float(np.mean(similarities)),
        "source_length_bins": {
            label: {"sources": data[0], "exact_match": data[1] / data[0],
                    "mean_edit_similarity": data[2] / data[0]}
            for label, data in bin_stats.items() if data[0]
        },
        "first_twelve_sources_before_looking_at_predictions": examples,
        "limits": "Multiple references per exact English source included; other valid translations may be absent. UNK output never equals a real unseen French word. Exact match and edit similarity are reference-based, not human semantic adequacy.",
    }
 
 
def plot_training(outcomes: dict) -> None:
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False

def alignment_example(model: Translator, part: dict, target_vocab: list[str], source_vocab: list[str], source_row: int) -> dict:
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    matches = [index for index, item in enumerate(part["metadata"]) if item["source_row_zero_based"] == source_row]
    if len(matches) != 1:
        raise ValueError("事前选定的验证句未恰好出现一次")
    row = matches[0]
    src_len = int(part["source_len"][row])
    tgt_len = int(part["target_len"][row])
    with torch.no_grad():
        logits, weights = model(
            part["source_ids"][row:row + 1],
            part["source_len"][row:row + 1],
            part["target_input"][row:row + 1, :tgt_len].long(),
            capture_weights=True,
        )
    matrix = weights[0, :tgt_len, :src_len].cpu().numpy()
    if not np.allclose(matrix.sum(axis=1), 1, atol=1e-6):
        raise AssertionError("每个法语目标步的来源权重不归一")
    src_labels = [source_vocab[int(value)] for value in part["source_ids"][row, :src_len].cpu()]
    target_labels = [target_vocab[int(value)] for value in part["target_output"][row, :tgt_len].cpu()]
    fig, ax = plt.subplots(figsize=(max(5.2, src_len * 0.7), max(4.3, tgt_len * 0.45)), layout="constrained")
    image = ax.imshow(matrix, cmap="Blues", vmin=0, vmax=1)
    ax.set(xticks=range(src_len), yticks=range(tgt_len),
           xticklabels=src_labels, yticklabels=target_labels,
           xlabel="英语源位置（含 EOS）", ylabel="法语参考目标步",
           title="给定正确法语前缀时的来源权重")
    ax.tick_params(axis="x", labelrotation=45)
    fig.colorbar(image, ax=ax, fraction=0.045, pad=0.04)
    fig.savefig(ALIGNMENT_FIGURE, dpi=180)
    plt.close(fig)
    return {
        "source_row_zero_based": source_row,
        "english_original": part["metadata"][row]["english_original"],
        "french_original": part["metadata"][row]["french_original"],
        "attribution": part["metadata"][row]["attribution"],
        "source_tokens_used": src_labels,
        "reference_target_tokens_used": target_labels,
        "teacher_forced_source_weight_matrix": matrix.tolist(),
        "max_row_sum_error": float(np.abs(matrix.sum(axis=1) - 1).max()),
        "note": "Weights are from the trained model while feeding reference French prefixes; no manual gold word alignment or proof of a human interpretation.",
    }
 
 
def main() -> None:
    # Import the exact tokenizer chosen in the data preparation, without rerunning preparation.
    from prepare_tatoeba_fra import tokens
 
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    if device.type == "cuda":
        torch.backends.cudnn.benchmark = False
        torch.backends.cudnn.deterministic = True
    manifest, data, source_vocab, target_vocab = load_data(device)
    source_vocab_size, target_vocab_size = len(source_vocab), len(target_vocab)
    rng = np.random.default_rng(SEED + 77)
    train_lengths = data["train"]["target_len"].cpu().numpy()
    orders = [batches_for_epoch(len(data["train"]["metadata"]), train_lengths, rng) for _ in range(EPOCHS)]
    identity = {
        "manifest_sha256": sha256(DATA / "manifest.json"),
        "source_zip_sha256": manifest["source_identity"]["zip_sha256"],
        "english_vocab_sha256": manifest["source_vocab_sha256"],
        "french_vocab_sha256": manifest["target_vocab_sha256"],
    }
    outcomes = {
        arm: train_arm(arm, data, source_vocab_size, target_vocab_size, orders, device, identity)
        for arm in ARMS
    }
    # 三路线的验证选择已完成。测试此后才首次进入预测与评价。
    for arm in ARMS:
        model = outcomes[arm].pop("_model")
        outcomes[arm]["test_teacher_forced"] = teacher_evaluate(model, data["test"], arm, with_bins=True)
        outcomes[arm]["test_greedy"] = greedy_evaluate(model, data["test"], arm, target_vocab, tokens)
        outcomes[arm]["_selected_model"] = model
        print(f"{arm}: test NLL {outcomes[arm]['test_teacher_forced']['mean_teacher_forced_nll_nats']:.4f}, "
              f"greedy exact {outcomes[arm]['test_greedy']['exact_match_to_any_reference']:.4f}", flush=True)
    example = alignment_example(
        outcomes["soft_read"]["_selected_model"],
        data["validation"], target_vocab, source_vocab,
        manifest["alignment_visualization_validation_source_row"],
    )
    for arm in ARMS:
        outcomes[arm].pop("_selected_model")
    RESULT.parent.mkdir(exist_ok=True)
    TRAIN_FIGURE.parent.mkdir(exist_ok=True)
    plot_training(outcomes)
    result = {
        "purpose": "local EN-to-FR short sentence conditional translation; three alternatives, not historical WMT14 reproduction",
        "source_manifest": str((DATA / "manifest.json").relative_to(WORK)).replace("\\", "/"),
        "source_manifest_sha256": identity["manifest_sha256"],
        "source_zip_sha256": identity["source_zip_sha256"],
        "split_counts": {name: manifest["stats"][name]["pairs"] for name in ("train", "validation", "test")},
        "vocab_size_source": source_vocab_size,
        "vocab_size_target": target_vocab_size,
        "source_unknown_fraction_test": manifest["stats"]["test"]["source_unk_fraction"],
        "target_unknown_fraction_test": manifest["stats"]["test"]["target_unk_fraction"],
        "special_ids": manifest["special_ids"],
        "training": {
            "seed": SEED, "embedding": EMBEDDING, "hidden": HIDDEN, "batch": BATCH,
            "max_epochs": EPOCHS, "early_stop_patience": PATIENCE,
            "optimizer": "Adam", "learning_rate": LEARNING_RATE, "gradient_clip_norm": GRAD_CLIP,
            "teacher_forcing": "decoder input BOS + reference y[:-1]; target y + EOS; PAD excluded from loss",
        },
        "device": str(device),
        "gpu_name": torch.cuda.get_device_name(device) if device.type == "cuda" else None,
        "torch_version": torch.__version__,
        "python_version": platform.python_version(),
        "arms": outcomes,
        "preselected_validation_alignment_example": example,
        "limits": "One seed, at most 18 input/target tokens, many short Tatoeba pairs, duplicate/alternative French targets, source grouping only. Teacher-forced NLL and reference-based greedy metrics do not equal WMT14 BLEU or human adequacy.",
    }
    RESULT.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(f"saved {RESULT}, {TRAIN_FIGURE}, {ALIGNMENT_FIGURE}", flush=True)
