"""Show authentic binary Fashion images and free Gibbs RBM samples."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
from fashion_rbm_cd import ROOT, DATA, digest
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c30_fashion_rbm_cd1"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/figures/c30_rbm_free_gibbs_samples.png"))
    parser.add_argument("--receipt", type=Path,
                        default=Path("work/results/c30_rbm_visual_check.json"))
    args = parser.parse_args()
    directory = ROOT / args.run_dir
    report = json.loads((directory / "train.json").read_text(encoding="utf-8"))
    sample_path = ROOT / report["generated_sample_path"]
    if digest(sample_path) != report["generated_sample_sha256"]:
        raise RuntimeError("RBM generated array changed")
    if digest(ROOT / report["checkpoint"]) != report["checkpoint_sha256"]:
        raise RuntimeError("RBM checkpoint changed")
    actual = np.load(DATA / "validation.npy", allow_pickle=False)[:16]
    generated = np.load(sample_path, allow_pickle=False)[:16]
    out, receipt = ROOT / args.out, ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing RBM figure")
    fig, axes = plt.subplots(4, 8, figsize=(14, 8))
    for row in range(4):
        chosen = actual if row < 2 else generated
        for col in range(8):
            axes[row, col].imshow(
                chosen[(row % 2) * 8 + col],
                cmap="gray", vmin=0, vmax=1, interpolation="nearest",
            )
            axes[row, col].axis("off")
    fig.text(0.01, 0.75, "真实验证图", va="center",
             fontproperties="Microsoft YaHei", fontsize=11)
    fig.text(0.01, 0.25, "自由链新样本", va="center",
             fontproperties="Microsoft YaHei", fontsize=11)
    fig.tight_layout(rect=(0.11, 0, 1, 1))
    out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=180)
    plt.close(fig)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps({
        "scope": "first16 real validation binary Fashion images versus first16 of64 RBM samples after 500-step free Gibbs chains; visual only",
        "source_manifest_sha256": report["source_manifest_sha256"],
        "checkpoint_sha256": report["checkpoint_sha256"],
        "sample_array_sha256": report["generated_sample_sha256"],
        "figure": str(out.relative_to(ROOT)).replace("\\", "/"),
        "official_test_opened": False,
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
 
 
if __name__ == "__main__":
    main()
