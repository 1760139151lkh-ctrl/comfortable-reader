"""Official CIFAR-10 scene pixels: label-free contrastive training, then a frozen probe.
 
The CIFAR binary record physically contains a label byte. Training slices only
the 3072 pixel bytes and never interprets that label byte. The official test
record is not opened until both linear readout choices are fixed.
"""
 
from __future__ import annotations
 
import argparse
import copy
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
 
from fashion_ssl_lab import (
    ContrastHead, Encoder, digest, extract, feature_spread, objective,
    validation_objective, write_json,
)
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/cifar10_binary"
MANIFEST = DATA / "manifest.json"
SEED = 20260924
TRAIN_FILES = tuple(f"data_batch_{i}.bin" for i in range(1, 6))
 
 
def official_manifest() -> dict:
    manifest = json.loads(MANIFEST.read_text(encoding="utf-8"))
    if digest(DATA / "cifar-10-binary.tar.gz") != manifest["archive_sha256"]:
        raise RuntimeError("official CIFAR archive changed")
    return manifest
 
 
def records(name: str, manifest: dict) -> np.ndarray:
    path = DATA / name
    if digest(path) != manifest["files"][name]["sha256"]:
        raise RuntimeError(f"official CIFAR member changed: {name}")
    data = np.fromfile(path, dtype=np.uint8)
    if data.size != 10000 * 3073:
        raise RuntimeError(f"unexpected CIFAR record count: {name}")
    return data.reshape(10000, 3073)
 
 
def split_indices() -> tuple[np.ndarray, np.ndarray, str]:
    rows = np.random.default_rng(SEED + 29).permutation(50000).astype(np.int32)
    train, validation = rows[:45000], rows[45000:]
    return train, validation, hashlib.sha256(rows.tobytes()).hexdigest()
 
 
def train_pixels(device: torch.device):
    manifest = official_manifest()
    # The first byte in every record is present in the file, but is discarded.
    # No class number or label-derived split enters representation training.
    pixels = np.concatenate(
        [records(name, manifest)[:, 1:].reshape(10000, 3, 32, 32)
         for name in TRAIN_FILES],
        axis=0,
    ).copy()
    train, validation, split_sha = split_indices()
    images = torch.from_numpy(pixels).to(device=device, dtype=torch.float32) / 255
    source = {
        "manifest_sha256": digest(MANIFEST),
        "official_archive_sha256": manifest["archive_sha256"],
        "train_members": {name: manifest["files"][name]["sha256"] for name in TRAIN_FILES},
        "random_unstratified_split_sha256": split_sha,
        "train_rows": len(train), "validation_rows": len(validation),
    }
    return images, train, validation, source
 
 
def train(epochs: int, out_dir: Path) -> None:
    if out_dir.exists() and any(out_dir.iterdir()):
        raise FileExistsError(f"preserve existing run: {out_dir}")
    out_dir.mkdir(parents=True, exist_ok=True)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    images, train_rows, validation_rows, source = train_pixels(device)
    torch.manual_seed(SEED)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED)
        torch.cuda.reset_peak_memory_stats()
    encoder = Encoder(channels=3, side=32).to(device)
    head = ContrastHead().to(device)
    initial_encoder = copy.deepcopy(encoder.state_dict())
    optimizer = torch.optim.AdamW(
        list(encoder.parameters()) + list(head.parameters()),
        lr=0.001, weight_decay=1e-4,
    )
    order_rng = np.random.default_rng(SEED)
    history = []
    best_epoch, best_val, best_state = 0, float("inf"), None
    for epoch in range(1, epochs + 1):
        encoder.train()
        head.train()
        order = order_rng.permutation(train_rows)
        online_total, steps = 0.0, 0
        for start in range(0, len(order), 256):
            chosen = order[start:start + 256]
            if len(chosen) < 2:
                continue
            loss = objective("contrast", encoder, head, images[chosen])
            if not torch.isfinite(loss):
                raise RuntimeError("nonfinite CIFAR contrastive loss")
            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(
                list(encoder.parameters()) + list(head.parameters()), 1.0
            )
            optimizer.step()
            online_total += float(loss.detach())
            steps += 1
        val = validation_objective(
            "contrast", encoder, head, images, validation_rows
        )
        spread = feature_spread(encoder, images, validation_rows)
        history.append({
            "epoch": epoch, "train_online_mean_loss": online_total / steps,
            "validation_fixed_views_loss": val,
            "steps": steps, "feature_spread": spread,
        })
        if val < best_val:
            best_epoch, best_val = epoch, val
            best_state = {
                "encoder": copy.deepcopy(encoder.state_dict()),
                "head": copy.deepcopy(head.state_dict()),
            }
        print("cifar contrast epoch", epoch, "val", round(val, 4), flush=True)
    if best_state is None:
        raise RuntimeError("no CIFAR checkpoint chosen")
    checkpoint = {
        "mode": "cifar_contrast", "selected_epoch": best_epoch,
        "encoder": {k: v.cpu().clone() for k, v in best_state["encoder"].items()},
        "head": {k: v.cpu().clone() for k, v in best_state["head"].items()},
        "initial_encoder": {k: v.cpu().clone() for k, v in initial_encoder.items()},
        "source_identity": source,
    }
    best_path = out_dir / "best.pt"
    torch.save(checkpoint, best_path)
    write_json(out_dir / "train.json", {
        "scope": "official CIFAR-10 32x32 RGB scenes, label byte ignored in pretraining; random unstratified 45k/5k split",
        "mode": "cifar_contrast", "epochs_planned": epochs,
        "batch_images": 256,
        "source_identity": source,
        "augmentation": "two independent shifts up to four pixels, horizontal flip, brightness, Gaussian noise; same-source positives",
        "objective": "2B normalized projected-view cross entropy, other originals as negatives, temperature0.2",
        "optimizer": "AdamW lr0.001 weight_decay1e-4 gradient_clip1",
        "selection": "lowest fixed-view self-supervised loss on first2048 validation rows; no labels",
        "history": history, "selected_epoch": best_epoch,
        "checkpoint": str(best_path.relative_to(ROOT)).replace("\\", "/"),
        "checkpoint_sha256": digest(best_path),
        "labels_interpreted_in_pretraining": False,
        "official_test_opened": False,
        "peak_gpu_allocated_bytes": (
            torch.cuda.max_memory_allocated() if device.type == "cuda" else None
        ),
    })
 
 
def probe(run_dir: Path, out: Path) -> None:
    if out.exists():
        raise FileExistsError(f"preserve existing probe: {out}")
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    images, train_rows, validation_rows, source = train_pixels(device)
    report = json.loads((run_dir / "train.json").read_text(encoding="utf-8"))
    checkpoint_path = ROOT / report["checkpoint"]
    if (report["source_identity"] != source or
            digest(checkpoint_path) != report["checkpoint_sha256"]):
        raise RuntimeError("CIFAR training identity changed")
    checkpoint = torch.load(checkpoint_path, map_location="cpu", weights_only=True)
    if checkpoint["source_identity"] != source:
        raise RuntimeError("CIFAR checkpoint source changed")
    manifest = official_manifest()
    # Class bytes from official train files are interpreted only here.
    labels = np.concatenate(
        [records(name, manifest)[:, 0] for name in TRAIN_FILES]
    )
    subset_rng = np.random.default_rng(SEED + 1)
    subset = np.array(sorted(
        index
        for label in range(10)
        for index in subset_rng.choice(
            train_rows[labels[train_rows] == label], 500, replace=False
        ).tolist()
    ), dtype=np.int32)
    torch.manual_seed(SEED)
    encoders = {"random_same_architecture": Encoder(channels=3, side=32).to(device)}
    trained = Encoder(channels=3, side=32).to(device)
    trained.load_state_dict(checkpoint["encoder"])
    encoders["contrast"] = trained
    selected_models, results = {}, {}
    for name, encoder in encoders.items():
        raw_train = extract(encoder, images, subset)
        raw_val = extract(encoder, images, validation_rows)
        scaler = StandardScaler().fit(raw_train)
        x_train, x_val = scaler.transform(raw_train), scaler.transform(raw_val)
        best_acc, chosen_model, chosen_c = -1.0, None, None
        candidates = []
        for c in (0.01, 0.1, 1.0, 10.0):
            model = LogisticRegression(C=c, max_iter=2000, solver="lbfgs")
            model.fit(x_train, labels[subset])
            accuracy = float((model.predict(x_val) == labels[validation_rows]).mean())
            candidates.append({
                "C": c, "validation_accuracy": accuracy,
                "max_solver_iterations": int(model.n_iter_.max()),
            })
            if accuracy > best_acc:
                best_acc, chosen_model, chosen_c = accuracy, model, c
        selected_models[name] = (encoder, scaler, chosen_model)
        results[name] = {
            "labeled_train_count": len(subset),
            "validation_count": len(validation_rows),
            "C_candidates": candidates, "selected_C": chosen_c,
            "validation_accuracy": best_acc,
            "feature_coordinate_std_mean_validation_raw": float(
                raw_val.std(axis=0).mean()
            ),
        }
    # Official test is first opened after both models and all C choices fixed.
    test_records = records("test_batch.bin", manifest)
    test_labels = test_records[:, 0].copy()
    test_images = torch.from_numpy(
        test_records[:, 1:].reshape(10000, 3, 32, 32).copy()
    ).to(device=device, dtype=torch.float32) / 255
    for name, (encoder, scaler, model) in selected_models.items():
        test_features = scaler.transform(
            extract(encoder, test_images, np.arange(10000))
        )
        results[name]["official_test_count"] = 10000
        results[name]["official_test_accuracy"] = float(
            (model.predict(test_features) == test_labels).mean()
        )
        print(name, "val", results[name]["validation_accuracy"],
              "test", results[name]["official_test_accuracy"], flush=True)
    write_json(out, {
        "scope": "official CIFAR-10 first local test opening for this SSL probe after validation selection; 500 labels per class, frozen encoder",
        "source_identity": source,
        "ssl_checkpoint_sha256": report["checkpoint_sha256"],
        "official_test_member_sha256": manifest["files"]["test_batch.bin"]["sha256"],
        "results": results,
    })
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("mode", choices=("train", "probe"))
    parser.add_argument("--epochs", type=int, default=8)
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c29_cifar_contrast"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c29_cifar_linear_probe.json"))
    args = parser.parse_args()
    if args.mode == "train":
        train(args.epochs, ROOT / args.run_dir)
    else:
        probe(ROOT / args.run_dir, ROOT / args.out)
 
 
if __name__ == "__main__":
    main()
