"""比较同数据/日程下轮界恢复与不中断训练的完整状态和验证曲线。"""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import torch
 
 
WORK = Path(__file__).resolve().parents[1]
RESTORED = WORK / "runs" / "c19_revision_20260924"
CONTINUOUS = WORK / "runs" / "c19_uninterrupted_20260924"
OUT = WORK / "results" / "c19_resume_comparison.json"
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def epoch_entries(path: Path) -> list[dict]:
    with path.open(encoding="utf-8") as stream:
        return [item for line in stream if (item := json.loads(line)).get("event") == "epoch_end"]
 
 
def compare_values(left, right, path: str, differences: list[dict]) -> None:
    if isinstance(left, torch.Tensor) and isinstance(right, torch.Tensor):
        if left.shape != right.shape or left.dtype != right.dtype:
            raise AssertionError(f"{path}: shape or dtype changed")
        a, b = left.detach().cpu(), right.detach().cpu()
        if torch.is_floating_point(a):
            gap = float((a.double() - b.double()).abs().max()) if a.numel() else 0.0
            differences.append({"path": path, "max_abs": gap, "exact": torch.equal(a, b)})
        elif not torch.equal(a, b):
            differences.append({"path": path, "nonfloating_tensor_differs": True})
        return
    if isinstance(left, dict) and isinstance(right, dict):
        if left.keys() != right.keys():
            raise AssertionError(f"{path}: dictionary keys changed")
        for key in left:
            compare_values(left[key], right[key], f"{path}/{key}", differences)
        return
    if isinstance(left, (list, tuple)) and isinstance(right, (list, tuple)):
        if len(left) != len(right):
            raise AssertionError(f"{path}: sequence length changed")
        for index, (a, b) in enumerate(zip(left, right)):
            compare_values(a, b, f"{path}/{index}", differences)
        return
    if left != right:
        differences.append({"path": path, "left": left, "right": right})
 
 
def resolve_in_project(path_text: str) -> Path:
    path = Path(path_text)
    return path.resolve() if path.is_absolute() else (WORK.parent / path).resolve()
 
 
def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--restored-dir", default=str(RESTORED))
    parser.add_argument("--continuous-dir", default=str(CONTINUOUS))
    parser.add_argument("--out", default=str(OUT))
    args = parser.parse_args()
    restored_dir = resolve_in_project(args.restored_dir)
    continuous_dir = resolve_in_project(args.continuous_dir)
    output = resolve_in_project(args.out)
    restored_path = restored_dir / "causal_lm_best.pt"
    continuous_path = continuous_dir / "causal_lm_best.pt"
    restored = torch.load(restored_path, map_location="cpu", weights_only=True)
    continuous = torch.load(continuous_path, map_location="cpu", weights_only=True)
    required_equal = ["config", "c18_manifest_sha256", "epoch", "next_batch", "global_step", "best_epoch"]
    for key in required_equal:
        if restored[key] != continuous[key]:
            raise AssertionError(f"两条训练在{key}身份上不一致")
    if restored["global_step"] != 1336 or restored["epoch"] != 4:
        raise AssertionError("未比较到完成四轮的同一步")
    model_differences, optimizer_differences = [], []
    compare_values(restored["model"], continuous["model"], "model", model_differences)
    compare_values(restored["optimizer"], continuous["optimizer"], "optimizer", optimizer_differences)
    rr = epoch_entries(restored_dir / "causal_lm_history.jsonl")
    cc = epoch_entries(continuous_dir / "causal_lm_history.jsonl")
    if len(rr) != len(cc) or len(rr) != 4:
        raise AssertionError("两条训练必须各有完整四轮记录")
    validation = [
        {"epoch": a["epoch"], "restored_nll": a["validation"]["nll"],
         "continuous_nll": b["validation"]["nll"],
         "abs_gap": abs(a["validation"]["nll"] - b["validation"]["nll"])}
        for a, b in zip(rr, cc)
    ]
    result = {
        "scope": "两条真正执行的四轮同数据/同日程训练：第一条第1轮后退出并恢复，第二条不中断；比较最佳检查点及四轮验证。没有打开测试资料。",
        "restored_directory": str(restored_dir),
        "continuous_directory": str(continuous_dir),
        "restored_checkpoint_sha256": sha256(restored_path),
        "continuous_checkpoint_sha256": sha256(continuous_path),
        "c18_manifest_sha256": restored["c18_manifest_sha256"],
        "global_step_each": restored["global_step"],
        "restored_resume_event": any('"event": "resume"' in line for line in (restored_dir / "causal_lm_history.jsonl").read_text(encoding="utf-8").splitlines()),
        "model_tensor_count": len(model_differences),
        "model_tensors_exactly_equal": sum(item.get("exact", False) for item in model_differences),
        "model_largest_parameter_abs_gap": max((item["max_abs"] for item in model_differences), default=0.0),
        "optimizer_largest_floating_abs_gap": max((item["max_abs"] for item in optimizer_differences if "max_abs" in item), default=0.0),
        "optimizer_nonfloating_differences": [item for item in optimizer_differences if "max_abs" not in item],
        "best_validation_nll_abs_gap": abs(restored["best_validation_nll"] - continuous["best_validation_nll"]),
        "validation_by_epoch": validation,
        "cpu_rng_exactly_equal": torch.equal(restored["torch_cpu_rng"], continuous["torch_cpu_rng"]),
        "cuda_rng_exactly_equal": all(torch.equal(a, b) for a, b in zip(restored["torch_cuda_rng_all"], continuous["torch_cuda_rng_all"])),
        "interpretation_limit": "小数差可能来自GPU浮点执行顺序；该比较核轮界恢复是否延续同一设置和近似轨迹，不证明所有断电时刻、机器或库版本逐位相同。",
    }
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({key: result[key] for key in ("global_step_each", "model_tensor_count", "model_tensors_exactly_equal", "model_largest_parameter_abs_gap", "optimizer_largest_floating_abs_gap", "best_validation_nll_abs_gap", "cpu_rng_exactly_equal", "cuda_rng_exactly_equal")}, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
