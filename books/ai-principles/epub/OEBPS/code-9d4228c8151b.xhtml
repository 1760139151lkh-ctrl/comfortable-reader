"""Enumerate a four-visible/two-hidden RBM to expose exact and CD-1 phases."""
 
from __future__ import annotations
 
import argparse
import itertools
import json
import math
from pathlib import Path
 
import numpy as np
import torch
 
 
ROOT = Path(__file__).resolve().parents[2]
 
 
def states(width: int) -> np.ndarray:
    return np.asarray(list(itertools.product((0.0, 1.0), repeat=width)))
 
 
def sigmoid(x):
    return 1 / (1 + np.exp(-x))
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c30_rbm_exact_probe.json"))
    args = parser.parse_args()
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError("preserve existing RBM probe")
    visible = states(4)
    hidden = states(2)
    observed = np.asarray([1.0, 0.0, 1.0, 0.0])
    W = np.asarray([
        [1.3, -0.7],
        [-0.9, 1.1],
        [0.8, 0.6],
        [-0.5, 1.2],
    ])
    b = np.asarray([-0.3, 0.1, 0.25, -0.2])
    c = np.asarray([0.2, -0.15])
    # Joint log weight is -E(v,h), with one row for each of 16*4 states.
    logits = (
        visible[:, None, :] @ b +
        hidden[None, :, :] @ c +
        np.einsum("vi,ij,hj->vh", visible, W, hidden)
    )
    normalizer = np.exp(logits - logits.max()).sum() * np.exp(logits.max())
    joint = np.exp(logits) / normalizer
    p_visible = joint.sum(axis=1)
    positive_hidden = sigmoid(c + observed @ W)
    positive = observed[:, None] * positive_hidden[None, :]
    model_negative = np.einsum("vh,vi,hj->ij", joint, visible, hidden)
    exact = positive - model_negative
    observed_index = int(np.where(np.all(visible == observed, axis=1))[0][0])
    # Evaluate the true gradient from the fully normalized visible likelihood.
    tensor_W = torch.tensor(W, dtype=torch.float64, requires_grad=True)
    tensor_b = torch.tensor(b, dtype=torch.float64)
    tensor_c = torch.tensor(c, dtype=torch.float64)
    vs = torch.tensor(visible, dtype=torch.float64)
    hs = torch.tensor(hidden, dtype=torch.float64)
    torch_logits = (
        vs[:, None, :] @ tensor_b +
        hs[None, :, :] @ tensor_c +
        torch.einsum("vi,ij,hj->vh", vs, tensor_W, hs)
    )
    log_probability = (
        torch.logsumexp(torch_logits[observed_index], dim=0) -
        torch.logsumexp(torch_logits.reshape(-1), dim=0)
    )
    gradient, = torch.autograd.grad(log_probability, tensor_W)
    error = float(np.max(np.abs(gradient.numpy() - exact)))
    if error > 1e-12:
        raise RuntimeError("RBM exact positive/negative gradient mismatch")
    # CD-1 starts at THIS observation, not at equilibrium. Enumerate its
    # one h|v and one v'|h transition exactly; no Monte Carlo noise here.
    p_h_given_observed = np.prod(
        np.where(hidden == 1, positive_hidden, 1 - positive_hidden),
        axis=1,
    )
    p_visible_after_one = np.zeros(len(visible))
    for h_index, h in enumerate(hidden):
        p_bit = sigmoid(b + W @ h)
        p_visible_given_h = np.prod(
            np.where(visible == 1, p_bit, 1 - p_bit), axis=1
        )
        p_visible_after_one += p_h_given_observed[h_index] * p_visible_given_h
    h_prob_after_one = sigmoid(visible @ W + c)
    cd_negative = np.einsum(
        "v,vi,vj->ij", p_visible_after_one, visible, h_prob_after_one
    )
    cd_one = positive - cd_negative
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps({
        "scope": "exact 4-visible/2-hidden finite RBM math check; CD-1 expectation enumerated from one observation, not natural-image training",
        "observed_visible": observed.tolist(),
        "W": W.tolist(), "visible_bias": b.tolist(),
        "hidden_bias": c.tolist(),
        "partition_function": normalizer,
        "log_probability_observed": float(log_probability.detach()),
        "positive_statistic": positive.tolist(),
        "equilibrium_negative_statistic": model_negative.tolist(),
        "exact_log_likelihood_gradient": exact.tolist(),
        "torch_autograd_gradient": gradient.numpy().tolist(),
        "maximum_exact_gradient_difference": error,
        "one_step_visible_distribution_sum": float(p_visible_after_one.sum()),
        "one_step_negative_statistic": cd_negative.tolist(),
        "cd1_expected_update": cd_one.tolist(),
        "cd1_minus_exact_max_abs": float(np.max(np.abs(cd_one - exact))),
        "stationary_visible_probability_observed": float(p_visible[observed_index]),
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("log p(v)", float(log_probability.detach()),
          "grad diff", error,
          "CD1 max bias", float(np.max(np.abs(cd_one - exact))))
 
 
if __name__ == "__main__":
    main()
