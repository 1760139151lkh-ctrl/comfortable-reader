"""Synchronized full-prefix vs cache and equal-length request batching on C19."""
from __future__ import annotations
 
import argparse
import json
import statistics
import time
from pathlib import Path
 
import numpy as np
import torch
 
from c37_cached_decode import DATA, decode_one, full_logits, load_serving, prefill, select_token
 
 
def sync(device):
    if device.type == "cuda":
        torch.cuda.synchronize(device)
 
 
@torch.inference_mode()
def fixed_length_decode(model, prompt: torch.Tensor, new_steps: int, engine: str):
    """Timing workload: always compute new_steps, record any EOS but do not early-exit."""
    if engine == "cached":
        logits, cache = prefill(model, prompt)
    else:
        logits, cache = full_logits(model, prompt), None
    tokens = prompt
    eos_count = 0
    rebuilds = 0
    for step in range(new_steps):
        chosen = select_token(logits)
        eos_count += int((chosen == 2).sum())
        if step + 1 == new_steps:
            break
        tokens = torch.cat([tokens, chosen[:, None]], dim=1)
        if engine == "cached":
            if cache[0][0].shape[2] == model.cfg.context:
                logits, cache = prefill(model, tokens[:, -model.cfg.context:])
                rebuilds += 1
            else:
                logits, cache = decode_one(model, chosen, cache)
        else:
            logits = full_logits(model, tokens)
    return eos_count, rebuilds
 
 
def time_call(fn, device, repeats=7, warmups=3):
    for _ in range(warmups):
        fn()
    sync(device)
    values = []
    for _ in range(repeats):
        sync(device)
        start = time.perf_counter()
        fn()
        sync(device)
        values.append((time.perf_counter() - start) * 1000)
    return {"repeats": repeats, "median_ms": statistics.median(values), "min_ms": min(values), "max_ms": max(values), "each_ms": values}
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("result exists")
    torch.set_num_threads(8)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model, _, state = load_serving(device)
    train = np.load(DATA / "train_blocks.npz")
    result = {"identity": "same frozen C19 four-layer model; FP32 inference, no autocast, no optimizer", "device": str(device), "selected_epoch": state["selected_epoch"], "measurement": "torch.inference_mode; GPU synchronize before and after every timed operation; median over seven repeats after three warmups except component probe 21 repeats; times may drift across sessions", "single_request": {}, "prefill_and_one_decode": {}, "eight_simultaneous_requests": {}}
    for length in (16, 64, 120):
        prompt = torch.as_tensor(train["input_ids"][0, :length].copy(), dtype=torch.long, device=device)[None]
        arms = {}
        for engine in ("full", "cached"):
            arms[engine] = time_call(lambda: fixed_length_decode(model, prompt, 16, engine), device)
            with torch.inference_mode():
                eos, rebuilds = fixed_length_decode(model, prompt, 16, engine)
            arms[engine]["fixed_steps"] = 16
            arms[engine]["eos_id_occurrences_not_early_stopped_for_timing"] = eos
            arms[engine]["cache_rebuilds"] = rebuilds
            arms[engine]["output_tokens_per_second_median"] = 16 / (arms[engine]["median_ms"] / 1000)
        result["single_request"][str(length)] = {"prompt_tokens": length, "arms": arms, "median_full_over_cached_speed_ratio": arms["full"]["median_ms"] / arms["cached"]["median_ms"]}
        prefill_full = time_call(lambda: full_logits(model, prompt), device, repeats=21)
        prefill_cache = time_call(lambda: prefill(model, prompt), device, repeats=21)
        _, one_cache = prefill(model, prompt)
        next_token = torch.as_tensor(train["input_ids"][0, length:length + 1].copy(), dtype=torch.long, device=device)
        next_window = torch.cat((prompt, next_token[:, None]), dim=1)
        next_full = time_call(lambda: full_logits(model, next_window), device, repeats=21)
        next_cached = time_call(lambda: decode_one(model, next_token, one_cache), device, repeats=21)
        result["prefill_and_one_decode"][str(length)] = {"full_prompt_ms": prefill_full["median_ms"], "cached_prompt_prefill_ms": prefill_cache["median_ms"], "next_full_prefix_ms": next_full["median_ms"], "next_one_token_cached_ms": next_cached["median_ms"], "cached_KV_bytes_for_this_prompt": sum(k.numel() * k.element_size() + v.numel() * v.element_size() for k, v in one_cache)}
 
    prompts = torch.as_tensor(train["input_ids"][:8, :16].copy(), dtype=torch.long, device=device)
    separate = time_call(lambda: [fixed_length_decode(model, prompts[i:i + 1], 16, "cached") for i in range(8)], device)
    batched = time_call(lambda: fixed_length_decode(model, prompts, 16, "cached"), device)
    # For simultaneous arrivals, a serial queue completes request i after
    # i+1 single-request durations. Equal-length batch completes all together.
    single_request_ms = separate["median_ms"] / 8
    queue_completion = [(i + 1) * single_request_ms for i in range(8)]
    result["eight_simultaneous_requests"] = {"prompt_tokens_each": 16, "new_steps_each": 16, "serial_cached": separate, "one_equal_length_batch_cached": batched, "serial_total_new_tokens_per_second": 128 / (separate["median_ms"] / 1000), "batched_total_new_tokens_per_second": 128 / (batched["median_ms"] / 1000), "estimated_serial_p50_completion_ms_given_same_arrival": float(np.median(queue_completion)), "estimated_serial_max_completion_ms_given_same_arrival": max(queue_completion), "batch_all_complete_ms_given_same_arrival": batched["median_ms"], "warning": "fixed-step microbenchmark ignores early EOS and assumes 8 simultaneous arrivals/equal prompt and generation lengths; service scheduling/real traffic differ"}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"single": {k:{"full_ms":v["arms"]["full"]["median_ms"],"cached_ms":v["arms"]["cached"]["median_ms"],"ratio":v["median_full_over_cached_speed_ratio"],"rebuilds":v["arms"]["cached"]["cache_rebuilds"]} for k,v in result["single_request"].items()}, "eight": {k:v for k,v in result["eight_simultaneous_requests"].items() if isinstance(v,(int,float))}}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
