"""Limited pixel/copy check for C31's real-CIFAR continuous flow samples."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
 
from c30_gan_diagnostic import mean_pairwise_rmse, nearest_training_rmse
from cifar_ssl_lab import ROOT, train_pixels, digest
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c31_cifar_flow"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c31_cifar_flow_pixel_diagnostic.json"))
    args = parser.parse_args()
    run, out = ROOT / args.run_dir, ROOT / args.out
    if out.exists():
        raise FileExistsError("preserve existing flow diagnostic")
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    if report["dataset"] != "cifar":
        raise RuntimeError("this diagnostic needs CIFAR RGB")
    chosen = next(item for item in report["samplers"] if item["name"] == "heun_64")
    sample_path = ROOT / chosen["path"]
    if digest(sample_path) != chosen["sha256"]:
        raise RuntimeError("CIFAR flow sample changed")
    generated_bytes = np.load(sample_path, allow_pickle=False)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    all_images, train_rows, val_rows, source = train_pixels(device)
    if any(source[key] != report["source_identity"][key] for key in source):
        raise RuntimeError("CIFAR flow source changed")
    training, real = all_images[train_rows], all_images[val_rows[:16]]
    generated = torch.from_numpy(generated_bytes.copy()).to(
        device=device, dtype=torch.float32
    ).permute(0, 3, 1, 2) / 255
    raw_training_bytes = (
        (training * 255).round().byte().permute(0, 2, 3, 1).cpu().numpy()
    )
    hashes = {hashlib.sha256(row.tobytes()).digest()
              for row in raw_training_bytes}
    copies = sum(
        hashlib.sha256(row.tobytes()).digest() in hashes
        for row in generated_bytes
    )
    result = {
        "scope": "16 Heun64 CIFAR flow images against 16 unpaired validation and 45k training images; raw pixel diagnostics only",
        "source_identity": report["source_identity"],
        "generated_array_sha256": chosen["sha256"],
        "exact_generated_train_byte_matches": copies,
        "unique_generated_byte_arrays": len({
            hashlib.sha256(row.tobytes()).digest()
            for row in generated_bytes
        }),
        "generated_pairwise_pixel_rmse_mean": mean_pairwise_rmse(generated),
        "real_validation_pairwise_pixel_rmse_mean": mean_pairwise_rmse(real),
        "generated_nearest_train_pixel_rmse": nearest_training_rmse(generated, training),
        "real_validation_nearest_train_pixel_rmse": nearest_training_rmse(real, training),
        "limitations": "no exact copies does not rule out changed memorization; raw pixel distance does not measure object correctness, coverage, or human preference",
    }
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n",
                   encoding="utf-8")
    print("exact copies", copies, "unique", result["unique_generated_byte_arrays"],
          "generated pair RMSE", result["generated_pairwise_pixel_rmse_mean"])
 
 
if __name__ == "__main__":
    main()
