"""用本机固定分词器验证同一字串能由多条生成编号路径解码。"""
 
from __future__ import annotations
 
import hashlib
import json
from pathlib import Path
 
from tokenizers import Tokenizer
 
 
WORK = Path(__file__).resolve().parents[1]
DATA = WORK / "data" / "wikitext2_causal"
OUT = WORK / "results" / "tokenization_ambiguity_probe.json"
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def main() -> None:
    manifest_path = DATA / "manifest.json"
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    tokenizer_path = DATA / "tokenizer.json"
    if sha256(tokenizer_path) != manifest["tokenizer"]["model_sha256"]:
        raise ValueError("分词器字节身份与C18准备清单不符")
    tokenizer = Tokenizer.from_file(str(tokenizer_path))
    raw = " the"
    canonical = tokenizer.encode(raw, add_special_tokens=False).ids
    token_paths = [
        canonical,
        [tokenizer.token_to_id("Ġt"), tokenizer.token_to_id("he")],
        [tokenizer.token_to_id("Ġ"), tokenizer.token_to_id("t"), tokenizer.token_to_id("he")],
    ]
    if any(item is None for path in token_paths for item in path):
        raise ValueError("当前已核分词器中缺少教学例子的片段")
    if len({tuple(path) for path in token_paths}) != len(token_paths):
        raise AssertionError("例子中的编号路径必须彼此不同")
    decoded = [tokenizer.decode(path, skip_special_tokens=False) for path in token_paths]
    if any(value != raw for value in decoded):
        raise AssertionError("多条编号路径未解回同一原字串")
    report = {
        "source_manifest_sha256": sha256(manifest_path),
        "tokenizer_sha256": sha256(tokenizer_path),
        "raw_string": raw,
        "canonical_encoder_ids": canonical,
        "alternative_generated_id_paths": token_paths[1:],
        "all_decoded_strings": decoded,
        "special_eos_id": manifest["tokenizer"]["special_ids"]["<|eos|>"],
        "meaning": "确定性编码器只选一条规范路径；生成模型的完整词表仍可输出其他编号路径，它们也会解回同一字串。一个规范路径的概率不是所有解码路径概率之和。",
    }
    OUT.parent.mkdir(exist_ok=True)
    OUT.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(report, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
