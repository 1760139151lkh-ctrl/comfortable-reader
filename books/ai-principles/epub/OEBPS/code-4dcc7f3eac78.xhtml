"""C25: one-batch gradient geometry between new response and old text losses."""
 
from __future__ import annotations
 
import argparse
import math
from pathlib import Path
 
import numpy as np
import torch
from torch.nn import functional as F
 
from train_dolly_sft import (
    ROOT, choose_device, load_base, load_sft_manifest, load_sft_split,
    torch_batch, write_json,
)
from train_causal_wikitext import (
    autocast_for, batch as wiki_batch, load_split as load_wiki_split,
)
 
 
def gradient(model: torch.nn.Module) -> torch.Tensor:
    return torch.cat([
        parameter.grad.detach().float().flatten()
        for parameter in model.parameters() if parameter.grad is not None
    ])
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument(
        "--out", type=Path, default=Path("work/results/c25_gradient_interference_probe.json")
    )
    args = parser.parse_args()
    out = args.out if args.out.is_absolute() else ROOT / args.out
    if out.exists():
        raise FileExistsError(f"keep previous gradient result: {out}")
    sft_manifest, sft_sha = load_sft_manifest()
    sft = load_sft_split(sft_manifest, "train")
    model, _, base = load_base()
    wiki = load_wiki_split(base["wiki_manifest"], "train")
    device = choose_device()
    model.to(device).train()
    sft_indices = np.arange(64)
    wiki_indices = np.arange(64)
    s_ids, s_labels, s_real = torch_batch(sft, sft_indices, device)
    w_ids, w_labels, w_real = wiki_batch(wiki, wiki_indices, device)
 
    model.zero_grad(set_to_none=True)
    with autocast_for(device):
        s_logits = model(s_ids, s_real)
    s_loss = F.cross_entropy(
        s_logits.float().transpose(1, 2), s_labels,
        ignore_index=-100, reduction="sum",
    ) / (s_labels != -100).sum()
    s_loss.backward()
    g_new = gradient(model)
 
    model.zero_grad(set_to_none=True)
    with autocast_for(device):
        w_logits = model(w_ids, w_real)
    w_loss = F.cross_entropy(
        w_logits.float().transpose(1, 2), w_labels,
        ignore_index=-100, reduction="sum",
    ) / (w_labels != -100).sum()
    w_loss.backward()
    g_old = gradient(model)
 
    dot = float(torch.dot(g_new, g_old))
    norm_new = float(g_new.norm())
    norm_old = float(g_old.norm())
    cosine = dot / (norm_new * norm_old)
    if not math.isfinite(cosine):
        raise RuntimeError("nonfinite gradient alignment")
    result = {
        "scope": "C19 pretrained checkpoint, first 64 Dolly train records and first 64 WikiText train blocks; two gradients at same fixed weights, no update, no validation/test opened",
        "base_checkpoint_sha256": base["c19_checkpoint_sha256"],
        "sft_manifest_sha256": sft_sha,
        "sft_source_row_first_last": [int(sft["source_row"][0]), int(sft["source_row"][63])],
        "sft_response_targets": int((s_labels != -100).sum()),
        "wiki_targets": int((w_labels != -100).sum()),
        "sft_loss": float(s_loss.detach()),
        "wiki_loss": float(w_loss.detach()),
        "new_gradient_norm": norm_new,
        "old_gradient_norm": norm_old,
        "gradient_dot": dot,
        "gradient_cosine": cosine,
        "plain_SGD_first_order_old_loss_change_for_rate_0_0002": -0.0002 * dot,
        "caution": "single batch at base weights and plain SGD local approximation; actual three-epoch AdamW forgetting has more causes",
    }
    write_json(out, result)
    print("gradient cosine", cosine, "dot", dot)
    print("plain-SGD local old-loss change estimate", result["plain_SGD_first_order_old_loss_change_for_rate_0_0002"])
 
 
if __name__ == "__main__":
    main()
