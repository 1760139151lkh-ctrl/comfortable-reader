"""Fetch one pinned public CommonRoad control example and its repo license."""
from __future__ import annotations
 
import argparse
import hashlib
import json
import urllib.request
import xml.etree.ElementTree as ET
from pathlib import Path
 
WORK = Path(__file__).resolve().parents[1]
OUT = WORK / "data/c42_commonroad"
COMMIT = "0b672f9d2164cd2b2698fbfcd5046ba492c75ccd"
BASE = f"https://raw.githubusercontent.com/CommonRoad/commonroad-control/{COMMIT}/"
FILES = {
    "DEU_Backnang-9_1_T-1.xml": ("scenarios/DEU_Backnang-9_1_T-1.xml", 177156,
                                    "f642ad66ecdb73ef4e15c57da201773fe739f6f707a3cf64c4efb5c265afc361"),
    "ITA_Foggia-6_1_T-1.xml": ("scenarios/ITA_Foggia-6_1_T-1.xml", 329353,
                                "2563a7dd4eedb60ef460d37afe4c0b718510092289f81341eb967bd3f741b8b4"),
    "LICENSE": ("LICENSE", 1612, "60a4cd8990098ce0cca20f090ba4af5906b21d3cb43c94e3bc987a2b62d44cb6"),
}
 
 
def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()
 
 
def fetch():
    OUT.mkdir(parents=True, exist_ok=True)
    items = []
    for name, (relative, size, digest) in FILES.items():
        target = OUT / name
        if target.exists():
            data = target.read_bytes()
        else:
            request = urllib.request.Request(BASE + relative, headers={"User-Agent": "C42 local textbook study"})
            with urllib.request.urlopen(request, timeout=30) as response:
                data = response.read()
            if len(data) != size or sha(data) != digest:
                raise RuntimeError("source bytes differ from pinned expectation")
            target.write_bytes(data)
        if len(data) != size or sha(data) != digest:
            raise RuntimeError(f"existing {name} differs from pinned source")
        items.append({"file": str(target.relative_to(WORK)), "url": BASE + relative,
                      "bytes": size, "sha256": digest})
    scenarios = []
    for name in ("DEU_Backnang-9_1_T-1.xml", "ITA_Foggia-6_1_T-1.xml"):
        root = ET.fromstring((OUT / name).read_bytes())
        if root.tag != "commonRoad" or root.get("benchmarkID") != name[:-4]:
            raise RuntimeError("unexpected scenario identity")
        scenarios.append({"file": name, "header": root.attrib,
                          "top_level_counts": {tag: len(root.findall(tag)) for tag in
                                               ("lanelet", "dynamicObstacle", "staticObstacle", "trafficSign", "intersection", "planningProblem")}})
    report = {"source_commit": COMMIT, "files": items, "scenarios": scenarios,
              "source_identity": "CommonRoad examples: road geometry from OpenStreetMap, other traffic from SUMO; not raw measured driving logs or proof of vehicle safety"}
    path = OUT / "manifest_v2.json"
    if path.exists() and json.loads(path.read_text(encoding="utf-8")) != report:
        raise RuntimeError("existing manifest differs")
    if not path.exists():
        path.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"scenarios": [{"name": x["header"]["benchmarkID"], "counts": x["top_level_counts"]} for x in scenarios]},
                     ensure_ascii=False))
 
 
if __name__ == "__main__":
    argparse.ArgumentParser().parse_args()
    fetch()
