"""Sample a frozen waveform GRU from a voiced held-out chapter prefix."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
import time
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
from scipy.signal import resample_poly, stft
import soundfile as sf
import torch
 
from c32_waveform_generator import SampleGRU, generate
from prepare_c32_wave_generation import decode_mulaw, encode_mulaw
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/librispeech_dev_clean"
WAVEGEN = DATA / "wavegen_speaker1272"
 
 
def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path, required=True)
    parser.add_argument("--out-dir", type=Path, required=True)
    parser.add_argument("--seconds", type=float, default=1.0)
    args = parser.parse_args()
    run = args.run_dir if args.run_dir.is_absolute() else ROOT / args.run_dir
    out = args.out_dir if args.out_dir.is_absolute() else ROOT / args.out_dir
    if out.exists():
        raise FileExistsError(f"preserve existing waveform samples: {out}")
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    checkpoint = torch.load(run / "best.pt", map_location="cpu", weights_only=False)
    if (report["best_checkpoint_sha256"] != sha256(run / "best.pt") or
            checkpoint["source_manifest_sha256"] != sha256(WAVEGEN / "manifest.json")):
        raise RuntimeError("frozen waveform model/data mismatch")
    split = json.loads((DATA / "chapter_split.json").read_text(encoding="utf-8"))
    candidate = next(row for row in split["rows"] if row["speaker"] == "1272"
                     and row["chapter"] == "135031")
    wave, sr = sf.read(ROOT / candidate["flac"], dtype="float32")
    if sr != 16000:
        raise RuntimeError("source audio changed")
    low = resample_poly(wave, 1, 2).astype(np.float32)
    continuation_count = int(round(args.seconds * 8000))
    if len(low) < continuation_count + 256:
        raise RuntimeError("validation audio too short")
    prefix_count = 128
    candidates = range(0, len(low) - continuation_count - prefix_count, 64)
    begin = max(candidates, key=lambda i: float(np.mean(low[i:i + prefix_count] ** 2)))
    prefix = encode_mulaw(low[begin:begin + prefix_count])
    true_continuation = low[begin:begin + prefix_count + continuation_count]
    torch.set_num_threads(1)
    model = SampleGRU()
    model.load_state_dict(checkpoint["model"])
    start = time.perf_counter()
    new_codes = generate(model, prefix, continuation_count, 0.9, 64, 20260924)
    sample_seconds = time.perf_counter() - start
    new_wave = decode_mulaw(np.concatenate([prefix, new_codes]))
    out.mkdir(parents=True)
    sf.write(out / "voiced_validation_prefix_then_new_audio.wav", new_wave, 8000)
    sf.write(out / "actual_validation_speech_reference.wav", true_continuation, 8000)
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    fig, axes = plt.subplots(2, 1, figsize=(10, 7), layout="constrained")
    for axis, title, audio in (
        (axes[0], "同一开头后的真实朗读（不是生成的唯一正确答案）", true_continuation),
        (axes[1], "同一开头后的模型新采样（没有输入文字）", new_wave),
    ):
        freqs, times, spectrum = stft(audio.astype(np.float32), fs=8000,
                                       window="hann", nperseg=200,
                                       noverlap=120, nfft=256,
                                       boundary="zeros", padded=True)
        db = 20 * np.log10(np.maximum(np.abs(spectrum), 1e-7))
        axis.pcolormesh(times, freqs, db - db.max(), shading="auto",
                        cmap="magma", vmin=-65, vmax=0)
        axis.set(ylim=(0, 4000), xlabel="时间（秒）", ylabel="频率（Hz）", title=title)
        axis.axvline(prefix_count / 8000, color="white", linestyle="--", linewidth=1)
    fig.savefig(out / "real_vs_model_waveform_spectrum.png", dpi=160)
    plt.close(fig)
    receipt = {
        "identity": "frozen next-sample GRU conditioned on 128 samples from a validation-chapter voice; no text-to-speech",
        "train_report_sha256": sha256(run / "train.json"),
        "checkpoint_sha256": sha256(run / "best.pt"),
        "source_archive_sha256": split["archive_sha256"],
        "prefix_clip_id": candidate["id"],
        "prefix_original_split": candidate["split"],
        "prefix_wavegen_partition": "validation",
        "prefix_begin_seconds": begin / 8000,
        "prefix_samples": prefix_count,
        "generated_samples": continuation_count,
        "generated_sample_rate": 8000,
        "temperature": 0.9,
        "top_k": 64,
        "generated_wall_seconds": sample_seconds,
        "generated_rms_after_prefix": float(np.sqrt(np.mean(decode_mulaw(new_codes) ** 2))),
        "real_reference_rms_after_prefix": float(np.sqrt(np.mean(true_continuation[prefix_count:] ** 2))),
        "generated_distinct_codes": len(np.unique(new_codes)),
        "figure_sha256": sha256(out / "real_vs_model_waveform_spectrum.png"),
        "limitation": "sample-level likelihood and a visible/audible output do not certify intelligible words, speaker identity, or text-controlled synthesis",
    }
    (out / "sample.json").write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + "\n",
                                     encoding="utf-8")
    print(json.dumps(receipt, ensure_ascii=False), flush=True)
 
 
if __name__ == "__main__":
    main()
