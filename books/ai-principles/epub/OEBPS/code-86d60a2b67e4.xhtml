"""Use C13's separately trained classifier as a limited Fashion guidance diagnostic."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
 
from fashion_ddpm import ROOT, DATA, digest
from fashion_spatial_training import CNN
from prepare_fashion_mnist import load_images, load_labels
 
 
def pairwise_pixel_rmse(images: np.ndarray) -> float:
    flat = images.astype(np.float32).reshape(len(images), -1) / 255
    values = [
        float(np.sqrt(np.mean((flat[i] - flat[j]) ** 2)))
        for i in range(len(images)) for j in range(i + 1, len(images))
    ]
    return float(np.mean(values))
 
 
@torch.no_grad()
def judge(classifier, images: np.ndarray, device):
    x = torch.from_numpy(images[:, None].copy()).to(
        device=device, dtype=torch.float32
    ) / 255
    probability = classifier(x).softmax(dim=1)
    return {
        "count": len(images),
        "predicted_class9_count": int((probability.argmax(dim=1) == 9).sum()),
        "mean_classifier_probability_class9": float(probability[:, 9].mean()),
        "pairwise_raw_pixel_rmse_mean": pairwise_pixel_rmse(images),
    }
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c31_fashion_ddpm"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c31_fashion_guidance_diagnostic.json"))
    args = parser.parse_args()
    run, out = ROOT / args.run_dir, ROOT / args.out
    if out.exists():
        raise FileExistsError("preserve existing guidance diagnostic")
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    if digest(ROOT / report["checkpoint"]) != report["checkpoint_sha256"]:
        raise RuntimeError("DDPM checkpoint changed")
    c13_report = json.loads(
        (ROOT / "work/results/fashion_spatial_training.json").read_text(encoding="utf-8")
    )
    c13 = c13_report["arms"]["cnn_original"]
    checkpoint_path = ROOT / "work" / c13["checkpoint"]
    if digest(checkpoint_path) != c13["checkpoint_sha256"]:
        raise RuntimeError("C13 supervised CNN changed")
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    classifier = CNN().to(device)
    checkpoint = torch.load(checkpoint_path, map_location="cpu", weights_only=True)
    classifier.load_state_dict(checkpoint["state_dict"])
    classifier.eval()
    label_path = DATA / "train-labels-idx1-ubyte.gz"
    image_path = DATA / "train-images-idx3-ubyte.gz"
    if (digest(label_path) != report["source_identity"]["train_labels_sha256"] or
            digest(image_path) != report["source_identity"]["train_images_sha256"]):
        raise RuntimeError("Fashion source changed")
    split = np.load(DATA / "chapter_split.npz")
    val_rows = split["validation_indices"]
    labels = load_labels(label_path, 60000)
    real_boot_rows = val_rows[labels[val_rows] == 9]
    real_images = load_images(image_path, 60000)[real_boot_rows]
    results = {"real_validation_class9": judge(classifier, real_images, device)}
    for item in report["samples"]:
        path = ROOT / item["path"]
        if digest(path) != item["sha256"]:
            raise RuntimeError(f"generated array changed: {item['name']}")
        images = np.load(path, allow_pickle=False)
        results[item["name"]] = judge(classifier, images, device)
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps({
        "scope": "C13 supervised Fashion gray CNN is an external, limited class9 recognizability probe; calibrated on C13 validation boots; neither human image quality nor user preference",
        "c13_checkpoint_sha256": c13["checkpoint_sha256"],
        "ddpm_checkpoint_sha256": report["checkpoint_sha256"],
        "source_identity": report["source_identity"],
        "results": results,
        "limitations": "C13 classifier was trained on the same Fashion corpus, can be fooled by generated artifacts, and pairwise pixel RMSE does not measure semantic diversity",
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("real boot recall", results["real_validation_class9"]["predicted_class9_count"],
          "/", results["real_validation_class9"]["count"])
    for name in ("ankle_boot_w0", "ankle_boot_w2", "ankle_boot_w5"):
        print(name, results[name]["predicted_class9_count"],
              "mean probability", results[name]["mean_classifier_probability_class9"],
              "pair RMSE", results[name]["pairwise_raw_pixel_rmse_mean"])
 
 
if __name__ == "__main__":
    main()
