"""Verify the exact two-direction image/text matching gradient on three pairs."""
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import torch
from torch.nn import functional as F
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("result exists")
    tau = 0.2
    cosine = torch.tensor([[0.7, 0.1, 0.2], [0.3, 0.6, 0.1], [0.2, 0.4, 0.5]], dtype=torch.float64, requires_grad=True)
    score = cosine / tau
    target = torch.arange(3)
    loss = (F.cross_entropy(score, target) + F.cross_entropy(score.T, target)) / 2
    loss.backward()
    with torch.no_grad():
        row_probability = score.softmax(dim=1)
        col_probability = score.softmax(dim=0)
        expected = (row_probability + col_probability - 2 * torch.eye(3, dtype=torch.float64)) / (2 * 3 * tau)
        maximum_difference = float((cosine.grad - expected).abs().max())
        if maximum_difference > 1e-12:
            raise RuntimeError("analytic and autograd gradient differ")
        result = {"cosine": cosine.tolist(), "temperature": tau, "symmetric_loss": float(loss), "row_probability": row_probability.tolist(), "column_probability": col_probability.tolist(), "analytic_derivative_wrt_raw_cosine": expected.tolist(), "autograd_max_absolute_difference": maximum_difference}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"loss": result["symmetric_loss"], "max_difference": maximum_difference}))
 
 
if __name__ == "__main__":
    main()
