"""Pairwise reward proxy on real CoVal votes, with prompt-grouped heldout data.
 
This is a TF-IDF/hand-feature model, separate from the C19/C25 language model.
It predicts rankings in the supplied English cohort; it is not the user's own
preference, visual aesthetic judgment, or a reward oracle.
"""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
import math
import re
from pathlib import Path
 
import numpy as np
from scipy import sparse
from scipy.special import expit
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/coval_eec757/pairs_v1"
LETTERS = "ABCD"
 
 
def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def load_manifest() -> tuple[dict, str]:
    path = DATA / "manifest.json"
    manifest = json.loads(path.read_text(encoding="utf-8"))
    for split, expected in manifest["files"].items():
        if digest(DATA / f"{split}.json") != expected:
            raise RuntimeError(f"{split} pair file changed")
    return manifest, digest(path)
 
 
def load_split(split: str) -> list[dict]:
    return json.loads((DATA / f"{split}.json").read_text(encoding="utf-8"))
 
 
def words(text: str) -> set[str]:
    return {item for item in re.findall(r"[a-zA-Z]{3,}", text.casefold())
            if item not in {"the", "and", "for", "you", "are", "that", "this", "with"}}
 
 
def numeric_features(question: str, response: str) -> list[float]:
    p = words(question)
    r = words(response)
    return [
        math.log1p(len(response)),
        len(p & r) / max(1, len(p)),
        len(r) / max(1, len(re.findall(r"[a-zA-Z]+", response))),
        response.count("?"),
    ]
 
 
def train_candidate_features(rows: list[dict]):
    documents = [row["responses"][letter]
                 for row in rows for letter in LETTERS]
    vectorizer = TfidfVectorizer(
        lowercase=True, strip_accents="unicode", ngram_range=(1, 2),
        min_df=3, max_features=12000, sublinear_tf=True,
    )
    vectorizer.fit(documents)
    numbers = np.array([numeric_features(row["prompt"], row["responses"][letter])
                        for row in rows for letter in LETTERS], dtype=np.float64)
    scaler = StandardScaler().fit(numbers)
    return vectorizer, scaler
 
 
def features(rows: list[dict], vectorizer, scaler):
    documents = [row["responses"][letter]
                 for row in rows for letter in LETTERS]
    text = vectorizer.transform(documents).tocsr()
    numbers = np.array([numeric_features(row["prompt"], row["responses"][letter])
                        for row in rows for letter in LETTERS], dtype=np.float64)
    numeric = sparse.csr_matrix(scaler.transform(numbers))
    return sparse.hstack([text, numeric], format="csr")
 
 
def fit_data(rows: list[dict], all_features):
    positive_rows = []
    negative_rows = []
    positive_weights = []
    negative_weights = []
    for position, row in enumerate(rows):
        base = 4 * position
        for pair, record in row["pairs"].items():
            n_a, n_b = record["world_votes"]
            if n_a + n_b == 0:
                continue
            difference = all_features[base + LETTERS.index(pair[0])]
            difference = difference - all_features[base + LETTERS.index(pair[1])]
            if n_a:
                positive_rows.append(difference)
                positive_weights.append(n_a)
            if n_b:
                negative_rows.append(difference)
                negative_weights.append(n_b)
    matrix = sparse.vstack(positive_rows + negative_rows, format="csr")
    labels = np.array([1] * len(positive_rows) + [0] * len(negative_rows))
    weights = np.array(positive_weights + negative_weights, dtype=np.float64)
    return matrix, labels, weights
 
 
def evaluate(rows: list[dict], all_features, model) -> dict:
    score = np.asarray(all_features @ model.coef_[0]).reshape(-1)
    vote_loss = 0.0
    ballots = 0
    majority_correct = 0
    majority_total = 0
    candidate_selected_win_rates = []
    candidate_longest_win_rates = []
    candidate_oracle_win_rates = []
    for position, row in enumerate(rows):
        candidate_scores = {letter: float(score[4 * position + i])
                            for i, letter in enumerate(LETTERS)}
        selected = max(LETTERS, key=lambda letter: candidate_scores[letter])
        longest = max(LETTERS, key=lambda letter: len(row["responses"][letter]))
        empirical_win = {}
        for a in LETTERS:
            wins = total = 0
            for b in LETTERS:
                if a == b:
                    continue
                key = "".join(sorted((a, b)))
                c0, c1 = row["pairs"][key]["world_votes"]
                count_a, count_b = (c0, c1) if a == key[0] else (c1, c0)
                wins += count_a
                total += count_a + count_b
            empirical_win[a] = wins / total if total else 0.5
        candidate_selected_win_rates.append(empirical_win[selected])
        candidate_longest_win_rates.append(empirical_win[longest])
        candidate_oracle_win_rates.append(max(empirical_win.values()))
        for pair, record in row["pairs"].items():
            a, b = pair
            n_a, n_b = record["world_votes"]
            if n_a + n_b == 0:
                continue
            probability = float(np.clip(
                expit(candidate_scores[a] - candidate_scores[b]), 1e-8, 1-1e-8
            ))
            vote_loss -= n_a * math.log(probability) + n_b * math.log(1-probability)
            ballots += n_a + n_b
            if n_a != n_b:
                majority_total += 1
                majority_correct += int((probability > 0.5) == (n_a > n_b))
    return {
        "prompts": len(rows),
        "world_non_tie_ballots_across_six_pairs": ballots,
        "vote_cross_entropy": vote_loss / ballots,
        "constant_half_vote_cross_entropy": math.log(2),
        "pair_majority_accuracy": majority_correct / majority_total,
        "pair_majority_items": majority_total,
        "mean_human_pairwise_win_rate_of_proxy_selected_candidate":
            float(np.mean(candidate_selected_win_rates)),
        "mean_human_pairwise_win_rate_of_longest_candidate":
            float(np.mean(candidate_longest_win_rates)),
        "mean_human_pairwise_win_rate_of_empirical_best_candidate":
            float(np.mean(candidate_oracle_win_rates)),
    }
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c28_coval_reward_fit.json"))
    args = parser.parse_args()
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError(f"preserve existing result: {out}")
    manifest, manifest_sha = load_manifest()
    train = load_split("train")
    validation = load_split("validation")
    vectorizer, scaler = train_candidate_features(train)
    train_features = features(train, vectorizer, scaler)
    validation_features = features(validation, vectorizer, scaler)
    matrix, labels, weights = fit_data(train, train_features)
    candidates = []
    selected = None
    best_loss = float("inf")
    for c in (0.01, 0.1, 1.0, 10.0):
        model = LogisticRegression(
            fit_intercept=False, C=c, solver="liblinear", max_iter=200,
            random_state=20260924,
        )
        model.fit(matrix, labels, sample_weight=weights)
        measured = evaluate(validation, validation_features, model)
        candidates.append({"C": c, "validation": measured})
        if measured["vote_cross_entropy"] < best_loss:
            best_loss = measured["vote_cross_entropy"]
            selected = model
            selected_c = c
    if selected is None:
        raise RuntimeError("no model chosen")
    train_result = evaluate(train, train_features, selected)
    validation_result = evaluate(validation, validation_features, selected)
    # Only the final chosen model opens the held-out prompt groups.
    test = load_split("test")
    test_features = features(test, vectorizer, scaler)
    test_result = evaluate(test, test_features, selected)
    numeric_coefs = selected.coef_[0][-4:].tolist()
    report = {
        "scope": "prompt-grouped CoVal world-view votes; small TF-IDF/length/overlap pairwise reward proxy; not C25 LM, not visual/user aesthetic; test opened after validation C choice",
        "source_manifest_sha256": manifest_sha,
        "source_counts": manifest["prompt_counts"],
        "feature_design": {
            "text": "train-response TF-IDF unigrams/bigrams max12000, min_df3",
            "numeric": ["log response length", "prompt-response word overlap",
                        "response distinct-word ratio", "question mark count"],
            "pair_feature": "candidate A minus B, no intercept or candidate index features",
            "training": "each human non-tie world vote weights logistic pair likelihood",
        },
        "train_fit": {"feature_count": int(train_features.shape[1]),
                      "pair_label_rows": int(matrix.shape[0]),
                      "weighted_individual_votes": float(weights.sum())},
        "validation_C_selection": candidates,
        "selected_C": selected_c,
        "selected_numeric_coefficients_on_standardized_features": numeric_coefs,
        "train": train_result,
        "validation": validation_result,
        "test": test_result,
        "test_opened_once": True,
    }
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("selected C", selected_c, "val loss", validation_result["vote_cross_entropy"],
          "test loss", test_result["vote_cross_entropy"],
          "test majority", test_result["pair_majority_accuracy"])
 
 
if __name__ == "__main__":
    main()
