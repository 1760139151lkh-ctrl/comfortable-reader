"""Check selected six-category real-photo object question model once on held-out IDs."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
 
from c35_image_text_retrieval import DATA
from c35_object_presence_vqa import OBJECT, ObjectQuestionNet, evaluate
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--checkpoint", type=Path, default=Path("work/runs/c35_object_presence30/best.pt"))
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("result exists")
    torch.set_num_threads(8)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    source = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    rows = json.loads((DATA / "image_records.json").read_text(encoding="utf-8"))
    by_image = {r["image_id"]: i for i, r in enumerate(rows)}
    images = np.load(DATA / f"images_{source['image_side']}_chw_uint8.npy", mmap_mode="r")
    records = json.loads((OBJECT / "records.json").read_text(encoding="utf-8"))
    indices = [i for i, item in enumerate(records) if item["split"] == "test"]
    if len(indices) != 990:
        raise RuntimeError("test changed")
    ckpt = torch.load(a.checkpoint, map_location="cpu", weights_only=True)
    model = ObjectQuestionNet().to(device)
    model.load_state_dict(ckpt["model"])
    actual = evaluate(model, records, indices, images, by_image, device)
    image_ids = sorted({records[i]["image_id"] for i in indices})
    shift = {iid: image_ids[(j + 73) % len(image_ids)] for j, iid in enumerate(image_ids)}
    replaced = [dict(r) for r in records]
    for i in indices:
        replaced[i]["image_id"] = shift[records[i]["image_id"]]
    wrong = evaluate(model, replaced, indices, images, by_image, device)
    result = {"identity": "Six-category fixed-template binary questions from COCO official instance annotations, NOT natural human VQA", "category_choice": "after viewing per-category counts across all splits; test models never used for selection", "test_questions": len(indices), "selected_epoch": ckpt["epoch"], "checkpoint_sha256": hashlib.sha256(a.checkpoint.read_bytes()).hexdigest(), "balanced_question_only_chance": 0.5, "real_image": actual, "cyclically_wrong_image": wrong}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"real": {k:v for k,v in actual.items() if k!='per_category'}, "wrong": {k:v for k,v in wrong.items() if k!='per_category'}, "per_category": actual['per_category']}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
