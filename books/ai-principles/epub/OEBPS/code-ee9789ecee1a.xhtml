"""按实际矩阵形状数一层自身读取和逐位置前馈的乘法量级。"""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
 
OUT = Path(__file__).resolve().parents[1] / "results" / "attention_cost_probe.json"
 
 
def one_length(n: int, width: int, inner: int, heads: int) -> dict:
    # 一次矩阵乘的 m×k 乘 k×p 约有 mkp 项乘法；加法、偏置、softmax 等另计。
    qkv_and_output = 4 * n * width * width
    score_and_weighted_values = 2 * n * n * width
    positionwise_ffn = 2 * n * width * inner
    included = qkv_and_output + score_and_weighted_values + positionwise_ffn
    return {
        "positions": n,
        "qkv_and_output_linear_multiplications": qkv_and_output,
        "score_and_weighted_value_multiplications": score_and_weighted_values,
        "two_ffn_linears_multiplications": positionwise_ffn,
        "included_total_multiplications": included,
        "position_pair_fraction_of_included_total": score_and_weighted_values / included,
        "one_attention_weight_matrix_entries": heads * n * n,
        "one_fp32_attention_weight_matrix_mib": heads * n * n * 4 / 1024**2,
    }
 
 
def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--lengths", nargs="+", type=int, default=[18, 128, 512])
    parser.add_argument("--width", type=int, default=96)
    parser.add_argument("--inner", type=int, default=192)
    parser.add_argument("--heads", type=int, default=4)
    parser.add_argument("--out", type=Path, default=OUT, help="结果 JSON；须写在 work/results/ 内")
    args = parser.parse_args()
    if min(args.lengths) <= 0 or min(args.width, args.inner, args.heads) <= 0 or args.width % args.heads:
        parser.error("长度、宽度、内宽和头数须为正；宽度须可整除头数")
    output = args.out.resolve()
    if not output.is_relative_to(OUT.parent.resolve()):
        parser.error("结果文件须在本项目 work/results/ 内")
    report = {
        "width": args.width,
        "ffn_inner_width": args.inner,
        "heads": args.heads,
        "rows": [one_length(n, args.width, args.inner, args.heads) for n in args.lengths],
        "scope": "一层密集自身读取的Q/K/V/输出四次线性映射、两次位置配对矩阵乘及两次逐位置FFN线性层；仅数乘法项，不含偏置、激活、归一化、softmax、词表输出、反向传播或硬件时间。",
    }
    output.parent.mkdir(exist_ok=True)
    output.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(report, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
