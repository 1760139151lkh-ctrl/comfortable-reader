"""Exact two-hop hub bottleneck and its dependence on aggregator choice."""
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import torch
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("result exists")
    result = []
    for leaves in (1, 4, 16, 64):
        signal = torch.tensor(1.0, dtype=torch.float64, requires_grad=True)
        others = torch.zeros(leaves - 1, dtype=torch.float64)
        leaf_values = torch.cat([signal[None], others])
        # hub sees its leaves and the target root (which starts at zero);
        # root sees just the hub on the next synchronous round.
        root_zero = torch.tensor([0.0], dtype=torch.float64)
        hub_mean = torch.cat([leaf_values, root_zero]).mean()
        root_two_hops = hub_mean
        root_two_hops.backward()
        mean_grad = float(signal.grad)
        expected = 1.0 / (leaves + 1)
        if abs(mean_grad - expected) > 1e-14:
            raise RuntimeError("mean bottleneck gradient mismatch")
        signal2 = torch.tensor(1.0, dtype=torch.float64, requires_grad=True)
        sum_output = torch.cat([signal2[None], others, root_zero]).sum()
        sum_output.backward()
        result.append({"source_leaves_behind_one_hub": leaves, "mean_aggregator_one_source_to_root_derivative": mean_grad, "formula_1_over_leaves_plus_one": expected, "sum_aggregator_same_source_derivative": float(signal2.grad), "hub_state_coordinates": 1})
    output = {"task": "2-hop star: one root wants a far leaf signal through one scalar hub; only aggregator is changed", "scope": "exact derivative example, not a universal impossibility theorem or a real-data performance result", "records": result}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(output, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
