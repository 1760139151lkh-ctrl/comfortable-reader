"""Compare one stochastic DDPM chain with deterministic DDIM-style skip paths."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
from fashion_ddpm import ROOT, DATA, digest
from prepare_fashion_mnist import load_images, load_labels
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--ddpm-run", type=Path,
                        default=Path("work/runs/c31_fashion_ddpm"))
    parser.add_argument("--ddim-run", type=Path,
                        default=Path("work/runs/c31_fashion_ddim"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/figures/c31_fashion_ddim_steps.png"))
    parser.add_argument("--receipt", type=Path,
                        default=Path("work/results/c31_fashion_ddim_figure.json"))
    args = parser.parse_args()
    ddpm, ddim = ROOT / args.ddpm_run, ROOT / args.ddim_run
    out, receipt = ROOT / args.out, ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing DDIM figure")
    parent = json.loads((ddpm / "train.json").read_text(encoding="utf-8"))
    record = json.loads((ddim / "sample.json").read_text(encoding="utf-8"))
    if (record["parent_training_report_sha256"] != digest(ddpm / "train.json")
            or record["checkpoint_sha256"] != parent["checkpoint_sha256"]):
        raise RuntimeError("DDIM parent changed")
    split = np.load(DATA / "chapter_split.npz")
    labels = load_labels(DATA / "train-labels-idx1-ubyte.gz", 60000)
    val = split["validation_indices"]
    selected = val[labels[val] == 9][:8]
    actual = load_images(DATA / "train-images-idx3-ubyte.gz", 60000)[selected]
    rows = [("真实验证踝靴", actual)]
    ddpm_item = next(
        item for item in parent["samples"] if item["name"] == "ankle_boot_w0"
    )
    ddpm_path = ROOT / ddpm_item["path"]
    if digest(ddpm_path) != ddpm_item["sha256"]:
        raise RuntimeError("DDPM samples changed")
    rows.append(("随机逆链300步", np.load(ddpm_path, allow_pickle=False)[:8]))
    for item in reversed(record["results"]):
        path = ROOT / item["path"]
        if digest(path) != item["sha256"]:
            raise RuntimeError("DDIM sample changed")
        rows.append((
            f"确定逆路{item['steps']}步",
            np.load(path, allow_pickle=False)[:8],
        ))
    fig, axes = plt.subplots(len(rows), 8, figsize=(13, 1.7 * len(rows)))
    for row, (title, images) in enumerate(rows):
        for col in range(8):
            axes[row, col].imshow(
                images[col], cmap="gray", vmin=0, vmax=255,
                interpolation="nearest",
            )
            axes[row, col].axis("off")
        fig.text(0.01, 1 - (row + 0.5) / len(rows), title,
                 va="center", fontsize=10, fontproperties="Microsoft YaHei")
    fig.tight_layout(rect=(0.15, 0, 1, 1))
    out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=180)
    plt.close(fig)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps({
        "scope": "old real Fashion class9 validation images, same initial Gaussian seed with stochastic 300-step ancestral versus deterministic eta0 DDIM-style 300/100/30/10 paths; visual only",
        "checkpoint_sha256": parent["checkpoint_sha256"],
        "parent_train_report_sha256": digest(ddpm / "train.json"),
        "ddim_report_sha256": digest(ddim / "sample.json"),
        "real_validation_rows": selected.astype(int).tolist(),
        "row_titles": [name for name, _ in rows],
        "figure": str(out.relative_to(ROOT)).replace("\\", "/"),
        "official_test_opened": False,
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
 
 
if __name__ == "__main__":
    main()
