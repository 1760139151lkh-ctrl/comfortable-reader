"""Count CoVal prompt/response fit under the existing 128-position C19 tokenizer."""
 
from __future__ import annotations
 
import argparse
import json
import statistics
from pathlib import Path
 
from tokenizers import Tokenizer
 
 
ROOT = Path(__file__).resolve().parents[2]
SOURCE = ROOT / "work/data/coval_eec757/comparisons.jsonl"
TOKENIZER = ROOT / "work/data/wikitext2_causal/tokenizer.json"
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c28_coval_length_audit.json"))
    args = parser.parse_args()
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError(out)
    tokenizer = Tokenizer.from_file(str(TOKENIZER))
    prompt_lengths = []
    response_lengths = []
    fit_counts = []
    for line in SOURCE.open(encoding="utf-8"):
        item = json.loads(line)
        user_prompt = "\n".join(message["content"] for message in item["prompt"]["messages"]
                                if message["role"] == "user")
        prompt_size = len(tokenizer.encode(user_prompt, add_special_tokens=False).ids)
        sizes = [len(tokenizer.encode(message["content"], add_special_tokens=False).ids)
                 for response in item["responses"] for message in response["messages"]
                 if message["role"] == "assistant"]
        if len(sizes) != 4:
            raise RuntimeError("current CoVal row is not four single assistant candidates")
        prompt_lengths.append(prompt_size)
        response_lengths.extend(sizes)
        fit_counts.append(sum(1 + prompt_size + size + 1 <= 128 for size in sizes))
    result = {
        "scope": "length-only audit of current four candidates and user prompt under existing C18 BPE/C19 128 positions; no preference label/model training",
        "prompts": len(fit_counts),
        "prompt_length_median": statistics.median(prompt_lengths),
        "response_length_median": statistics.median(response_lengths),
        "all_four_fit": sum(count == 4 for count in fit_counts),
        "at_least_two_fit": sum(count >= 2 for count in fit_counts),
        "none_fit": sum(count == 0 for count in fit_counts),
        "fit_count_distribution": {str(n): sum(count == n for count in fit_counts)
                                   for n in range(5)},
    }
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(result)
 
 
if __name__ == "__main__":
    main()
