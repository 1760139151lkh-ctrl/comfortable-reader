"""Limited pixel-distance, diversity and exact-byte-copy checks for C30 GAN."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
 
from cifar_ssl_lab import ROOT, train_pixels, digest
 
 
def mean_pairwise_rmse(images: torch.Tensor) -> float:
    flat = images.flatten(1)
    return float(torch.pdist(flat).mean() / flat.shape[1] ** 0.5)
 
 
@torch.no_grad()
def nearest_training_rmse(query: torch.Tensor, training: torch.Tensor):
    query = query.flatten(1)
    training = training.flatten(1)
    q_norm = query.square().sum(dim=1, keepdim=True)
    answers = torch.full((len(query),), float("inf"), device=query.device)
    for start in range(0, len(training), 5000):
        candidate = training[start:start + 5000]
        squared = (q_norm + candidate.square().sum(dim=1)[None, :] -
                   2 * query @ candidate.T).clamp_min(0)
        answers = torch.minimum(answers, squared.min(dim=1).values)
    values = (answers / query.shape[1]).sqrt().cpu().numpy()
    return {"mean": float(values.mean()),
            "min": float(values.min()), "max": float(values.max())}
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c30_cifar_small_gan"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c30_cifar_gan_pixel_diagnostic.json"))
    args = parser.parse_args()
    run = ROOT / args.run_dir
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError("preserve existing GAN diagnostic")
    final_preview = report["fixed_noise_previews"][-1]
    path = ROOT / final_preview["path"]
    if digest(path) != final_preview["sha256"]:
        raise RuntimeError("GAN final samples changed")
    generated_bytes = np.load(path, allow_pickle=False)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    all_images, train_rows, validation_rows, source = train_pixels(device)
    if source != report["source_identity"]:
        raise RuntimeError("GAN source identity changed")
    training = all_images[train_rows]
    real = all_images[validation_rows[:64]]
    generated = torch.from_numpy(
        generated_bytes.copy()
    ).to(device=device, dtype=torch.float32).permute(0, 3, 1, 2) / 255
    training_bytes = (training * 255).round().byte().permute(
        0, 2, 3, 1
    ).cpu().numpy()
    seen = {hashlib.sha256(row.tobytes()).digest() for row in training_bytes}
    exact_matches = sum(
        hashlib.sha256(row.tobytes()).digest() in seen
        for row in generated_bytes
    )
    unique_generated = len({
        hashlib.sha256(row.tobytes()).digest() for row in generated_bytes
    })
    result = {
        "scope": "64 fixed-noise generated versus 64 fixed validation images and 45k training images; raw normalized-pixel checks only, not semantic or perceptual quality",
        "source_identity": source,
        "generated_array_sha256": final_preview["sha256"],
        "exact_generated_train_byte_matches": exact_matches,
        "unique_generated_byte_arrays": unique_generated,
        "generated_pairwise_pixel_rmse_mean": mean_pairwise_rmse(generated),
        "real_validation_pairwise_pixel_rmse_mean": mean_pairwise_rmse(real),
        "generated_nearest_train_pixel_rmse": nearest_training_rmse(generated, training),
        "real_validation_nearest_train_pixel_rmse": nearest_training_rmse(real, training),
        "limitations": "no exact-byte copies does not rule out altered memorization; raw-pixel distances do not rate semantic realism or class coverage",
    }
    out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n",
                   encoding="utf-8")
    print("exact copies", exact_matches, "unique", unique_generated,
          "generated pair RMSE", result["generated_pairwise_pixel_rmse_mean"])
 
 
if __name__ == "__main__":
    main()
