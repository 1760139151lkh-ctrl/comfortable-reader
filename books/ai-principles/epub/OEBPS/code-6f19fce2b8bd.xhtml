"""同一绿色观测下，按历史推测电量2的机会如何改变最后一步选择。"""
 
from __future__ import annotations
 
import argparse
import json
from fractions import Fraction
from pathlib import Path
 
from delivery_decision_lab import lamp, outcomes
 
 
WORK = Path(__file__).resolve().parents[1]
DEFAULT = WORK / "results" / "c21_green_belief_probe.json"
 
 
def expected_reward(energy: int, action: str) -> Fraction:
    return sum(Fraction(str(probability)) * reward for probability, reward, _ in outcomes(energy, action))
 
 
def score(belief_two: Fraction) -> dict:
    fast = (1 - belief_two) * expected_reward(1, "express") + belief_two * expected_reward(2, "express")
    slow = (1 - belief_two) * expected_reward(1, "slow") + belief_two * expected_reward(2, "slow")
    return {"probability_energy_2_given_green": str(belief_two),
            "fast_expected_last_reward": float(fast), "slow_expected_last_reward": float(slow),
            "better_one_step_action": "express" if fast > slow else "slow" if slow > fast else "tie"}
 
 
def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--belief", help="另给一项电量2机会，例如 9/10 或 0.95")
    parser.add_argument("--out", type=Path, default=DEFAULT)
    args = parser.parse_args()
    output = args.out if args.out.is_absolute() else WORK.parent / args.out
    if output.exists():
        raise FileExistsError(f"结果已存在，请另选输出路径：{output}")
    if lamp(1) != "green" or lamp(2) != "green":
        raise AssertionError("两个电量须显示同一绿色观测")
    fast_at_one, fast_at_two = expected_reward(1, "express"), expected_reward(2, "express")
    slow_at_one, slow_at_two = expected_reward(1, "slow"), expected_reward(2, "slow")
    threshold = (slow_at_one - fast_at_one) / ((fast_at_two - fast_at_one) - (slow_at_two - slow_at_one))
    if threshold != Fraction(12, 13):
        raise AssertionError("当前配送规则的最后一步选择门槛改变")
    points = [Fraction(0), Fraction(1, 2), Fraction(9, 10), threshold, Fraction(19, 20), Fraction(1)]
    if args.belief is not None:
        candidate = Fraction(args.belief)
        if not 0 <= candidate <= 1:
            parser.error("--belief 须在 0 到 1 之间")
        points.append(candidate)
    report = {
        "scope": "只在最后一次行动、当前灯为绿且电量可能为1或2时比较两行动期望；不是多轮最优策略或从未知环境学习。",
        "full_state_one_step_expectations": {
            "energy_1_fast": float(fast_at_one), "energy_2_fast": float(fast_at_two),
            "energy_1_slow": float(slow_at_one), "energy_2_slow": float(slow_at_two),
        },
        "after_known_charge_probability_energy_2": 1,
        "after_known_slow_from_energy_2_probability_energy_2": 0,
        "fast_minus_slow_formula": "(13/2)*b - 6; b=P(S_t=2 | green and available history)",
        "fast_and_slow_tie_belief": str(threshold),
        "examples": [score(point) for point in points],
    }
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(report, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
