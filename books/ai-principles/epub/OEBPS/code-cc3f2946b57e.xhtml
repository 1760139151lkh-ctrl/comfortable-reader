"""Show reconstructions versus genuine prior-sampled VAE binary Fashion images."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import torch
 
from fashion_ae_vae import AE, VAE, ROOT, DATA, LATENT, SEED, digest
 
 
def selected(mode: str, directory: Path, device: torch.device):
    report = json.loads((directory / "train.json").read_text(encoding="utf-8"))
    path = ROOT / report["checkpoint"]
    if (report["mode"] != mode or digest(path) != report["checkpoint_sha256"]):
        raise RuntimeError(f"{mode} checkpoint identity changed")
    state = torch.load(path, map_location="cpu", weights_only=True)
    if state["mode"] != mode or state["source_identity"] != report["source_identity"]:
        raise RuntimeError(f"{mode} checkpoint metadata changed")
    model = (AE() if mode == "ae" else VAE()).to(device)
    model.load_state_dict(state["model"])
    model.eval()
    return model, report
 
 
@torch.no_grad()
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--ae-dir", type=Path, default=Path("work/runs/c30_fashion_ae"))
    parser.add_argument("--vae-dir", type=Path, default=Path("work/runs/c30_fashion_vae"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/figures/c30_ae_vae_reconstruct_and_sample.png"))
    parser.add_argument("--receipt", type=Path,
                        default=Path("work/results/c30_ae_vae_visual_check.json"))
    args = parser.parse_args()
    out, receipt = ROOT / args.out, ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing AE/VAE figure")
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    ae, ae_report = selected("ae", ROOT / args.ae_dir, device)
    vae, vae_report = selected("vae", ROOT / args.vae_dir, device)
    if ae_report["source_identity"] != vae_report["source_identity"]:
        raise RuntimeError("AE/VAE different derived data")
    val = np.load(DATA / "validation.npy", allow_pickle=False)[:12]
    images = torch.from_numpy(val[:, None].copy()).to(
        device=device, dtype=torch.float32
    )
    ae_logits, _ = ae(images)
    vae_logits, _, _ = vae(images, use_mean=True)
    torch.manual_seed(SEED + 7000)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED + 7000)
    prior = torch.randn(12, LATENT, device=device)
    prior_prob = vae.decoder(prior).sigmoid()
    prior_binary = torch.bernoulli(prior_prob)
    rows = [
        images, ae_logits.sigmoid(), vae_logits.sigmoid(),
        prior_prob, prior_binary,
    ]
    titles = [
        "原二值图", "AE重建概率", "VAE后验均值重建概率",
        "新抽先验z的像素概率", "再逐像素抽样所得新图",
    ]
    fig, axes = plt.subplots(5, 12, figsize=(18, 8))
    for row, pictures in enumerate(rows):
        for col in range(12):
            axes[row, col].imshow(
                pictures[col, 0].cpu().numpy(),
                cmap="gray", vmin=0, vmax=1, interpolation="nearest",
            )
            axes[row, col].axis("off")
        fig.text(0.012, 1 - (row + 0.5) / 5, titles[row],
                 va="center", fontsize=10, fontproperties="Microsoft YaHei")
    fig.tight_layout(rect=(0.12, 0, 1, 1))
    out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=180)
    plt.close(fig)
    result = {
        "scope": "first12 binary Fashion validation images for reconstruction; 12 independent VAE prior z draws, then conditional Bernoulli pixel draws; visual examples, not all-sample score",
        "source_identity": ae_report["source_identity"],
        "ae_checkpoint_sha256": ae_report["checkpoint_sha256"],
        "vae_checkpoint_sha256": vae_report["checkpoint_sha256"],
        "prior_seed": SEED + 7000,
        "rows": titles,
        "figure": str(out.relative_to(ROOT)).replace("\\", "/"),
        "labels_loaded": False,
    }
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n",
                       encoding="utf-8")
 
 
if __name__ == "__main__":
    main()
