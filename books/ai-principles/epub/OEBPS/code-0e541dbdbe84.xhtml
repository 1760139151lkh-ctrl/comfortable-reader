"""用真实红裤子句的编号画出三种读取各自允许的列；颜色不是学得权重。"""
 
from __future__ import annotations
 
import hashlib
import json
from pathlib import Path
 
import matplotlib
 
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.colors import ListedColormap
from matplotlib.patches import Patch
import numpy as np
import torch
 
from transformer_bridge import PAD, causal_mask, source_mask
 
 
WORK = Path(__file__).resolve().parents[1]
DATA = WORK / "data" / "tatoeba_fra_eng"
RESULT = WORK / "results" / "attention_mask_figure.json"
FIGURE = WORK / "figures" / "transformer_three_visibility_masks.png"
SOURCE_ROW = 26766
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def main() -> None:
    manifest_path = DATA / "manifest.json"
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    if manifest["alignment_visualization_validation_source_row"] != SOURCE_ROW:
        raise ValueError("事前定的机制句来源行号已改变")
    expected = {
        DATA / "validation_pairs.npz": manifest["files"]["validation"]["arrays_sha256"],
        DATA / "validation_attribution.jsonl": manifest["files"]["validation"]["attribution_sha256"],
        DATA / "english_vocab.json": manifest["source_vocab_sha256"],
        DATA / "french_vocab.json": manifest["target_vocab_sha256"],
    }
    for path, digest in expected.items():
        if sha256(path) != digest:
            raise ValueError(f"画图输入与C16来源清单不符：{path.name}")
    with (DATA / "validation_attribution.jsonl").open(encoding="utf-8") as stream:
        metadata = [json.loads(line) for line in stream]
    hits = [i for i, item in enumerate(metadata) if item["source_row_zero_based"] == SOURCE_ROW]
    if len(hits) != 1:
        raise ValueError("事前固定红裤子句须恰有一条验证记录")
    row = hits[0]
    source_vocab = json.loads((DATA / "english_vocab.json").read_text(encoding="utf-8"))
    target_vocab = json.loads((DATA / "french_vocab.json").read_text(encoding="utf-8"))
    with np.load(DATA / "validation_pairs.npz") as data:
        source_length = int(data["source_len"][row])
        target_length = int(data["target_len"][row])
        source_ids = data["source_ids"][row, :source_length + 2].astype(np.int64)
        target_ids = data["target_input"][row, :target_length + 2].astype(np.int64)
    source = torch.from_numpy(source_ids).unsqueeze(0)
    target = torch.from_numpy(target_ids).unsqueeze(0)
    source_labels = [source_vocab[int(i)] for i in source_ids]
    target_labels = [target_vocab[int(i)] for i in target_ids]
    masks = {
        "source_self": source_mask(source, source.shape[1])[0, 0, :source_length].numpy(),
        "target_causal_self": causal_mask(target)[0, 0, :target_length].numpy(),
        "target_to_source": source_mask(source, target.shape[1])[0, 0, :target_length].numpy(),
    }
    if any(mask.dtype != bool for mask in masks.values()):
        raise AssertionError("可见表必须是布尔表")
    if masks["target_causal_self"][:, target_length:].any() or masks["target_to_source"][:, source_length:].any():
        raise AssertionError("填充列不应可读")
    for i in range(target_length):
        if masks["target_causal_self"][i, i + 1:].any():
            raise AssertionError("译文未来列不应可读")
 
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    fig, axes = plt.subplots(1, 3, figsize=(14, 5.1), layout="constrained")
    panels = [
        ("source_self", "英语来源内部", source_labels, source_labels[:source_length], "英语可读列", "英语查询"),
        ("target_causal_self", "法语译文内部", target_labels, target_labels[:target_length], "法语可读列", "法语查询"),
        ("target_to_source", "法语读取英语", source_labels, target_labels[:target_length], "英语可读列", "法语查询"),
    ]
    colors = ListedColormap(["#e6e9ed", "#3978ae"])
    for ax, (key, title, columns, rows, xlabel, ylabel) in zip(axes, panels):
        matrix = masks[key]
        ax.imshow(matrix.astype(int), cmap=colors, vmin=0, vmax=1, interpolation="nearest")
        ax.set(xticks=range(len(columns)), yticks=range(len(rows)), xticklabels=columns, yticklabels=rows,
               xlabel=xlabel, ylabel=ylabel, title=title)
        ax.tick_params(axis="x", labelrotation=45)
        ax.set_xticks(np.arange(-0.5, len(columns), 1), minor=True)
        ax.set_yticks(np.arange(-0.5, len(rows), 1), minor=True)
        ax.grid(which="minor", color="white", linewidth=1)
        ax.tick_params(which="minor", bottom=False, left=False)
    fig.legend(handles=[Patch(color="#3978ae", label="允许读取"), Patch(color="#e6e9ed", label="禁止读取")],
               loc="upper center", ncol=2, bbox_to_anchor=(0.5, 1.04))
    FIGURE.parent.mkdir(exist_ok=True)
    fig.savefig(FIGURE, dpi=180, bbox_inches="tight")
    plt.close(fig)
    report = {
        "source_manifest_sha256": sha256(manifest_path),
        "source_row_zero_based": SOURCE_ROW,
        "source": metadata[row]["english_original"],
        "target": metadata[row]["french_original"],
        "attribution": metadata[row]["attribution"],
        "source_tokens_including_two_padding_columns": source_labels,
        "decoder_input_including_two_padding_columns": target_labels,
        "valid_source_positions": source_length,
        "valid_target_input_positions": target_length,
        "masks_valid_query_rows": {name: mask.astype(int).tolist() for name, mask in masks.items()},
        "meaning": "蓝色是结构允许读取的键列，不是训练得出的注意力大小；真实查询行以外的PAD行未画。",
    }
    RESULT.parent.mkdir(exist_ok=True)
    RESULT.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"figure": str(FIGURE), "source": report["source"], "mask_shapes":
                      {name: list(mask.shape) for name, mask in masks.items()}}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
