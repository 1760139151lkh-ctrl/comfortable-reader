"""Visualize authentic and step-by-step autoregressive binary Fashion images."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
from fashion_pixelcnn import ROOT, DATA, digest
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c30_fashion_pixelcnn"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/figures/c30_pixelcnn_real_and_generated.png"))
    parser.add_argument("--receipt", type=Path,
                        default=Path("work/results/c30_pixelcnn_visual_check.json"))
    args = parser.parse_args()
    run = ROOT / args.run_dir
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    generated_path = ROOT / report["samples"]
    if digest(generated_path) != report["samples_sha256"]:
        raise RuntimeError("PixelCNN generated array changed")
    actual = np.load(DATA / "validation.npy", allow_pickle=False)[:12]
    generated = np.load(generated_path, allow_pickle=False)[:12]
    out, receipt = ROOT / args.out, ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing PixelCNN figure")
    fig, axes = plt.subplots(2, 12, figsize=(18, 4))
    for row, batch in enumerate((actual, generated)):
        for col in range(12):
            axes[row, col].imshow(
                batch[col], cmap="gray", vmin=0, vmax=1,
                interpolation="nearest",
            )
            axes[row, col].axis("off")
    fig.text(0.01, 0.75, "真实验证图", va="center",
             fontproperties="Microsoft YaHei", fontsize=10)
    fig.text(0.01, 0.25, "逐像素新样本", va="center",
             fontproperties="Microsoft YaHei", fontsize=10)
    fig.tight_layout(rect=(0.11, 0, 1, 1))
    out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=180)
    plt.close(fig)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps({
        "scope": "first12 real binary Fashion validation images and first12 of24 PixelCNN samples from 784 sequential draws; visual only",
        "checkpoint_sha256": report["checkpoint_sha256"],
        "generated_sha256": report["samples_sha256"],
        "figure": str(out.relative_to(ROOT)).replace("\\", "/"),
        "labels_loaded": False,
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
 
 
if __name__ == "__main__":
    main()
