"""第二章学习版本：同一张表、另一把误差尺，以及完整批次更新。"""
from __future__ import annotations
import json
from pathlib import Path
 
HERE = Path(__file__).resolve().parent
FIRST_MODEL = HERE / "output" / "first_perceptron_model.json"
if not FIRST_MODEL.is_file():
    raise SystemExit("先运行第一章的 01_first_perceptron.py，再运行本文件。")
 
records = [
    ((2.0, 2.0), 1.0),
    ((1.0, 3.0), 1.0),
    ((3.0, 1.0), 1.0),
    ((0.0, 0.0), -1.0),
    ((1.0, 1.0), -1.0),
    ((2.0, 0.0), -1.0),
    ((0.0, 2.0), -1.0),
]
 
 
def score(theta, features):
    return theta[0] * features[0] + theta[1] * features[1] + theta[2]
 
 
def mean_half_square(theta, examples):
    return sum(0.5 * (score(theta, x) - y) ** 2 for x, y in examples) / len(examples)
 
 
def mean_square_gradient(theta, examples):
    gradient = [0.0, 0.0, 0.0]
    for features, target in examples:
        residual = score(theta, features) - target
        gradient[0] += residual * features[0]
        gradient[1] += residual * features[1]
        gradient[2] += residual
    return [component / len(examples) for component in gradient]
 
 
model = json.loads(FIRST_MODEL.read_text("utf-8"))
theta = [*model["weights"], model["bias"]]
initial = mean_half_square(theta, records)
first_gradient = mean_square_gradient(theta, records)
rate = 0.1
theta = [theta[j] - rate * first_gradient[j] for j in range(3)]
after_first = mean_half_square(theta, records)
print("初始损失:", initial)
print("第一步平均梯度:", first_gradient)
print("一步后参数和损失:", theta, after_first)
for _ in range(2999):
    gradient = mean_square_gradient(theta, records)
    theta = [theta[j] - rate * gradient[j] for j in range(3)]
final = mean_half_square(theta, records)
print("3000 步后参数和损失:", theta, final)
output = HERE / "output" / "second_squared_loss.json"
output.write_text(json.dumps({
    "source_model": str(FIRST_MODEL.name),
    "rate": rate,
    "initial_loss": initial,
    "first_gradient": first_gradient,
    "loss_after_first_step": after_first,
    "steps": 3000,
    "theta": theta,
    "final_loss": final,
}, ensure_ascii=False, indent=2), "utf-8")
print("独立练习记录:", output)
