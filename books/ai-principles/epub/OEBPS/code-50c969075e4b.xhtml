"""Show genuine validation past/current/future versus copy and learned prediction."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import torch
 
from c33_future_frame import FutureNet, batch
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/penn_action_publication.xhtml"
 
 
def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path, required=True)
    parser.add_argument("--out", type=Path, required=True)
    parser.add_argument("--receipt", type=Path, required=True)
    args = parser.parse_args()
    run = args.run_dir if args.run_dir.is_absolute() else ROOT / args.run_dir
    out = args.out if args.out.is_absolute() else ROOT / args.out
    receipt = args.receipt if args.receipt.is_absolute() else ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing next-frame figure/receipt")
    manifest_path = DATA / "manifest.json"
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    train_report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    checkpoint_path = run / "best.pt"
    if (train_report["data_manifest_sha256"] != sha256(manifest_path) or
            train_report["best_checkpoint_sha256"] != sha256(checkpoint_path)):
        raise RuntimeError("frozen future model/data mismatch")
    checkpoint = torch.load(checkpoint_path, map_location="cpu", weights_only=False)
    if checkpoint["data_manifest_sha256"] != sha256(manifest_path):
        raise RuntimeError("future checkpoint source mismatch")
    validation = np.load(DATA / "validation_triplets.npy", mmap_mode="r", allow_pickle=False)
    # The first validation entry is fixed by source-id order, not cherry-picked
    # after viewing model outputs.
    selected = 0
    row = manifest["metadata"]["validation"][selected]
    torch.set_num_threads(4)
    model = FutureNet()
    model.load_state_dict(checkpoint["model"])
    model.eval()
    with torch.no_grad():
        past, current, future = batch(validation, np.array([selected]), torch.device("cpu"))
        predicted = model(past, current)
    first = past[0].permute(1, 2, 0).numpy()
    now = current[0].permute(1, 2, 0).numpy()
    truth = future[0].permute(1, 2, 0).numpy()
    guess = predicted[0].permute(1, 2, 0).numpy()
    diff = np.abs(guess - truth).mean(axis=2)
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    fig, axes = plt.subplots(2, 3, figsize=(13, 6), layout="constrained")
    panels = [
        (axes[0, 0], first, "较早真帧"), (axes[0, 1], now, "当前真帧"),
        (axes[0, 2], truth, "隔5帧后的真实画面"),
        (axes[1, 0], now, "复制当前：零训练基线"),
        (axes[1, 1], guess, "网络根据前两帧预测"),
        (axes[1, 2], diff, "预测与真实的平均颜色差"),
    ]
    for axis, pixels, title in panels:
        if pixels.ndim == 2:
            axis.imshow(pixels, cmap="magma", vmin=0, vmax=0.5)
        else:
            axis.imshow(np.clip(pixels, 0, 1))
        axis.set_title(title)
        axis.axis("off")
    fig.suptitle(f"Penn官方训练组内另留的视频 {row['video_id']}；只按帧序号，不冒称秒数",
                 fontsize=14)
    out.parent.mkdir(parents=True, exist_ok=True)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=160)
    plt.close(fig)
    mask = np.abs(now - truth).mean(axis=2) > 0.08
    result = {
        "source_archive_sha256": manifest["source_archive_sha256"],
        "data_manifest_sha256": sha256(manifest_path),
        "train_report_sha256": sha256(run / "train.json"),
        "checkpoint_sha256": sha256(checkpoint_path),
        "validation_metadata": row,
        "selected_validation_index": selected,
        "copy_mse_per_rgb_coordinate": float(np.mean((now - truth) ** 2)),
        "model_mse_per_rgb_coordinate": float(np.mean((guess - truth) ** 2)),
        "future_defined_moving_mask_fraction": float(mask.mean()),
        "copy_mse_on_future_defined_moving_mask": float(np.mean((now[mask] - truth[mask]) ** 2)) if mask.any() else None,
        "model_mse_on_future_defined_moving_mask": float(np.mean((guess[mask] - truth[mask]) ** 2)) if mask.any() else None,
        "figure_sha256": sha256(out),
        "limits": "single fixed validation example, low resolution letterbox, one-frame deterministic prediction; mask uses future only for analysis",
    }
    receipt.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"video": row["video_id"],
                      "copy_mse": result["copy_mse_per_rgb_coordinate"],
                      "model_mse": result["model_mse_per_rgb_coordinate"]}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
