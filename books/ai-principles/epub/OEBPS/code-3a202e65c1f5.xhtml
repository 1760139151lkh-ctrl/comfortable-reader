"""Plot real Fashion ankle boots beside fixed-noise diffusion guidance variants."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
from fashion_ddpm import ROOT, DATA, digest
from prepare_fashion_mnist import load_images, load_labels
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c31_fashion_ddpm"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/figures/c31_fashion_guidance.png"))
    parser.add_argument("--receipt", type=Path,
                        default=Path("work/results/c31_fashion_guidance_figure.json"))
    args = parser.parse_args()
    run, out, receipt = ROOT / args.run_dir, ROOT / args.out, ROOT / args.receipt
    if out.exists() or receipt.exists():
        raise FileExistsError("preserve existing DDPM figure")
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    if digest(ROOT / report["checkpoint"]) != report["checkpoint_sha256"]:
        raise RuntimeError("DDPM checkpoint changed")
    image_path = DATA / "train-images-idx3-ubyte.gz"
    label_path = DATA / "train-labels-idx1-ubyte.gz"
    if (digest(image_path) != report["source_identity"]["train_images_sha256"] or
            digest(label_path) != report["source_identity"]["train_labels_sha256"]):
        raise RuntimeError("Fashion source changed")
    split = np.load(DATA / "chapter_split.npz")
    validation = split["validation_indices"]
    classes = load_labels(label_path, 60000)
    selected = validation[classes[validation] == 9][:8]
    original = load_images(image_path, 60000)[selected]
    rows = [("真实验证踝靴", original)]
    descriptions = {
        "unconditional": "不提供类别",
        "ankle_boot_w0": "类别9，w=0",
        "ankle_boot_w2": "类别9，w=2",
        "ankle_boot_w5": "类别9，w=5",
    }
    for item in report["samples"]:
        path = ROOT / item["path"]
        if digest(path) != item["sha256"]:
            raise RuntimeError(f"DDPM generated array changed: {item['name']}")
        rows.append((descriptions[item["name"]],
                     np.load(path, allow_pickle=False)[:8]))
    fig, axes = plt.subplots(len(rows), 8, figsize=(13, 1.75 * len(rows)))
    for row, (title, pictures) in enumerate(rows):
        for col in range(8):
            axes[row, col].imshow(
                pictures[col], cmap="gray", vmin=0, vmax=255,
                interpolation="nearest",
            )
            axes[row, col].axis("off")
        fig.text(0.01, 1 - (row + 0.5) / len(rows), title,
                 va="center", fontsize=10, fontproperties="Microsoft YaHei")
    fig.tight_layout(rect=(0.12, 0, 1, 1))
    out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(out, dpi=180)
    plt.close(fig)
    receipt.parent.mkdir(parents=True, exist_ok=True)
    receipt.write_text(json.dumps({
        "scope": "eight old Fashion validation class9 ankle boots versus same fixed reverse-noise sequence with null/conditional class9 guidance w0,w2,w5; visual only",
        "source_identity": report["source_identity"],
        "checkpoint_sha256": report["checkpoint_sha256"],
        "real_validation_source_rows": selected.astype(int).tolist(),
        "generated_sample_sha256": {
            item["name"]: item["sha256"] for item in report["samples"]
        },
        "figure": str(out.relative_to(ROOT)).replace("\\", "/"),
        "official_test_opened": False,
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
 
 
if __name__ == "__main__":
    main()
