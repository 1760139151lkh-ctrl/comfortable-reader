"""Finite two-state check of the ELBO identity before a continuous-image VAE."""
 
from __future__ import annotations
 
import argparse
import json
import math
from pathlib import Path
 
 
ROOT = Path(__file__).resolve().parents[2]
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c30_elbo_toy.json"))
    args = parser.parse_args()
    out = ROOT / args.out
    if out.exists():
        raise FileExistsError(f"preserve existing toy result: {out}")
    prior = [0.5, 0.5]
    likelihood_of_x1 = [0.1, 0.9]
    q = [0.3, 0.7]  # Intentionally unlike the actual posterior [0.1, 0.9].
    joint = [prior[i] * likelihood_of_x1[i] for i in range(2)]
    marginal = sum(joint)
    posterior = [x / marginal for x in joint]
    elbo = sum(q[i] * math.log(joint[i] / q[i]) for i in range(2))
    gap = sum(q[i] * math.log(q[i] / posterior[i]) for i in range(2))
    difference = math.log(marginal) - elbo - gap
    if abs(difference) > 1e-14:
        raise RuntimeError("ELBO/KL identity failed")
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps({
        "scope": "two-state exact probability check, not learned image generation",
        "observed_x": 1,
        "prior_z0_z1": prior,
        "likelihood_x1_given_z0_z1": likelihood_of_x1,
        "joint_x1_z0_z1": joint,
        "marginal_p_x1": marginal,
        "posterior_z0_z1_given_x1": posterior,
        "chosen_approximate_q": q,
        "log_marginal": math.log(marginal),
        "elbo": elbo,
        "kl_q_to_true_posterior": gap,
        "identity_difference": difference,
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("log p", math.log(marginal), "ELBO", elbo, "gap", gap)
 
 
if __name__ == "__main__":
    main()
