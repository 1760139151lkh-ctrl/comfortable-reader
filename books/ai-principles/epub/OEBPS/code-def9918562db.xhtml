"""Verify and keep three exact files from the UCI Adult archive locally."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
from urllib.request import Request, urlopen
from zipfile import ZipFile
 
WORK = Path(__file__).resolve().parents[1]
OUT = WORK / "data/c44_adult"
URL = "https://archive.ics.uci.edu/static/public/2/adult.zip"
ARCHIVE_SHA = "7537312dd56c2b98035880805ce99e68183a30ee468aa5329d6df0fbb3cc21bb"
ARCHIVE_BYTES = 620237
FILES = {
    "adult.data": (3974305, "5b00264637dbfec36bdeaab5676b0b309ff9eb788d63554ca0a249491c86603d"),
    "adult.test": (2003153, "a2a9044bc167a35b2361efbabec64e89d69ce82d9790d2980119aac5fd7e9c05"),
    "adult.names": (5229, "c248284c0b5de30c9e1958d6cdd168a34a654758b620e68f46aefa83fc0a576a"),
}
 
 
def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()
 
 
def main():
    OUT.mkdir(parents=True, exist_ok=True)
    archive = OUT / "adult.zip"
    if archive.exists():
        data = archive.read_bytes()
    else:
        with urlopen(Request(URL, headers={"User-Agent": "AI textbook local research"}), timeout=90) as response:
            data = response.read()
    if len(data) != ARCHIVE_BYTES or sha(data) != ARCHIVE_SHA:
        raise RuntimeError("UCI archive differs from pinned bytes")
    if not archive.exists():
        archive.write_bytes(data)
    entries = []
    with ZipFile(archive) as z:
        if z.testzip() is not None:
            raise RuntimeError("UCI archive CRC failure")
        if set(z.namelist()) != {"Index", "adult.data", "adult.test", "adult.names", "old.adult.names"}:
            raise RuntimeError("unexpected UCI archive members")
        for name, (size, digest) in FILES.items():
            raw = z.read(name)  # exact whitelist; never extract archive paths
            if len(raw) != size or sha(raw) != digest:
                raise RuntimeError(f"UCI member differs: {name}")
            target = OUT / name
            if target.exists() and target.read_bytes() != raw:
                raise RuntimeError(f"existing local file differs: {name}")
            if not target.exists():
                target.write_bytes(raw)
            entries.append({"name": name, "bytes": size, "sha256": digest,
                            "nonempty_lines": sum(bool(line.strip()) for line in raw.splitlines())})
    report = {"source": URL, "uci_page": "https://archive.ics.uci.edu/dataset/2/adult",
              "uci_doi": "10.24432/C5XW20", "archive_bytes": ARCHIVE_BYTES,
              "archive_sha256": ARCHIVE_SHA, "files_kept": entries,
              "other_archive_members": ["Index", "old.adult.names"],
              "official_split": "adult.data 32561; adult.test has one header plus 16281 rows; original names says MLC++ random 2/3-1/3 from 1994 census extraction",
              "license_page_claim": "UCI dataset page states CC BY 4.0 with attribution; legal/privacy suitability of any downstream use not decided by this script",
              "privacy": "Local research; no individual census row is printed or uploaded by this program."}
    manifest = OUT / "manifest.json"
    if manifest.exists():
        if json.loads(manifest.read_text(encoding="utf-8")) != report:
            raise RuntimeError("existing manifest differs")
    else:
        manifest.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"archive_sha256": ARCHIVE_SHA,
                      "files": [(x["name"], x["nonempty_lines"]) for x in entries]}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    argparse.ArgumentParser().parse_args()
    main()
