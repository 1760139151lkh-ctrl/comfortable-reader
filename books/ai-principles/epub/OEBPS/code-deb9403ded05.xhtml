"""Check exact/normalized prompt leakage across current preference splits."""
 
from __future__ import annotations
 
import hashlib
import json
from pathlib import Path
 
 
ROOT = Path(__file__).resolve().parents[2]
BASE = ROOT / "work/data/coval_eec757"
 
 
def audit(path: Path) -> dict:
    seen: dict[str, set[str]] = {}
    rows = 0
    for split in ("train", "validation", "test"):
        for record in json.loads((path / f"{split}.json").read_text(encoding="utf-8")):
            normalized = " ".join(record["prompt"].casefold().split())
            key = hashlib.sha256(normalized.encode("utf-8")).hexdigest()
            seen.setdefault(key, set()).add(split)
            rows += 1
    return {"rows": rows, "normalized_prompt_groups": len(seen),
            "groups_crossing_splits": sum(len(groups) > 1 for groups in seen.values())}
 
 
if __name__ == "__main__":
    print("all_pairs", audit(BASE / "pairs_v1"))
    print("short_dpo", audit(BASE / "short_dpo_v2"))
