"""C25: same real prompt/response input, two different next-token loss regions."""
 
from __future__ import annotations
 
import argparse
import json
import math
from pathlib import Path
 
import numpy as np
import torch
from torch.nn import functional as F
 
from train_dolly_sft import (
    ROOT,
    choose_device,
    load_base,
    load_sft_manifest,
    load_sft_split,
    torch_batch,
    write_json,
)
from train_causal_wikitext import autocast_for
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument(
        "--out", type=Path, default=Path("work/results/c25_loss_region_probe.json")
    )
    args = parser.parse_args()
    out = args.out if args.out.is_absolute() else ROOT / args.out
    if out.exists():
        raise FileExistsError(f"keep previous loss-region result: {out}")
    manifest, manifest_sha = load_sft_manifest()
    train = load_sft_split(manifest, "train")
    model, _, base = load_base()
    device = choose_device()
    model.to(device).train()
    ids, response_labels, real = torch_batch(train, np.asarray([0]), device)
    true_length = int(real[0].sum().item())
    prompt_tokens = int(train["prompt_tokens"][0])
    full_labels = torch.full_like(response_labels, -100)
    full_labels[0, : true_length - 1] = ids[0, 1:true_length]
    full_labels[0, true_length - 1] = 2  # EOS is the final target, not an input.
    if not torch.equal(
        full_labels[0, prompt_tokens:true_length],
        response_labels[0, prompt_tokens:true_length],
    ):
        raise AssertionError("the response part changed when prompt loss was unmasked")
    with autocast_for(device):
        logits = model(ids, real)
    all_count = int((full_labels != -100).sum().item())
    answer_count = int((response_labels != -100).sum().item())
    all_loss = F.cross_entropy(
        logits.float().transpose(1, 2), full_labels,
        ignore_index=-100, reduction="sum",
    ) / all_count
    response_loss = F.cross_entropy(
        logits.float().transpose(1, 2), response_labels,
        ignore_index=-100, reduction="sum",
    ) / answer_count
    model.zero_grad(set_to_none=True)
    all_loss.backward(retain_graph=True)
    full_gradient = model.output.weight.grad.detach().float().clone()
    model.zero_grad(set_to_none=True)
    response_loss.backward()
    answer_gradient = model.output.weight.grad.detach().float().clone()
    cosine = float(F.cosine_similarity(
        full_gradient.flatten(), answer_gradient.flatten(), dim=0
    ))
    if not math.isfinite(cosine):
        raise RuntimeError("nonfinite gradient comparison")
    result = {
        "scope": "same C19 base checkpoint, same Dolly training row and forward pass; only loss labels differ; no optimizer step or test data opened",
        "base_checkpoint_sha256": base["c19_checkpoint_sha256"],
        "sft_manifest_sha256": manifest_sha,
        "source_row": int(train["source_row"][0]),
        "visible_input_positions_in_both": true_length,
        "prompt_targets_added_only_to_full_text_objective": prompt_tokens,
        "response_plus_eos_targets_in_both": answer_count,
        "full_text_scored_targets": all_count,
        "response_only_nll": float(response_loss.detach()),
        "full_text_nll": float(all_loss.detach()),
        "output_weight_gradient_norm_full_text": float(full_gradient.norm()),
        "output_weight_gradient_norm_response_only": float(answer_gradient.norm()),
        "output_weight_gradient_cosine": cosine,
        "optimizer_step_executed": False,
    }
    write_json(out, result)
    print(
        "same visible", true_length, "full targets", all_count,
        "response-only targets", answer_count,
    )
    print(
        "loss full/response", result["full_text_nll"], result["response_only_nll"],
        "output gradient cosine", cosine,
    )
 
 
if __name__ == "__main__":
    main()
