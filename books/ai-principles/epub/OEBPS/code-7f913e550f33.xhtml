"""Fixed-step DDIM image changes and an external C13 class9 probe."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import numpy as np
import torch
 
from c31_guidance_diagnostic import judge
from fashion_ddpm import ROOT, digest
from fashion_spatial_training import CNN
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--ddim-run", type=Path,
                        default=Path("work/runs/c31_fashion_ddim"))
    parser.add_argument("--out", type=Path,
                        default=Path("work/results/c31_fashion_ddim_diagnostic.json"))
    args = parser.parse_args()
    run, out = ROOT / args.ddim_run, ROOT / args.out
    if out.exists():
        raise FileExistsError("preserve existing DDIM diagnostic")
    report = json.loads((run / "sample.json").read_text(encoding="utf-8"))
    c13_report = json.loads(
        (ROOT / "work/results/fashion_spatial_training.json").read_text(encoding="utf-8")
    )
    c13 = c13_report["arms"]["cnn_original"]
    c13_checkpoint = ROOT / "work" / c13["checkpoint"]
    if digest(c13_checkpoint) != c13["checkpoint_sha256"]:
        raise RuntimeError("C13 supervised classifier changed")
    model = CNN()
    state = torch.load(c13_checkpoint, map_location="cpu", weights_only=True)
    model.load_state_dict(state["state_dict"])
    model.eval()
    outputs = {}
    arrays = {}
    for item in report["results"]:
        path = ROOT / item["path"]
        if digest(path) != item["sha256"]:
            raise RuntimeError("DDIM sample changed")
        arrays[item["steps"]] = np.load(path, allow_pickle=False)
    reference = arrays[300].astype(np.float32) / 255
    for steps, images in arrays.items():
        scaled = images.astype(np.float32) / 255
        result = judge(model, images, torch.device("cpu"))
        result["pixel_rmse_vs_300_step_display_images"] = float(
            np.sqrt(np.mean((scaled - reference) ** 2))
        )
        outputs[str(steps)] = result
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps({
        "scope": "16 fixed initial-noise class9 eta0 DDIM-style samples, C13 classifier probe and pixel differences versus same model's 300-step deterministic display output; no human quality test",
        "ddim_report_sha256": digest(run / "sample.json"),
        "c13_classifier_sha256": c13["checkpoint_sha256"],
        "results": outputs,
        "limitations": "class9 classifier may be fooled by generated artifacts; display-clipped pixel RMSE measures path change, not error to true data distribution",
    }, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    for steps in (10, 30, 100, 300):
        item = outputs[str(steps)]
        print(steps, item["predicted_class9_count"],
              item["pixel_rmse_vs_300_step_display_images"])
 
 
if __name__ == "__main__":
    main()
