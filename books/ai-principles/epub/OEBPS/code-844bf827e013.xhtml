"""Post-hoc stronger image-copy baseline for an already evaluated C34 field."""
 
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import cv2
import matplotlib.pyplot as plt
import numpy as np
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/eth3d_pipes/views_518"
plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "SimHei", "DejaVu Sans"]
plt.rcParams["axes.unicode_minus"] = False
 
 
def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 20), b""):
            h.update(block)
    return h.hexdigest()
 
 
def small_rgb(view: dict, shape: tuple[int, int]) -> np.ndarray:
    path = ROOT / view["small_path"]
    if sha256(path) != view["small_sha256"]:
        raise RuntimeError(f"prepared view changed: {path}")
    bgr = cv2.imdecode(np.fromfile(path, dtype=np.uint8), cv2.IMREAD_COLOR)
    return cv2.cvtColor(cv2.resize(bgr, (shape[1], shape[0]),
                                    interpolation=cv2.INTER_AREA),
                        cv2.COLOR_BGR2RGB).astype(np.float32) / 255.0
 
 
def axis(view: dict) -> np.ndarray:
    rotation = np.asarray(view["rotation_world_to_camera"])
    return rotation.T @ np.array([0.0, 0.0, 1.0])
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path, required=True)
    parser.add_argument("--out-dir", type=Path, required=True)
    args = parser.parse_args()
    run = args.run_dir if args.run_dir.is_absolute() else ROOT / args.run_dir
    out = args.out_dir if args.out_dir.is_absolute() else ROOT / args.out_dir
    if out.exists():
        raise FileExistsError(out)
    source = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    train = json.loads((run / "train.json").read_text(encoding="utf-8"))
    renders = np.load(run / "heldout_renders.npz", allow_pickle=False)
    views = source["views"]
    train_indices = train["train_view_indices"]
    rows = []
    out.mkdir(parents=True)
    fig, axes = plt.subplots(2, 4, figsize=(14, 7), layout="constrained")
    for position, (name, index) in enumerate((("val", train["validation_view_index"]),
                                              ("test", train["test_view_index"]))):
        real = renders[name + "_real"]
        generated = renders[name]
        center = np.asarray(views[index]["center_world"])
        forward = axis(views[index])
        options = []
        for candidate in train_indices:
            distance = float(np.linalg.norm(np.asarray(views[candidate]["center_world"]) - center))
            cosine = float(np.dot(axis(views[candidate]), forward))
            options.append((distance + 2.0 * (1.0 - cosine), candidate, distance, cosine))
        _, chosen, distance, cosine = min(options)
        copied = small_rgb(views[chosen], real.shape[:2])
        copied_mse = float(np.mean((copied - real) ** 2))
        learned_mse = float(np.mean((generated - real) ** 2))
        rows.append({"partition": name, "target_view_index": index,
                     "chosen_train_view_index": chosen,
                     "pose_score": float(min(options)[0]),
                     "center_distance": distance, "view_axis_cosine": cosine,
                     "copy_mse": copied_mse, "trained_grid_mse": learned_mse})
        for ax, array, title in zip(axes[position],
                                    [real, copied, generated, np.abs(generated - real).mean(axis=2)],
                                    ["真实留出视角", "位置+朝向最近的训练原图", "已训练三维场", "三维场绝对颜色误差"]):
            ax.imshow(array, cmap="magma" if array.ndim == 2 else None,
                      vmin=0, vmax=0.35 if array.ndim == 2 else 1)
            ax.set_title(f"{name} {title}")
            ax.set_xticks([])
            ax.set_yticks([])
    fig.suptitle("相机位置近却朝向相反：只按位置复制会挑错训练图", fontsize=15)
    picture = out / "pose_aware_copy_vs_grid.png"
    fig.savefig(picture, dpi=140)
    plt.close(fig)
    receipt = {"identity": "post-hoc diagnostic after C34 original training/test was already inspected; not new held-out evaluation",
               "field_run_report_sha256": sha256(run / "train.json"),
               "precomputed_renders_sha256": sha256(run / "heldout_renders.npz"),
               "selection_formula": "Euclidean camera-center distance + 2*(1 - cosine between optical axes); uses known camera poses, no target pixels in selection; coefficient introduced after inspecting first center-only baseline",
               "rows": rows, "figure_sha256": sha256(picture),
               "limit": "same-pixel image copy without geometric reprojection, not a mature novel-view system; val/test pixels were already opened by first radiance-grid run"}
    (out / "receipt.json").write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(rows, ensure_ascii=False), flush=True)
 
 
if __name__ == "__main__":
    main()
