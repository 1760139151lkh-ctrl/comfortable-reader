"""ManyThings/Tatoeba 英法句对：保留署名、按英语原句分组切分并建词表。"""
 
from __future__ import annotations
 
import collections
import hashlib
import json
import re
import unicodedata
import urllib.request
import zipfile
from pathlib import Path
 
import numpy as np
 
 
WORK = Path(__file__).resolve().parents[1]
DATA = WORK / "data" / "tatoeba_fra_eng"
SOURCE_ZIP = DATA / "fra-eng.zip"
URL = "https://www.manythings.org/anki/fra-eng.zip"
EXPECTED_SHA256 = "9e35994767f7307dc1fa9d5efcc36681f3756a1bdaaa02c1381b3eb11461261e"
TOKEN_PATTERN = re.compile(r"\w+|[^\w\s]", flags=re.UNICODE)
SPECIAL = ("<pad>", "<unk>", "<bos>", "<eos>")
SOURCE_VOCAB_LIMIT = 8_000
TARGET_VOCAB_LIMIT = 8_000
MIN_TOKENS = 2
MAX_TOKENS = 18
ALIGNMENT_ROW = 26766
 
 
def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def tokens(text: str) -> list[str]:
    normalized = unicodedata.normalize("NFC", text).lower().strip()
    return TOKEN_PATTERN.findall(normalized)
 
 
def get_zip() -> None:
    DATA.mkdir(parents=True, exist_ok=True)
    if not SOURCE_ZIP.exists():
        request = urllib.request.Request(
            URL,
            headers={"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/130.0 Safari/537.36",
                     "Accept": "*/*"},
        )
        temporary = DATA / "fra-eng.zip.part"
        with urllib.request.urlopen(request, timeout=120) as response, temporary.open("wb") as output:
            while chunk := response.read(1024 * 1024):
                output.write(chunk)
        temporary.replace(SOURCE_ZIP)
    if sha256(SOURCE_ZIP) != EXPECTED_SHA256:
        raise ValueError("ManyThings 2026-02-13 来源 ZIP 字节身份不符")
 
 
def split_for_source(source_tokens: list[str]) -> str:
    signature = " ".join(source_tokens).encode("utf-8")
    bucket = int.from_bytes(hashlib.sha256(signature).digest()[:8], "big") % 100
    return "train" if bucket < 80 else "validation" if bucket < 90 else "test"
 
 
def load_pairs() -> tuple[dict[str, list[dict]], dict]:
    get_zip()
    with zipfile.ZipFile(SOURCE_ZIP) as archive:
        if set(archive.namelist()) != {"fra.txt", "_about.txt"}:
            raise ValueError("ManyThings ZIP 内成员不符")
        raw = archive.read("fra.txt")
        about = archive.read("_about.txt").decode("utf-8", errors="replace")
    lines = raw.decode("utf-8-sig").splitlines()
    by_split = {"train": [], "validation": [], "test": []}
    seen_pair = set()
    filtered_length, invalid_columns, exact_duplicates = 0, 0, 0
    for original_row, line in enumerate(lines):
        columns = line.split("\t", 2)
        if len(columns) != 3:
            invalid_columns += 1
            continue
        english, french, attribution = columns
        source = tokens(english)
        target = tokens(french)
        if not (MIN_TOKENS <= len(source) <= MAX_TOKENS and MIN_TOKENS <= len(target) <= MAX_TOKENS):
            filtered_length += 1
            continue
        key = (tuple(source), tuple(target))
        if key in seen_pair:
            exact_duplicates += 1
            continue
        seen_pair.add(key)
        split = split_for_source(source)
        by_split[split].append({
            "source_row_zero_based": original_row,
            "english_original": english,
            "french_original": french,
            "attribution": attribution,
            "source_tokens": source,
            "target_tokens": target,
            "source_group_sha256": hashlib.sha256(" ".join(source).encode("utf-8")).hexdigest(),
        })
    return by_split, {
        "zip_sha256": sha256(SOURCE_ZIP),
        "fra_txt_sha256": hashlib.sha256(raw).hexdigest(),
        "about_sha256": hashlib.sha256(about.encode("utf-8")).hexdigest(),
        "raw_line_count": len(lines),
        "invalid_column_rows": invalid_columns,
        "filtered_by_length_rows": filtered_length,
        "exact_token_pair_duplicates_removed": exact_duplicates,
        "about_excerpt": about[:350],
    }
 
 
def build_vocab(pairs: list[dict], side: str, limit: int) -> list[str]:
    counts = collections.Counter(
        word for item in pairs for word in item[side + "_tokens"]
    )
    return list(SPECIAL) + sorted(counts, key=lambda word: (-counts[word], word))[:limit]
 
 
def make_arrays(pairs: list[dict], src_map: dict, tgt_map: dict) -> tuple[dict, dict]:
    size = len(pairs)
    source_ids = np.zeros((size, MAX_TOKENS + 1), dtype=np.int32)
    target_input = np.zeros((size, MAX_TOKENS + 1), dtype=np.int32)
    target_output = np.zeros((size, MAX_TOKENS + 1), dtype=np.int32)
    source_len = np.zeros(size, dtype=np.int16)
    target_len = np.zeros(size, dtype=np.int16)
    unknown_source, unknown_target = 0, 0
    total_source, total_target = 0, 0
    for row, pair in enumerate(pairs):
        src = [src_map.get(word, 1) for word in pair["source_tokens"]] + [src_map["<eos>"]]
        tgt = [tgt_map.get(word, 1) for word in pair["target_tokens"]] + [tgt_map["<eos>"]]
        src_in = src[:-1]
        tgt_in = [tgt_map["<bos>"]] + tgt[:-1]
        source_ids[row, :len(src)] = src
        target_input[row, :len(tgt_in)] = tgt_in
        target_output[row, :len(tgt)] = tgt
        source_len[row] = len(src)
        target_len[row] = len(tgt)
        unknown_source += sum(word == 1 for word in src_in)
        unknown_target += sum(word == 1 for word in tgt[:-1])
        total_source += len(src_in)
        total_target += len(tgt) - 1
    if not np.all(source_len >= MIN_TOKENS + 1) or not np.all(target_len >= MIN_TOKENS + 1):
        raise AssertionError("已保留句对长度异常")
    return {
        "source_ids": source_ids,
        "source_len": source_len,
        "target_input": target_input,
        "target_output": target_output,
        "target_len": target_len,
    }, {
        "pairs": size,
        "source_token_count_without_eos": total_source,
        "target_token_count_without_eos": total_target,
        "source_unk_fraction": unknown_source / total_source,
        "target_unk_fraction": unknown_target / total_target,
        "different_source_target_length_fraction": sum(
            len(pair["source_tokens"]) != len(pair["target_tokens"]) for pair in pairs
        ) / size,
        "source_length_bins": {
            "2-5": sum(2 <= len(pair["source_tokens"]) <= 5 for pair in pairs),
            "6-10": sum(6 <= len(pair["source_tokens"]) <= 10 for pair in pairs),
            "11-18": sum(11 <= len(pair["source_tokens"]) <= 18 for pair in pairs),
        },
    }
 
 
def main() -> None:
    by_split, source_identity = load_pairs()
    source_sets = {split: {item["source_group_sha256"] for item in pairs} for split, pairs in by_split.items()}
    for left, right in (("train", "validation"), ("train", "test"), ("validation", "test")):
        if source_sets[left] & source_sets[right]:
            raise AssertionError(f"相同英语句子跨 {left}/{right}")
    french_sets = {
        split: {" ".join(item["target_tokens"]) for item in pairs}
        for split, pairs in by_split.items()
    }
    target_overlap = {
        left + "_vs_" + right: len(french_sets[left] & french_sets[right])
        for left, right in (("train", "validation"), ("train", "test"), ("validation", "test"))
    }
    source_vocab = build_vocab(by_split["train"], "source", SOURCE_VOCAB_LIMIT)
    target_vocab = build_vocab(by_split["train"], "target", TARGET_VOCAB_LIMIT)
    src_map = {word: index for index, word in enumerate(source_vocab)}
    tgt_map = {word: index for index, word in enumerate(target_vocab)}
    source_vocab_file = DATA / "english_vocab.json"
    target_vocab_file = DATA / "french_vocab.json"
    source_vocab_file.write_text(json.dumps(source_vocab, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    target_vocab_file.write_text(json.dumps(target_vocab, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    stats, files = {}, {}
    for split, pairs in by_split.items():
        arrays, split_stats = make_arrays(pairs, src_map, tgt_map)
        array_path = DATA / (split + "_pairs.npz")
        np.savez_compressed(array_path, **arrays)
        meta_path = DATA / (split + "_attribution.jsonl")
        with meta_path.open("w", encoding="utf-8") as output:
            for item in pairs:
                output.write(json.dumps({
                    "source_row_zero_based": item["source_row_zero_based"],
                    "english_original": item["english_original"],
                    "french_original": item["french_original"],
                    "attribution": item["attribution"],
                    "source_group_sha256": item["source_group_sha256"],
                }, ensure_ascii=False) + "\n")
        stats[split] = split_stats
        files[split] = {
            "arrays": str(array_path.relative_to(WORK)).replace("\\", "/"),
            "arrays_sha256": sha256(array_path),
            "attribution": str(meta_path.relative_to(WORK)).replace("\\", "/"),
            "attribution_sha256": sha256(meta_path),
        }
    sample = next(
        item for item in by_split["validation"]
        if 6 <= len(item["source_tokens"]) <= 10
        and len(item["source_tokens"]) != len(item["target_tokens"])
    )
    alignment_candidates = [
        item for item in by_split["validation"] if item["source_row_zero_based"] == ALIGNMENT_ROW
    ]
    if len(alignment_candidates) != 1:
        raise ValueError("事前选定的形容词位置示例未在验证句对里恰好出现一次")
    manifest = {
        "source_page": "https://www.manythings.org/anki/",
        "source_zip_url": URL,
        "source_identity": source_identity,
        "citation": "ManyThings.org Anki sentence pairs selected from Tatoeba; each row preserves CC-BY 2.0 France attribution to sentence contributors",
        "tokenizer_rule": "Unicode NFC, lowercase, regex words or each non-whitespace punctuation; not historical paper tokenization",
        "split_rule": "SHA-256 of normalized English token sequence, first 8 bytes modulo 100: train 0-79, validation 80-89, test 90-99",
        "source_group_overlap_count": 0,
        "exact_target_text_overlap_counts": target_overlap,
        "min_tokens": MIN_TOKENS,
        "max_tokens": MAX_TOKENS,
        "source_vocab_size": len(source_vocab),
        "target_vocab_size": len(target_vocab),
        "source_vocab_sha256": sha256(source_vocab_file),
        "target_vocab_sha256": sha256(target_vocab_file),
        "source_vocab_file": str(source_vocab_file.relative_to(WORK)).replace("\\", "/"),
        "target_vocab_file": str(target_vocab_file.relative_to(WORK)).replace("\\", "/"),
        "special_ids": {word: index for index, word in enumerate(SPECIAL)},
        "stats": stats,
        "files": files,
        "validation_example": sample,
        "alignment_visualization_validation_source_row": ALIGNMENT_ROW,
        "alignment_visualization_validation_example": {
            key: alignment_candidates[0][key]
            for key in ("source_row_zero_based", "english_original", "french_original", "attribution")
        },
        "limits": "Sentence pairs contain multiple valid targets, noisy links and short-phrase bias; no document provenance or exact alignment labels; near-duplicate sentence paraphrases not fully checked",
    }
    manifest_file = DATA / "manifest.json"
    manifest_file.write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({
        "source_identity": source_identity,
        "stats": stats,
        "target_overlap": target_overlap,
        "validation_example": {key: sample[key] for key in ("source_row_zero_based", "english_original", "french_original", "attribution")},
    }, ensure_ascii=False, indent=2))
 
 
if __name__ == "__main__":
    main()
