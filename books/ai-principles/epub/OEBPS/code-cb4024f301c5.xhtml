"""Deterministic eta=0 DDIM-style paths from C31's already trained epsilon net."""
 
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import numpy as np
import torch
 
from fashion_ddpm import (
    ROOT, SEED, TIMESTEPS, NoiseUNet, BAR, digest
)
 
 
@torch.no_grad()
def sample(model, device, steps: int, label: int, count=16):
    schedule = np.linspace(TIMESTEPS, 0, steps + 1, dtype=np.int64)
    if not np.all(schedule[:-1] > schedule[1:]):
        raise RuntimeError("DDIM skip schedule not strictly decreasing")
    model.eval()
    torch.manual_seed(SEED + 9000)
    if device.type == "cuda":
        torch.cuda.manual_seed_all(SEED + 9000)
    current = torch.randn(count, 1, 28, 28, device=device)
    abar = BAR.to(device=device, dtype=torch.float32)
    labels = torch.full((count,), label, device=device, dtype=torch.long)
    for t, s in zip(schedule[:-1], schedule[1:]):
        timestep = torch.full((count,), int(t), device=device, dtype=torch.long)
        epsilon = model(current, timestep, labels)
        x0_prediction = (
            current - torch.sqrt(1 - abar[t]) * epsilon
        ) / torch.sqrt(abar[t])
        current = (
            torch.sqrt(abar[s]) * x0_prediction +
            torch.sqrt(1 - abar[s]) * epsilon
        )
    raw_min_max = [float(current.min()), float(current.max())]
    image = ((current.clamp(-1, 1) + 1) * 127.5).round().byte()
    return image[:, 0].cpu().numpy(), raw_min_max, schedule.tolist()
 
 
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path,
                        default=Path("work/runs/c31_fashion_ddpm"))
    parser.add_argument("--out-dir", type=Path,
                        default=Path("work/runs/c31_fashion_ddim"))
    args = parser.parse_args()
    run, out = ROOT / args.run_dir, ROOT / args.out_dir
    if out.exists() and any(out.iterdir()):
        raise FileExistsError("preserve existing DDIM run")
    out.mkdir(parents=True, exist_ok=True)
    parent = json.loads((run / "train.json").read_text(encoding="utf-8"))
    checkpoint_path = ROOT / parent["checkpoint"]
    if digest(checkpoint_path) != parent["checkpoint_sha256"]:
        raise RuntimeError("DDPM selected checkpoint changed")
    checkpoint = torch.load(checkpoint_path, map_location="cpu",
                            weights_only=True)
    if checkpoint["source_identity"] != parent["source_identity"]:
        raise RuntimeError("DDPM source identity changed")
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = NoiseUNet().to(device)
    model.load_state_dict(checkpoint["model"])
    records = []
    for steps in (10, 30, 100, 300):
        images, raw_range, schedule = sample(model, device, steps, 9)
        path = out / f"ankle_boot_ddim_{steps}.npy"
        np.save(path, images, allow_pickle=False)
        records.append({
            "steps": steps, "model_evaluations": steps,
            "path": str(path.relative_to(ROOT)).replace("\\", "/"),
            "sha256": digest(path),
            "raw_before_display_clip_min_max": raw_range,
            "schedule": schedule,
        })
        print("DDIM eta0", steps, "raw", raw_range, flush=True)
    report = {
        "scope": "no retraining; one selected Fashion class9 epsilon model, same initial Gaussian for each deterministic skip schedule; C13 official test not used",
        "parent_training_report": str((run / "train.json").relative_to(ROOT)).replace("\\", "/"),
        "parent_training_report_sha256": digest(run / "train.json"),
        "checkpoint_sha256": parent["checkpoint_sha256"],
        "schedule_bar_alpha_terminal": parent["bar_alpha_terminal"],
        "formula": "x0_hat=(x_t-sqrt(1-bar_t)*eps_theta)/sqrt(bar_t); x_s=sqrt(bar_s)*x0_hat+sqrt(1-bar_s)*eps_theta, s<t, eta=0",
        "note": "DDIM-style deterministic path with original discrete-trained model; skipping steps changes trajectory and is not proven equal to 300-step ancestral DDPM or exact probability flow ODE",
        "label": 9,
        "initial_noise_seed": SEED + 9000,
        "results": records,
    }
    (out / "sample.json").write_text(
        json.dumps(report, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )
 
 
if __name__ == "__main__":
    main()
