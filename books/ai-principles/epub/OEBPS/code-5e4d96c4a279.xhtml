"""A citation graph is unchanged by renaming every node consistently."""
from __future__ import annotations
 
import argparse
import json
from pathlib import Path
 
import numpy as np
import torch
 
from c36_cora_node_models import DATA, NodeNet, normalized_adjacency
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--checkpoint", type=Path, default=Path("work/runs/c36_cora_gcn250/best.pt"))
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("result exists")
    torch.set_num_threads(8)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    x = torch.from_numpy(np.load(DATA / "features_uint8.npy").astype(np.float32)).to(device)
    x = x / x.sum(dim=1, keepdim=True).clamp_min(1)
    edges = np.load(DATA / "undirected_unique_edges_int64.npy")
    adj = normalized_adjacency(edges, len(x), device)
    ckpt = torch.load(a.checkpoint, map_location="cpu", weights_only=True)
    model = NodeNet("gcn").to(device)
    model.load_state_dict(ckpt["model"])
    model.eval()
    rng = np.random.default_rng(20260925)
    perm = rng.permutation(len(x))  # new row i contains original row perm[i]
    inverse = np.empty_like(perm)
    inverse[perm] = np.arange(len(perm))
    shuffled_edges = inverse[edges]
    adj_consistent = normalized_adjacency(shuffled_edges, len(x), device)
    order = torch.from_numpy(perm).long().to(device)
    with torch.inference_mode():
        old = model(x, adj)
        new = model(x[order], adj_consistent)
        wrong = model(x[order], adj)
    max_equivariant_error = float((new - old[order]).abs().max())
    mean_readout_error = float((new.mean(dim=0) - old.mean(dim=0)).abs().max())
    inconsistent_error = float((wrong - old[order]).abs().mean())
    if max_equivariant_error > 1e-4:
        raise RuntimeError("consistent renaming did not commute with GCN")
    result = {"nodes": len(x), "real_citation_pairs": len(edges), "trained_gcn_selected_epoch": ckpt["epoch"], "consistent_node_renaming_max_logit_difference": max_equivariant_error, "graph_mean_readout_invariance_max_difference": mean_readout_error, "inconsistent_reorder_features_only_mean_logit_difference": inconsistent_error, "meaning": "PAP^T and PX yield P f(A,X); permuting only X changes the graph input"}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(result))
 
 
if __name__ == "__main__":
    main()
