"""First evaluation on Penn's official test videos for an author-derived task."""
 
from __future__ import annotations
 
import argparse
import collections
import hashlib
import json
from pathlib import Path
 
import cv2
import numpy as np
from scipy.io import loadmat
import torch
 
from c33_real_motion_direction import DirectionCNN, evaluate, sha256
from prepare_c33_real_motion import (GAP_FRAMES, MAX_CENTERS_PER_VIDEO,
                                     MIN_FRACTION_WIDTH, RIGHT_WRIST, letterbox)
 
 
ROOT = Path(__file__).resolve().parents[2]
DATA = ROOT / "work/data/penn_action_original"
MOTION = DATA / "motion_gap5_v1"
 
 
def test_triplets(rows: list[dict]) -> tuple[np.ndarray, np.ndarray, list[dict]]:
    images = []
    signs = []
    metadata = []
    for row in rows:
        annotation = loadmat(ROOT / row["label_relative_path"], squeeze_me=True)
        x = np.asarray(annotation["x"], dtype=np.float64)[:, RIGHT_WRIST]
        visible = np.asarray(annotation["visibility"]).astype(bool)[:, RIGHT_WRIST]
        eligible = []
        for center in range(GAP_FRAMES, len(x) - GAP_FRAMES, 2):
            before, after = center - GAP_FRAMES, center + GAP_FRAMES
            if not (visible[before] and visible[center] and visible[after]):
                continue
            d1, d2 = x[center] - x[before], x[after] - x[center]
            if (np.isfinite(d1) and np.isfinite(d2) and d1 * d2 > 0 and
                    min(abs(d1), abs(d2)) >= MIN_FRACTION_WIDTH * row["width"]):
                eligible.append(center)
        if len(eligible) > MAX_CENTERS_PER_VIDEO:
            choices = np.linspace(0, len(eligible) - 1, MAX_CENTERS_PER_VIDEO,
                                  dtype=int)
            eligible = [eligible[i] for i in choices]
        for center in eligible:
            indices = (center - GAP_FRAMES, center, center + GAP_FRAMES)
            triplet = []
            for index in indices:
                path = ROOT / row["frames_relative_dir"] / f"{index + 1:06d}.jpg"
                frame = cv2.imdecode(np.fromfile(path, dtype=np.uint8),
                                     cv2.IMREAD_COLOR)
                if frame is None:
                    raise RuntimeError(f"unreadable official test frame: {path}")
                triplet.append(letterbox(frame))
            images.append(np.stack(triplet))
            sign = int(x[center] > x[indices[0]])
            signs.append(sign)
            metadata.append({"video_id": row["id"], "action": row["action"],
                             "center_frame_index_zero_based": center,
                             "past_to_current_direction": "right" if sign else "left"})
    if not images:
        raise RuntimeError("no eligible official test triplets")
    return np.stack(images), np.asarray(signs, dtype=np.int64), metadata
 
 
def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--run-dir", type=Path, required=True)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    run = args.run_dir if args.run_dir.is_absolute() else ROOT / args.run_dir
    out = args.out if args.out.is_absolute() else ROOT / args.out
    if out.exists():
        raise FileExistsError(f"preserve previous official test: {out}")
    motion_path = MOTION / "manifest.json"
    motion = json.loads(motion_path.read_text(encoding="utf-8"))
    report = json.loads((run / "train.json").read_text(encoding="utf-8"))
    if (report["data_manifest_sha256"] != sha256(motion_path) or
            report["best_checkpoint_sha256"] != sha256(run / "best.pt") or
            report["official_test_accessed"]):
        raise RuntimeError("training/checkpoint identity mismatch")
    checkpoint = torch.load(run / "best.pt", map_location="cpu", weights_only=False)
    if checkpoint["data_manifest_sha256"] != sha256(motion_path):
        raise RuntimeError("checkpoint source mismatch")
    inventory_path = DATA / "inventory.json"
    inventory = json.loads(inventory_path.read_text(encoding="utf-8"))
    if motion["inventory_sha256"] != sha256(inventory_path):
        raise RuntimeError("official inventory changed")
    rows = [row for row in inventory["rows"] if row["official_split"] == "official_test"]
    if len(rows) != 1068:
        raise RuntimeError("official Penn test count changed")
    images, signs, metadata = test_triplets(rows)
    torch.set_num_threads(4)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = DirectionCNN().to(device)
    model.load_state_dict(checkpoint["model"])
    result = evaluate(model, images, signs, 32, device)
    by_video = collections.defaultdict(lambda: [0, 0])
    by_action = collections.defaultdict(lambda: [0, 0])
    details = []
    for i, entry in enumerate(metadata):
        truth = (int(signs[i]), 1 - int(signs[i]))
        guesses = (result["predictions"][2 * i], result["predictions"][2 * i + 1])
        number_correct = sum(a == b for a, b in zip(truth, guesses))
        by_video[entry["video_id"]][0] += number_correct
        by_video[entry["video_id"]][1] += 2
        by_action[entry["action"]][0] += number_correct
        by_action[entry["action"]][1] += 2
        details.append({**entry, "truth_past_pair_and_future_pair": list(truth),
                        "predictions": list(guesses)})
    test_report = {
        "identity": "first Penn official test split use for this author-derived two-frame right-wrist image-plane direction task; NOT an official Penn 15-action benchmark result",
        "source_archive_sha256": inventory["source_archive_sha256"],
        "inventory_sha256": sha256(inventory_path),
        "motion_manifest_sha256": sha256(motion_path),
        "train_report_sha256": sha256(run / "train.json"),
        "checkpoint_sha256": sha256(run / "best.pt"),
        "selected_epoch": checkpoint["epoch"],
        "official_test_video_count": len(rows),
        "eligible_official_test_videos": len(by_video),
        "eligible_triplets": len(metadata),
        "single_current_frame_accuracy_by_exact_pair_symmetry": 0.5,
        "two_frame_model_metrics": {k: v for k, v in result.items() if k != "predictions"},
        "by_action_pair_counts": {key: value for key, value in sorted(by_action.items())},
        "by_video_pair_counts": {key: value for key, value in sorted(by_video.items())},
        "details": details,
        "limits": "x displacement of annotated right wrist in image pixels over five frame indices; excludes invisible/low displacement cases, camera geometry/true 3-D velocity unknown; test results can never be reopened as fresh blind data",
    }
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(test_report, ensure_ascii=False, indent=2) + "\n",
                   encoding="utf-8")
    print(json.dumps({"eligible_triplets": len(metadata),
                      "eligible_videos": len(by_video),
                      "two_frame_accuracy": result["accuracy"],
                      "single_current_theoretical": 0.5}, ensure_ascii=False), flush=True)
 
 
if __name__ == "__main__":
    main()
