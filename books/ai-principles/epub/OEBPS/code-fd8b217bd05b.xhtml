"""One fixed internal test read after true/wrong retrieval validation selection."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import numpy as np
import torch
from torch.nn import functional as F
 
from c35_image_text_retrieval import DATA, PairModel, batch_tokens
 
 
@torch.inference_mode()
def encode(model, images, rows, ids, device):
    model.eval()
    image_vectors, text_vectors = [], []
    for start in range(0, len(ids), 64):
        batch = ids[start:start + 64]
        x = torch.from_numpy(np.asarray(images[batch], dtype=np.float32).copy()).to(device) / 255.0
        tok, lengths = batch_tokens([rows[i]["caption_tokens"][0] for i in batch], device)
        image_vectors.append(F.normalize(model.image(x), dim=-1).cpu())
        text_vectors.append(F.normalize(model.text(tok, lengths), dim=-1).cpu())
    return torch.cat(image_vectors), torch.cat(text_vectors)
 
 
def ranks_for(sim):
    answer = {}
    for name, scores in (("image_to_text", sim), ("text_to_image", sim.T)):
        ranks = ((scores.argsort(dim=1, descending=True) == torch.arange(scores.shape[0])[:, None]).nonzero()[:, 1] + 1).float()
        answer[name] = {"r1": float((ranks <= 1).float().mean()), "r5": float((ranks <= 5).float().mean()), "r10": float((ranks <= 10).float().mean()), "median_rank": float(ranks.median()), "mean_rank": float(ranks.mean())}
    return answer
 
 
def main():
    p = argparse.ArgumentParser()
    p.add_argument("--true-checkpoint", type=Path, default=Path("work/runs/c35_retrieval_true30/best.pt"))
    p.add_argument("--wrong-checkpoint", type=Path, default=Path("work/runs/c35_retrieval_wrong30/best.pt"))
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    if a.out.exists():
        raise RuntimeError("test result exists; do not silently overwrite")
    torch.set_num_threads(8)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    manifest = json.loads((DATA / "manifest.json").read_text(encoding="utf-8"))
    rows = json.loads((DATA / "image_records.json").read_text(encoding="utf-8"))
    images = np.load(DATA / f"images_{manifest['image_side']}_chw_uint8.npy", mmap_mode="r")
    ids = [i for i, row in enumerate(rows) if row["split"] == "test"]
    vocab = json.loads((DATA / "vocab.json").read_text(encoding="utf-8"))
    if len(ids) != 500:
        raise RuntimeError("test size changed")
    results = {}
    for label, ckpt_path in (("true", a.true_checkpoint), ("wrong", a.wrong_checkpoint)):
        ckpt = torch.load(ckpt_path, map_location="cpu", weights_only=True)
        model = PairModel(len(vocab)).to(device)
        model.load_state_dict(ckpt["model"])
        u, v = encode(model, images, rows, ids, device)
        sim = u @ v.T
        diag = sim.diag()
        shifted = sim[torch.arange(len(ids)), (torch.arange(len(ids)) + 1) % len(ids)]
        first = []
        for rowidx in range(3):
            top = int(sim[rowidx].argmax())
            first.append({"image_id": rows[ids[rowidx]]["image_id"], "true_caption": rows[ids[rowidx]]["captions"][0], "top_caption_image_id": rows[ids[top]]["image_id"], "top_caption": rows[ids[top]]["captions"][0], "top_is_same_image": rowidx == top})
        results[label] = {"selected_epoch": ckpt["epoch"], "checkpoint_sha256": hashlib.sha256(ckpt_path.read_bytes()).hexdigest(), "metrics": ranks_for(sim), "mean_cosine_true_pair": float(diag.mean()), "mean_cosine_shifted_text": float(shifted.mean()), "first_three_fixed_examples": first}
    # Chance R@k is k/500; no random model is required to state this combinatorial baseline.
    output = {"test_identity": "author-internal 500 complete images from official COCO2017 validation; one fixed first caption/image", "test_images": len(ids), "candidate_texts": len(ids), "chance_r1": 1 / len(ids), "chance_r5": 5 / len(ids), "test_opened_for_metrics": True, "results": results}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(output, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({k: v["metrics"] for k, v in results.items()}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    main()
