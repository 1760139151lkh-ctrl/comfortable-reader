"""Draw rates and denominators from the already opened C44 test report."""
from __future__ import annotations
 
import argparse
import hashlib
import json
from pathlib import Path
 
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
 
WORK = Path(__file__).resolve().parents[1]
RUNS = WORK / "runs"
 
 
def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
 
 
def main(test_report: Path, out_dir: Path):
    source = test_report.resolve()
    out = out_dir.resolve()
    if not source.is_relative_to(RUNS.resolve()) or not out.is_relative_to(RUNS.resolve()) or out.exists():
        raise ValueError("use a C44 report and new directory inside work/runs")
    d = json.loads(source.read_text(encoding="utf-8"))
    if d["test_rows"] != 16281 or not d["first_chapter_test_after_model_and_threshold_selection"]:
        raise RuntimeError("expected the frozen C44 official-source test")
    names = ["Female", "Male"]  # Original file codes, not a complete social identity taxonomy.
    before = d["global_threshold_0_5"]["by_group"]
    after = d["validation_chosen_group_thresholds"]["by_group"]
    plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "SimHei", "DejaVu Sans"]
    plt.rcParams["axes.unicode_minus"] = False
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    x = np.arange(2)
    w = 0.25
    base = [before[g]["label_rate"] for g in names]
    first = [before[g]["selection_rate"] for g in names]
    second = [after[g]["selection_rate"] for g in names]
    for shift, data, label, color in ((-w, base, "原文件>50K标签率", "#555555"),
                                      (0, first, "同一0.5阈值预测率", "#d48735"),
                                      (w, second, "验证定组阈值预测率", "#29847c")):
        axes[0].bar(x + shift, data, width=w, label=label, color=color)
    axes[0].set_xticks(x, [f"{g}\n(n={before[g]['n']})" for g in names])
    axes[0].set_ylabel("该编码群体中的比例")
    axes[0].set_ylim(0, 0.5)
    axes[0].set_title("标签出现率与预测正类率，分母各自不同")
    axes[0].legend(fontsize=8)
    w2 = 0.19
    bars = [
        (-1.5 * w2, [before[g]["tpr_given_label_positive"] for g in names], "0.5阈值 TPR", "#d48735"),
        (-0.5 * w2, [after[g]["tpr_given_label_positive"] for g in names], "组阈值 TPR", "#29847c"),
        (0.5 * w2, [before[g]["fpr_given_label_negative"] for g in names], "0.5阈值 FPR", "#dbac7e"),
        (1.5 * w2, [after[g]["fpr_given_label_negative"] for g in names], "组阈值 FPR", "#8ac1bc"),
    ]
    for shift, values, label, color in bars:
        axes[1].bar(x + shift, values, width=w2, label=label, color=color)
    axes[1].set_xticks(x, [f"{g}\n(正例{before[g]['actual_positive']})" for g in names])
    axes[1].set_ylim(0, 0.9)
    axes[1].set_ylabel("按真实收入标签分母算的率")
    axes[1].set_title("TPR只在Y=1中算；FPR只在Y=0中算")
    axes[1].legend(fontsize=8)
    for ax in axes:
        ax.grid(axis="y", alpha=0.15)
    fig.tight_layout()
    out.mkdir(parents=True)
    fig.savefig(out / "adult_coded_group_rates.png", dpi=165)
    plt.close(fig)
    receipt = {"source_test_report_sha256": sha(source),
               "source_test_report": str(source.relative_to(WORK)),
               "figure": "adult_coded_group_rates.png",
               "scope": "1994 filtered UCI Adult sex field has only Female/Male codes; bar heights are observed file-row rates, not population-weighted fairness guarantees"}
    (out / "figure_receipt.json").write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"figure": str(out / "adult_coded_group_rates.png"),
                      "source_test_sha256": receipt["source_test_report_sha256"]}, ensure_ascii=False))
 
 
if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--test-report", type=Path, required=True)
    parser.add_argument("--out-dir", type=Path, required=True)
    args = parser.parse_args()
    main(args.test_report, args.out_dir)
